20   function init() {

22       
23       // Scene
24       scene = new THREE.Scene();
25       scene.background = new THREE.Color(0x0a0a14);
26       scene.fog = new THREE.Fog(0x0a0a14, 500, 2000);
27       
28       // Camera
29       camera = new THREE.PerspectiveCamera(
30           75,
31           window.innerWidth / window.innerHeight,
32           0.1,
33           3000
34       );
35       camera.position.set(0, 200, 600);
36

39       renderer.setSize(window.innerWidth, window.innerHeight);

86   function createStarField() {
87       const starGeometry = new THREE.BufferGeometry();
88       const starCount = 5000;
89       const positions = new Float32Array(starCount * 3);
90       
91       for (let i = 0; i < starCount * 3; i++) {
92           positions[i] = (Math.random() - 0.5) * 4000;
93       }
94       
95       starGeometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
96       const starMaterial = new THREE.PointsMaterial({
97           color: 0xffffff,
98           size: 2,
99           transparent: true,
100          opacity: 0.8
101      });
102      
103      const stars = new THREE.Points(starGeometry, starMaterial);
104      scene.add(stars);
105  }

210  function createSpaceObject(data) {
211      const group = new THREE.Group();
212      group.position.set(...data.position);
213      
214      // Sphere
215      const geometry = new THREE.SphereGeometry(data.size, 32, 32);
216      const material = new THREE.MeshPhongMaterial({
217          color: data.color,
218          emissive: data.color,
219          emissiveIntensity: 0.2,
220          shininess: 100
221      });
222      const sphere = new THREE.Mesh(geometry, material);
223      sphere.castShadow = true;
224      group.add(sphere);
225      
226      // Glow
227      const glowGeometry = new THREE.SphereGeometry(data.size * 1.2, 32, 32);
228      const glowMaterial = new THREE.MeshBasicMaterial({
229          color: data.color,
230          transparent: true,
231          opacity: 0.2
232      });
233      const glow = new THREE.Mesh(glowGeometry, glowMaterial);
234      group.add(glow);
235      
236      // Ring
237      const ringGeometry = new THREE.TorusGeometry(data.size * 1.5, 2, 16, 100);
238      const ringMaterial = new THREE.MeshBasicMaterial({
239          color: data.color,
240          transparent: true,
241          opacity: 0.4
242      });
243      const ring = new THREE.Mesh(ringGeometry, ringMaterial);
244      ring.rotation.x = Math.PI / 2;
245      group.add(ring);
246      
247      // Label
248      const canvas = document.createElement('canvas');
249      const context = canvas.getContext('2d');
250      canvas.width = 256;
251      canvas.height = 64;
252      context.fillStyle = 'white';
253      context.font = 'Bold 24px Arial';
254      context.textAlign = 'center';
255      context.fillText(data.name, 128, 40);
256      
257      const texture = new THREE.CanvasTexture(canvas);
258      const spriteMaterial = new THREE.SpriteMaterial({ map: texture });
259      const sprite = new THREE.Sprite(spriteMaterial);
260      sprite.position.y = data.size + 30;
261      sprite.scale.set(100, 25, 1);
262      group.add(sprite);
263      
264      return {
265          group: group,
266          sphere: sphere,
267          glow: glow,
268          ring: ring,
269          data: data,
270          rotation: Math.random() * Math.PI * 2
271      };
272  }

274  function createConnections() {
275      const lineMaterial = new THREE.LineBasicMaterial({
276          color: 0x4facfe,
277          transparent: true,
278          opacity: 0.3
279      });
280      
281      for (let i = 0; i < spaceObjects.length; i++) {
282          for (let j = i + 1; j < spaceObjects.length; j++) {
283              const obj1 = spaceObjects[i];
284              const obj2 = spaceObjects[j];
285              
286              const distance = obj1.group.position.distanceTo(obj2.group.position);
287              
288              if (distance < 400) {
289                  const points = [obj1.group.position, obj2.group.position];
290                  const geometry = new THREE.BufferGeometry().setFromPoints(points);
291                  const line = new THREE.Line(geometry, lineMaterial);
292                  scene.add(line);
293              }
294          }
295      }
296  }

298  function setupMouseInteraction() {
299      const raycaster = new THREE.Raycaster();
300      const mouse = new THREE.Vector2();
301      
302      renderer.domElement.addEventListener('click', (event) => {
303          mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
304          mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;
305          
306          raycaster.setFromCamera(mouse, camera);
307          
308          const spheres = spaceObjects.map(obj => obj.sphere);
309          const intersects = raycaster.intersectObjects(spheres);
310          
311          if (intersects.length > 0) {
312              const intersectedObject = spaceObjects.find(
313                  obj => obj.sphere === intersects[0].object
314              );
315              selectObject(intersectedObject);
316          }
317      });
318  }

320  function selectObject(obj) {
321      if (selectedObject) {
322          selectedObject.sphere.material.emissiveIntensity = 0.2;
323          selectedObject.glow.material.opacity = 0.2;
324      }
325      
326      selectedObject = obj;
327      
328      if (obj) {
329          obj.sphere.material.emissiveIntensity = 0.6;
330          obj.glow.material.opacity = 0.5;
331          
332          updateInfoPanel(obj);
333          animateCameraToObject(obj);
334      }
335  }

358  function animateCameraToObject(obj) {
359      const targetPosition = obj.group.position.clone();
360      targetPosition.z += 200;
361      targetPosition.y += 100;
362      
363      const startPosition = camera.position.clone();
364      const duration = 1000;
365      const startTime = Date.now();
366      
367      function animateCamera() {
368          const elapsed = Date.now() - startTime;
369          const progress = Math.min(elapsed / duration, 1);
370          
371          camera.position.lerpVectors(startPosition, targetPosition, progress);
372          controls.target.lerp(obj.group.position, progress);
373          
374          if (progress < 1) {
375              requestAnimationFrame(animateCamera);
376          }
377      }
378      
379      animateCamera();
380  }

382  function animate() {
383      requestAnimationFrame(animate);
384      
385      spaceObjects.forEach(obj => {
386          obj.rotation += 0.005;
387          obj.ring.rotation.z = obj.rotation;
388          obj.glow.rotation.y += 0.01;
389          
390          const scale = 1 + Math.sin(Date.now() * 0.002 + obj.rotation) * 0.05;
391          obj.glow.scale.set(scale, scale, scale);
392      });
393      
394      if (autoRotate) {
395          camera.position.x = Math.cos(Date.now() * 0.0002) * 600;
396          camera.position.z = Math.sin(Date.now() * 0.0002) * 600;
397          camera.lookAt(0, 0, 0);
398      }
399      
400      controls.update();
401      updateFPS();
402      renderer.render(scene, camera);
403  }

408  function updateFPS() {
409      frames++;
410      const currentTime = Date.now();
411      
412      if (currentTime > lastTime + 1000) {
413          document.getElementById('fps').textContent = frames;
414          frames = 0;
415          lastTime = currentTime;
416      }

465  async function fetchLiveData() {
466      for (const obj of spaceObjects) {
467          if (obj.data.endpoint) {
468              try {
469                  const response = await fetch(`${API_BASE}${obj.data.endpoint}`);
470                  if (response.ok) {
471                      const data = await response.json();
472                      updateObjectData(obj, data);
473                  }
474              } catch (error) {
475                  console.warn(`⚠️ Could not fetch data for ${obj.data.name}`);
476              }
477          }
478      }
479  }

481  function updateObjectData(obj, data) {
482      const id = obj.data.id;
483      
484      if (id === 'robot-fleet' && data.totalRobots !== undefined) {
485          obj.data.metadata.robots = data.totalRobots.toString();
486          obj.data.metadata.missions = (data.deployedRobots || 0).toString();
487          obj.data.metadata.battery = '78%';
488      } else if (id === 'drone-swarm' && data.totalDrones !== undefined) {
489          obj.data.metadata.drones = data.totalDrones.toString();
490          obj.data.metadata.inFlight = (data.availableCount || 0).toString();
491      } else if (id === 'finance' && data.totalBalance !== undefined) {
492          obj.data.metadata.balance = `€${(data.totalBalance / 1000).toFixed(1)}K`;
493          obj.data.metadata.wallets = (data.walletsCount || 0).toString();
494      } else if (id === 'infrastructure' && data.totalInstances !== undefined) {
495          obj.data.metadata.instances = data.totalInstances.toString();
496          obj.data.metadata.cost = `€${(data.totalMonthlyCost || 0).toFixed(0)}`;
497      } else if (id === 'blueprints' && data.totalBlueprints !== undefined) {
498          obj.data.metadata.total = data.totalBlueprints.toString();
499          obj.data.metadata.production = (data.productionCount || 0).toString();
500      }
501      
502      if (selectedObject === obj) {
503          updateInfoPanel(obj);
504      }
505  }
