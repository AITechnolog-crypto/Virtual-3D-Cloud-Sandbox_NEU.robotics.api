575      // 3D DRONES ANIMATION
576      // ========================================
577      function init3DDrones() {
578          const canvas = document.getElementById('droneCanvas');
579          const ctx = canvas.getContext('2d');
580          canvas.width = canvas.offsetWidth;
581          canvas.height = canvas.offsetHeight;
582  
583          // Generate drones
584          for (let i = 0; i < 12; i++) {
585              drones.push({
586                  x: Math.random() * canvas.width,
587                  y: Math.random() * canvas.height,
588                  vx: (Math.random() - 0.5) * 2,
589                  vy: (Math.random() - 0.5) * 2,
590                  size: Math.random() * 4 + 2
591              });
592          }
593  
594          function animate() {
595              ctx.fillStyle = 'rgba(10, 14, 39, 0.1)';
596              ctx.fillRect(0, 0, canvas.width, canvas.height);
597  
598              drones.forEach(drone => {
599                  // Update position
600                  drone.x += drone.vx;
601                  drone.y += drone.vy;
602  
603                  // Bounce off walls
604                  if (drone.x < 0 || drone.x > canvas.width) drone.vx *= -1;
605                  if (drone.y < 0 || drone.y > canvas.height) drone.vy *= -1;
606  
607                  // Draw drone
608                  ctx.fillStyle = '#00ff88';
609                  ctx.beginPath();
610                  ctx.arc(drone.x, drone.y, drone.size, 0, Math.PI * 2);
611                  ctx.fill();
612  
613                  // Draw trail
614                  ctx.strokeStyle = 'rgba(0, 255, 136, 0.3)';
615                  ctx.lineWidth = 1;
616                  ctx.beginPath();
617                  ctx.moveTo(drone.x, drone.y);
618                  ctx.lineTo(drone.x - drone.vx * 5, drone.y - drone.vy * 5);
619                  ctx.stroke();
620              });
621  
622              animationFrame = requestAnimationFrame(animate);
623          }
624  
625          animate();
626      }
