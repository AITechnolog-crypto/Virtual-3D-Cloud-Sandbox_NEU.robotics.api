25       /**
26        * Create a new wallet
27        */
28       @Transactional
29       public Wallet createWallet(Wallet wallet) {
30           System.out.println("Creating new wallet: " + wallet.getWalletName());
31           
32           if (walletRepository.existsByWalletName(wallet.getWalletName())) {
33               throw new RuntimeException("Wallet with name '" + wallet.getWalletName() + "' already exists");
34           }
35           
36           return walletRepository.save(wallet);
37       }
38       
39       /**
40        * Get wallet by ID
41        */
42       public Wallet getWallet(Long id) {
43           return walletRepository.findById(id)
44               .orElseThrow(() -> new RuntimeException("Wallet not found with id: " + id));
45       }

48        * Get all wallets
49        */
50       public List<Wallet> getAllWallets() {
51           return walletRepository.findAll();
52       }

55        * Get wallets by type

61       /**
62        * Add transaction to wallet
63        */
64       @Transactional
65       public Transaction addTransaction(Long walletId, Transaction transaction) {
66           System.out.println("Adding transaction to wallet " + walletId + ": " + 
67                            transaction.getAmount() + " " + transaction.getTransactionType());
68           
69           Wallet wallet = getWallet(walletId);
70           transaction.setWallet(wallet);
71           
72           // Update wallet balance
73           BigDecimal amount = transaction.getAmount();
74           switch (transaction.getTransactionType()) {
75               case "INCOME":
76                   wallet.setBalance(wallet.getBalance().add(amount));
77                   break;
78               case "EXPENSE":
79                   wallet.setBalance(wallet.getBalance().subtract(amount));
80                   break;
81               case "CHARITY":
82                   wallet.setBalance(wallet.getBalance().subtract(amount));
83                   break;
84               default:
85                   throw new RuntimeException("Unknown transaction type: " + transaction.getTransactionType());
86           }
87           
88           walletRepository.save(wallet);
89           return transactionRepository.save(transaction);
90       }

92       /**
93        * Transfer money between wallets
94        */
95       @Transactional
96       public void transferBetweenWallets(Long fromWalletId, Long toWalletId, 
97                                           BigDecimal amount, String description) {
98           System.out.println("Transferring " + amount + " from wallet " + fromWalletId + " to wallet " + toWalletId);
99           
100          Wallet fromWallet = getWallet(fromWalletId);
101          Wallet toWallet = getWallet(toWalletId);
102          
103          if (fromWallet.getBalance().compareTo(amount) < 0) {
104              throw new RuntimeException("Insufficient balance in source wallet");
105          }
106          
107          // Deduct from source wallet
108          Transaction outgoingTransaction = new Transaction();
109          outgoingTransaction.setWallet(fromWallet);
110          outgoingTransaction.setTransactionType("TRANSFER");
111          outgoingTransaction.setAmount(amount.negate());
112          outgoingTransaction.setCategory("INTERNAL_TRANSFER");
113          outgoingTransaction.setDescription("Transfer to " + toWallet.getWalletName() + ": " + description);
114          outgoingTransaction.setTransactionDate(LocalDateTime.now());
115          
116          // Add to destination wallet
117          Transaction incomingTransaction = new Transaction();
118          incomingTransaction.setWallet(toWallet);
119          incomingTransaction.setTransactionType("TRANSFER");
120          incomingTransaction.setAmount(amount);
121          incomingTransaction.setCategory("INTERNAL_TRANSFER");
122          incomingTransaction.setDescription("Transfer from " + fromWallet.getWalletName() + ": " + description);
123          incomingTransaction.setTransactionDate(LocalDateTime.now());
124          
125          // Update balances
126          fromWallet.setBalance(fromWallet.getBalance().subtract(amount));
127          toWallet.setBalance(toWallet.getBalance().add(amount));
128          
129          walletRepository.save(fromWallet);
130          walletRepository.save(toWallet);
131          transactionRepository.save(outgoingTransaction);
132          transactionRepository.save(incomingTransaction);
133      }

165      /**
166       * Get financial overview
167       */
168      public Map<String, Object> getFinancialOverview() {
169          Map<String, Object> overview = new HashMap<>();
170          
171          List<Wallet> allWallets = walletRepository.findAll();
172          BigDecimal totalBalance = allWallets.stream()
173              .filter(w -> w.getCurrency().equals("EUR"))
174              .map(Wallet::getBalance)
175              .reduce(BigDecimal.ZERO, BigDecimal::add);
176          
177          overview.put("totalBalance", totalBalance);
178          overview.put("walletsCount", allWallets.size());
179          overview.put("wallets", allWallets);
180          
181          List<Object[]> walletStats = walletRepository.getWalletStatistics();
182          overview.put("walletStatistics", walletStats);
183          
184          List<Object[]> transactionStats = transactionRepository.getTransactionStatistics();
185          overview.put("transactionStatistics", transactionStats);
186          
187          LocalDateTime lastMonth = LocalDateTime.now().minusMonths(1);
188          List<Transaction> recentTransactions = transactionRepository.findRecentTransactions(lastMonth);
189          overview.put("recentTransactions", recentTransactions);
190          
191          return overview;
192      }
