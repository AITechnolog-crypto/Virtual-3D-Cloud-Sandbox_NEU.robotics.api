570          // Check system status
571          async function checkSystemStatus() {
572              const statusDot = document.getElementById('status-dot');
573              const statusText = document.getElementById('status-text');
574              
575              try {
576                  const controller = new AbortController();
577                  const timeoutId = setTimeout(() => controller.abort(), 5000);
578                  
579                  const response = await fetch(`${API_BASE}/drones/count`, { 
580                      signal: controller.signal 
581                  });
582                  
583                  clearTimeout(timeoutId);
584                  
585                  if (response.ok) {
586                      statusDot.className = 'status-dot status-online';
587                      statusText.textContent = 'System Online';
588                  } else {
589                      statusDot.className = 'status-dot status-warning';
590                      statusText.textContent = 'System Warnung';
591                  }
592              } catch (error) {
593                  statusDot.className = 'status-dot status-offline';
594                  statusText.textContent = 'System Offline';
595                  console.warn('System Status Check failed:', error);
596              }
597          }

652              
653              try {
654                  const response = await fetch(`${API_BASE}/drones`);
655                  if (!response.ok) throw new Error(`HTTP ${response.status}`);

665                  drones.forEach((drone, index) => {
666                      const droneItem = document.createElement('div');
667                      droneItem.className = 'drone-item';
668                      droneItem.style.animationDelay = `${index * 0.05}s`;
669                      
670                      const statusColor = getStatusColor(drone.status);
671                      
672                      droneItem.innerHTML = `
673                          <div class="drone-info"> 
674                              <span class="drone-id">🚁 Drohne #${drone.id || 'N/A'}</span> 
675                              <span class="drone-status"> 
676                                  Status: <strong style="color: ${statusColor}">${drone.status || 'Aktiv'}</strong> |  
677                                  Typ: ${drone.type || 'Standard'}
678                                  ${drone.battery ? ` | Batterie: ${drone.battery}%` : ''}
679                              </span> 
680                          </div> 
681                          <span style="color: ${statusColor}; font-size: 1.5rem;">●</span> 
682                      `;
683                      droneList.appendChild(droneItem);
684                  });
685              } catch (error) {
686                  console.error('Fehler beim Laden der Drohnen-Liste:', error);
687                  droneList.innerHTML = 
688                      '<p style="text-align: center; padding: 2rem; color: var(--danger);">❌ Fehler beim Laden der Daten</p>';
689              } finally {
690                  updateProgress();
691              }

731          // Add new drone with validation
732          async function addNewDrone() {
733              const droneName = prompt('Drohnen-Name eingeben:');
734              if (!droneName || droneName.trim() === '') {
735                  showToast('Ungültiger Name', 'warning');
736                  return;
737              }
738  
739              showToast('Drohne wird hinzugefügt...', 'info');
740  
741              try {
742                  const response = await fetch(`${API_BASE}/drones`, {
743                      method: 'POST',
744                      headers: { 'Content-Type': 'application/json' },
745                      body: JSON.stringify({ 
746                          name: droneName.trim(),
747                          status: 'Aktiv',
748                          type: 'Standard'
749                      })
750                  });
751  
752                  if (response.ok) {
753                      showToast('✅ Drohne erfolgreich hinzugefügt!', 'success');
754                      await loadAllData();
755                  } else {
