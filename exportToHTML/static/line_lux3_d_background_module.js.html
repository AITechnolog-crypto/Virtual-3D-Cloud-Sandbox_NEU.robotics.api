2    // LineLux 3D Background Module
3    // Usage: import './linelux-3d.js' or <script src="linelux-3d.js"></script>
4    
5    (function() {
6        'use strict';
7    
8        function init3DBackground() {
9            const canvas = document.getElementById('bg-canvas');
10           if (!canvas || !window.THREE) {
11               console.warn('⚠️ Canvas or THREE.js not found');
12               return;
13           }
14   
15           const scene = new THREE.Scene();
16           scene.fog = new THREE.Fog(0x000000, 1, 1000);
17   
18           const camera = new THREE.PerspectiveCamera(
19               75,
20               window.innerWidth / window.innerHeight,
21               0.1,
22               1000
23           );
24           camera.position.z = 400;
25   
26           const renderer = new THREE.WebGLRenderer({
27               canvas: canvas,
28               antialias: true,
29               alpha: true
30           });
31           renderer.setSize(window.innerWidth, window.innerHeight);
32           renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
33   
34           // Particles
35           const particleCount = 1000;
36           const geometry = new THREE.BufferGeometry();
37           const vertices = [];
38           const particles = [];
39   
40           for (let i = 0; i < particleCount; i++) {
41               const x = (Math.random() - 0.5) * 1000;
42               const y = (Math.random() - 0.5) * 1000;
43               const z = (Math.random() - 0.5) * 1000;
44   
45               vertices.push(x, y, z);
46               particles.push({
47                   x, y, z,
48                   vx: (Math.random() - 0.5) * 0.5,
49                   vy: (Math.random() - 0.5) * 0.5,
50                   vz: (Math.random() - 0.5) * 0.5
51               });
52           }
53   
54           geometry.setAttribute('position', new THREE.Float32BufferAttribute(vertices, 3));
55   
56           const material = new THREE.PointsMaterial({
57               color: 0x00ff88,
58               size: 2,
59               transparent: true,
60               opacity: 0.8
61           });
62   
63           const points = new THREE.Points(geometry, material);
64           scene.add(points);
65   
66           // Animation
67           function animate() {
68               requestAnimationFrame(animate);
69               scene.rotation.y += 0.001;
70   
71               const positions = points.geometry.attributes.position.array;
72               for (let i = 0; i < particles.length; i++) {
73                   particles[i].x += particles[i].vx;
74                   particles[i].y += particles[i].vy;
75                   particles[i].z += particles[i].vz;
76   
77                   if (Math.abs(particles[i].x) > 500) particles[i].vx *= -1;
78                   if (Math.abs(particles[i].y) > 500) particles[i].vy *= -1;
79                   if (Math.abs(particles[i].z) > 500) particles[i].vz *= -1;
80   
81                   positions[i * 3] = particles[i].x;
82                   positions[i * 3 + 1] = particles[i].y;
83                   positions[i * 3 + 2] = particles[i].z;
84               }
85   
86               points.geometry.attributes.position.needsUpdate = true;
87               renderer.render(scene, camera);
88           }
89   
90           // Resize handler
91           window.addEventListener('resize', () => {
92               camera.aspect = window.innerWidth / window.innerHeight;
93               camera.updateProjectionMatrix();
94               renderer.setSize(window.innerWidth, window.innerHeight);
95           });
96   
97           animate();
98       }
99   
100      // Auto-initialize when DOM ready
101      if (document.readyState === 'loading') {
102          document.addEventListener('DOMContentLoaded', init3DBackground);
103      } else {
104          init3DBackground();
105      }
106  
107      // Export for manual initialization
108      window.LineLux = { init3D: init3DBackground };
109  })();
