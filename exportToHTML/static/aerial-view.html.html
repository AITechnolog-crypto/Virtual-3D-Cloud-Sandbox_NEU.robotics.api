103  (function(){
104    const API = location.origin + '/api/aerial/lookup';
105    const q = new URLSearchParams(location.search);
106    const elVal = document.getElementById('inputVal');
107    const elFmt = document.getElementById('format');
108    const elOri = document.getElementById('orientation');
109    const elMsg = document.getElementById('msg');
110    const v = document.getElementById('vid');
111    const i = document.getElementById('img');
112  
113    // init from query
114    if (q.get('address')) elVal.value = q.get('address');
115    if (q.get('videoId')) elVal.value = q.get('videoId');
116    if (q.get('format')) elFmt.value = q.get('format');
117    if (q.get('orientation')) elOri.value = q.get('orientation');
118  
119    async function fetchMedia(){
120      try{
121        elMsg.textContent = 'Lade…';
122        const qp = new URLSearchParams();
123        const val = (elVal.value||'').trim() || '1600 Amphitheatre Parkway, Mountain View, CA';
124        if (/^[0-9a-zA-Z\-_]{22}$/.test(val)) qp.set('videoId', val); else qp.set('address', val);
125        qp.set('format', elFmt.value);
126        qp.set('orientation', elOri.value);
127        const r = await fetch(API + '?' + qp.toString(), { cache: 'no-store' });
128        const j = await r.json();
129        if (!r.ok){ elMsg.textContent = '❌ Fehler: ' + (j && j.error || ('HTTP '+r.status)); return; }
130        const key = (elFmt.value||'mp4_medium').toUpperCase();
131        const pack = j.uris && j.uris[key];
132        if (!pack){ elMsg.textContent = '❌ Keine URIs gefunden.'; return; }
133        const src = elOri.value === 'portrait' ? (pack.portraitUri||pack.landscapeUri) : (pack.landscapeUri||pack.portraitUri);
134        if (elFmt.value === 'image'){
135          v.style.display='none'; i.style.display='block'; i.src = src; elMsg.textContent = 'Bild geladen.';
136        } else {
137          i.style.display='none'; v.style.display='block'; v.src = src; try{ await v.play(); }catch(_){} elMsg.textContent = 'Video bereit.';
138        }
139      }catch(e){ elMsg.textContent = '❌ Fehler: ' + (e.message||e); }
140    }
141  
142    document.getElementById('btnFetch').addEventListener('click', fetchMedia);
143    document.getElementById('btnPlayPause').addEventListener('click', ()=>{ try{ if (v.style.display!=='none'){ v.paused? v.play(): v.pause(); } }catch(_){} });
144  
145    // auto

147  })();
