45   <script>
46   (function(){
47     const frameAR  = document.getElementById('frameAR');
48     const frameMap = document.getElementById('frameMap');
49     const logEl    = document.getElementById('log');
50   
51     let lastMapTarget = null;   // {lat,lng}
52     let lastARPosition = null;  // {lat,lng,acc}
53   
54     function log(msg){ logEl.textContent = String(msg) }
55   
56     // Nachrichten von Frames entgegennehmen und puffern
57     window.addEventListener('message', ev => {
58       const m = ev.data || {};
59       // Von map.html: Zielpunkt gewählt
60       if (m.type === 'map:setTarget'){
61         lastMapTarget = {lat:m.lat, lng:m.lng};
62         try { frameAR.contentWindow.postMessage({type:'hub:setTarget', lat:m.lat, lng:m.lng}, '*'); } catch(_){}
63         log('Ziel von Karte empfangen und an AR gesendet');
64       }
65       // Von ar.html: aktuelle Position oder Status
66       if (m.type === 'ar:position'){
67         lastARPosition = {lat:m.lat, lng:m.lng, acc:m.acc||null};
68       }
69       if (m.type === 'ar:status'){
70         log('AR Status ' + m.msg);
71       }
72     });
73   
74     document.getElementById('btnSyncMapToAR').onclick = ()=>{
75       if (!lastMapTarget){ log('Kein Kartenziel gesetzt'); return; }
76       try { frameAR.contentWindow.postMessage({type:'hub:setTarget', ...lastMapTarget}, '*'); } catch(_){}
77       log('Ziel erneut an AR übergeben');
78     };
79   
80     document.getElementById('btnSyncARToMap').onclick = ()=>{
81       if (!lastARPosition){ log('Keine AR Positionsdaten'); return; }
82       try { frameMap.contentWindow.postMessage({type:'hub:centerOn', lat:lastARPosition.lat, lng:lastARPosition.lng}, '*'); } catch(_){}
83       log('AR Position an Karte gesendet');
84     };
85   
86     // Layoutumschalter
87     const root = document.getElementById('layoutRoot');
88     document.getElementById('layout').onchange = (e)=>{
89       const v = e.target.value;
90       if (v === 'split'){
91         root.style.gridTemplateColumns = '1fr 1fr';
92         root.style.gridTemplateRows = 'auto';
93       } else {
94         root.style.gridTemplateColumns = '1fr';
95         root.style.gridTemplateRows = '1fr 1fr';
96       }
97     };
98   })();
99   </script>

104  // Vision toggle, logging, dump, and FPS meter
105  (function(){
106    const frameVision = document.getElementById('frameVision');
107    const btnVision = document.getElementById('btnVision');
108    const btnLog = document.getElementById('btnLog');
109    const btnDump = document.getElementById('btnDump');
110    const pf = document.getElementById('pf');
111  
112    let lastVision = null;
113    let logging = false;
114    let logBuf = [];
115    let lastSensorTs = Date.now();
116  
117    // Toggle vision iframe visibility
118    if (btnVision){
119      btnVision.onclick = ()=>{
120        if (!frameVision) return;
121        const isHidden = frameVision.style.display === 'none' || !frameVision.style.display;
122        frameVision.style.display = isHidden ? 'block' : 'none';
123      };
124    }
125  
126    // Logging controls
127    if (btnLog){
128      btnLog.onclick = ()=>{
129        logging = !logging;
130        logBuf.length = 0;
131        btnLog.textContent = logging ? 'Log aus' : 'Log an';
132      };
133    }
134    if (btnDump){
135      btnDump.onclick = ()=>{
136        try{
137          const blob = new Blob([JSON.stringify(logBuf)], { type:'application/json' });
138          const a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download = 'telemetry.json'; a.click(); URL.revokeObjectURL(a.href);
139        }catch(e){ alert('Dump fehlgeschlagen'); }
140      };
141    }
142  
143    // Message handling (vision + optional sensors/learn stats)
144    window.addEventListener('message', (ev)=>{
145      const m = ev.data || {};
146      if (m && m.type === 'vision:blob'){
147        lastVision = m.payload || m; // tolerate {payload:{...}} or flat
148      }
149      if (m && m.type === 'hal:sensors'){
150        lastSensorTs = Date.now();
151      }
152      if (logging && m && (m.type === 'hal:sensors' || m.type === 'learn:stats' || m.type === 'vision:blob')){
153        logBuf.push({ t: Date.now(), type: m.type, payload: m.payload ?? m });
154        if (logBuf.length > 10000) logBuf.shift();
155      }
156    });
157  
158    // Watchdog (no-op here if no sim frame exists)
159    setInterval(()=>{
160      if (Date.now() - lastSensorTs > 1500){
161        // If a simulation iframe/window is available as simWin, you could reset:
162        // try { bus.post(simWin, 'sim:reset', {}); } catch(_){}
163        lastSensorTs = Date.now();
164      }
165    }, 800);
166  
167    // Simple FPS meter (hub-level render loop)
168    try{
169      let fCnt = 0; let t0 = performance.now();
170      (function loop(){
171        fCnt++;
172        const now = performance.now();
173        if (now - t0 > 1000){
174          if (pf) pf.textContent = 'fps ' + fCnt;
175          fCnt = 0; t0 = now;
176        }
177        requestAnimationFrame(loop);
178      })();
179    }catch(_){}
180  })();
181  </script>
