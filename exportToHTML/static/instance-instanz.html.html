178  function initRadar(){
179    try{
180      const cv = document.getElementById('radar'); if(!cv) return;
181      const ctx = cv.getContext('2d');
182      let pxr = Math.min(window.devicePixelRatio||1, 2);
183      function resize(){ const w=cv.clientWidth, h=cv.clientHeight; cv.width=Math.max(1,Math.floor(w*pxr)); cv.height=Math.max(1,Math.floor(h*pxr)); ctx.setTransform(pxr,0,0,pxr,0,0); }
184      window.addEventListener('resize', resize); resize();
185      function center(){ try{ if (window.gmap){ const c=gmap.getCenter(); return {lat:c.lat(), lng:c.lng()}; } if (window.leaflet){ const c=leaflet.getCenter(); return {lat:c.lat, lng:c.lng}; } }catch(_){} return {lat:45.81,lng:15.98}; }
186      function project(lat,lng){ const c=center(); const dlat = lat - c.lat; const dlng = (lng - c.lng) * Math.cos(c.lat*Math.PI/180); return { x: dlng, y: -dlat }; }
187      let angle = 0; const SCALE_DEG = 2.0; // 2° bis Rand
188      function draw(){
189        const w=cv.clientWidth, h=cv.clientHeight; const cx=w/2, cy=h/2; const r=Math.min(w,h)*0.46;
190        // Fade Hintergrund
191        ctx.fillStyle='#000'; ctx.globalAlpha=0.18; ctx.fillRect(0,0,w,h); ctx.globalAlpha=1;
192        // Grid
193        ctx.strokeStyle='rgba(168,255,223,.12)'; ctx.lineWidth=1;
194        [0.25,0.5,0.75,1].forEach(k=>{ ctx.beginPath(); ctx.arc(cx,cy,r*k,0,Math.PI*2); ctx.stroke(); });
195        for(let a=0;a<360;a+=30){ const rad=a*Math.PI/180; ctx.beginPath(); ctx.moveTo(cx,cy); ctx.lineTo(cx+Math.cos(rad)*r, cy+Math.sin(rad)*r); ctx.stroke(); }
196        // Sweep
197        const sweepAng = angle; const sweepRad = sweepAng*Math.PI/180;
198        const grad = ctx.createRadialGradient(cx,cy,0,cx,cy,r);
199        grad.addColorStop(0,'rgba(0,255,136,.18)'); grad.addColorStop(1,'rgba(0,255,136,0)');
200        ctx.save(); ctx.translate(cx,cy); ctx.rotate(sweepRad); ctx.beginPath(); ctx.moveTo(0,0); ctx.arc(0,0,r,0,Math.PI/36); ctx.closePath(); ctx.fillStyle=grad; ctx.fill(); ctx.restore();
201        function blip(lat,lng,color){ const c=center(); const dlat=lat-c.lat; const dlng=(lng-c.lng)*Math.cos(c.lat*Math.PI/180); const m=Math.hypot(dlng/SCALE_DEG,dlat/SCALE_DEG); const rr = Math.min(1,m); const x = (dlng/SCALE_DEG)/ (m||1) * rr * r; const y = (-dlat/SCALE_DEG)/ (m||1) * rr * r; const bx=cx+x, by=cy+y; const bearing=(Math.atan2(y,x)*180/Math.PI+360)%360; const diff=Math.min(Math.abs(bearing-sweepAng),360-Math.abs(bearing-sweepAng)); const pulse = diff<6; ctx.fillStyle=color; ctx.beginPath(); ctx.arc(bx,by,pulse?3.8:2.4,0,Math.PI*2); ctx.fill(); if(pulse){ ctx.strokeStyle=color; ctx.globalAlpha=0.6; ctx.beginPath(); ctx.arc(bx,by,8,0,Math.PI*2); ctx.stroke(); ctx.globalAlpha=1; } }
202        try{ (window.drones||[]).forEach(d=> blip(d.lat,d.lng,'#66e0ff')); }catch(_){ }
203        try{ (window.grylls||[]).forEach(g=> blip(g.lat,g.lng,'#a8ffdf')); }catch(_){ }
204        // Center
205        ctx.fillStyle='#0f0'; ctx.globalAlpha=0.7; ctx.beginPath(); ctx.arc(cx,cy,2.2,0,Math.PI*2); ctx.fill(); ctx.globalAlpha=1;
206        angle = (angle + 1.2) % 360; requestAnimationFrame(draw);
207      }
208      draw();
209    }catch(e){ /* no-op */ }
210  }
