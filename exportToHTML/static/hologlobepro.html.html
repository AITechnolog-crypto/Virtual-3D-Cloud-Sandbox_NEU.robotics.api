375  (function(){
376    const pad = document.getElementById('pad');
377    let down=false,lastA=null,acc=0;
378    const getA = (e)=>{ const r=pad.getBoundingClientRect(), cx=r.left+r.width/2, cy=r.top+r.height/2;
379      const x=(e.touches?e.touches[0].clientX:e.clientX)-cx, y=(e.touches?e.touches[0].clientY:e.clientY)-cy;
380      return Math.atan2(y,x);
381    };
382    const onDown = e=>{down=true; lastA=getA(e); acc=0; pad.setPointerCapture?.(e.pointerId);}
383    const onMove = e=>{
384      if(!down) return; const a=getA(e); if(lastA==null){lastA=a;return;}
385      let d=a-lastA; if(d>Math.PI) d-=2*Math.PI; if(d<-Math.PI) d+=2*Math.PI; acc += d; lastA=a;
386      if(Math.abs(acc) > 2*Math.PI){ // eine volle Umdrehung
387        ritual.wind(6); acc=0;
388      }
389    };
390    const onUp = ()=>{down=false; lastA=null;};
391    pad.addEventListener('pointerdown',onDown); pad.addEventListener('pointermove',onMove); pad.addEventListener('pointerup',onUp);
392    pad.addEventListener('touchstart',onDown,{passive:true}); pad.addEventListener('touchmove',onMove,{passive:true}); pad.addEventListener('touchend',onUp);

401    let need = watts / 25; // scale
402    const pool = state.energy.solar + state.energy.plasma;
403    if(pool < need){ state.vcpu.on=false; ghz=state.vcpu.base*0.7; state.vcpu.load=Math.min(state.vcpu.load,0.6); }
404    // „spend“
405    const ratio = pool? state.energy.solar/pool : 0.5;
406    state.energy.solar = clamp(state.energy.solar - need*ratio, 0, 100);
407    state.energy.plasma= clamp(state.energy.plasma - need*(1-ratio), 0, 100);
408    // Temp & Ops
409    state.vcpu.temp = clamp(state.vcpu.temp + (watts*0.02) - (state.energy.lev*0.01), 28, 95);
410    if(state.vcpu.temp>82){ state.vcpu.on=false; state.vcpu.load=Math.min(state.vcpu.load,0.7); }
411    state.vcpu.ops = state.vcpu.cores*ghz*1000*state.vcpu.load;
412    ui.sync();
413  }, 250);
