20   <script>
21   (function(){
22     const map = L.map('map', {zoomControl:true}).setView([48.137, 11.575], 5);
23     const tiles = L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png',{
24       maxZoom: 19,
25       attribution: '&copy; OpenStreetMap'
26     }).addTo(map);
27   
28     let marker = null;
29   
30     function setTarget(lat,lng){
31       if (!marker){ marker = L.marker([lat,lng]).addTo(map); }
32       marker.setLatLng([lat,lng]);
33       map.setView([lat,lng], Math.max(12, map.getZoom()));
34       try { window.parent.postMessage({type:'map:setTarget', lat, lng}, '*'); } catch(_){ }
35     }
36   
37     // Klick setzt Ziel und informiert Hub
38     map.on('click', e=> setTarget(e.latlng.lat, e.latlng.lng));
39   
40     // Hub kann die Karte zentrieren
41     window.addEventListener('message', ev=>{
42       const m = ev.data || {};
43       if (m.type === 'hub:centerOn'){
44         map.setView([m.lat, m.lng], Math.max(12, map.getZoom()));
45         if (!marker) marker = L.marker([m.lat,m.lng]).addTo(map);
46         marker.setLatLng([m.lat,m.lng]);
47       }
48     });
49   
50     // Zu mir
51     document.getElementById('btnLocate').onclick = ()=>{
52       map.locate({setView:true, maxZoom:15, enableHighAccuracy:true});
53     };
54     map.on('locationfound', e=>{
55       if (!marker) marker = L.marker(e.latlng).addTo(map);
56       marker.setLatLng(e.latlng);
57       try { window.parent.postMessage({type:'map:setTarget', lat:e.latlng.lat, lng:e.latlng.lng}, '*'); } catch(_){ }
58     });
59   })();
60   </script>
