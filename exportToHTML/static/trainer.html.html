28   // Minimal NurBus client compatible with hub-postMessage usage
29   const NurBus=(function(){
30     function createClient(){
31       const handlers=new Map();
32       window.addEventListener('message',ev=>{
33         const m=ev.data||{}; const list=handlers.get(m.type)||[]; const any=handlers.get('*')||[];
34         list.forEach(fn=>fn({type:m.type,payload:m})); any.forEach(fn=>fn({type:m.type,payload:m}));
35       });
36       return {
37         on(type,fn){ handlers.set(type,(handlers.get(type)||[]).concat(fn)); },
38         post(win,type,payload){ try{ win.postMessage(Object.assign({type}, payload||{}),'*'); }catch(_){} }
39       };
40     }
41     return { createClient };
42   })();
43   </script>
