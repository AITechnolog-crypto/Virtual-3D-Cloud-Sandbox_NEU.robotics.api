1    // saas-client.js — minimal client for Azure Functions or similar SaaS endpoints
2    // Persisted config (localStorage): saas.base, saas.token
3    (function(){
4      const LS_BASE = 'saas.base';
5      const LS_TOKEN = 'saas.token';
6    
7      function getBase(){ return (localStorage.getItem(LS_BASE)||'').trim(); }
8      function getToken(){ return (localStorage.getItem(LS_TOKEN)||'').trim(); }
9      function setBase(v){ try{ localStorage.setItem(LS_BASE, String(v||'').trim()); }catch(_){} SaaS.base = getBase(); return SaaS.base; }
10     function setToken(v){ try{ localStorage.setItem(LS_TOKEN, String(v||'').trim()); }catch(_){} SaaS.token = getToken(); return SaaS.token; }
11     function head(){ const h = { 'Content-Type':'application/json' }; const t = getToken(); if (t) h['Authorization'] = 'Bearer ' + t; return h; }
12   
13     async function health(){ const r = await fetch(getBase() + '/health'); return r.json(); }
14     async function config(){ const r = await fetch(getBase() + '/config', { headers: head() }); return r.json(); }
15     async function skills(){ const r = await fetch(getBase() + '/skillpack', { headers: head() }); return r.json(); }
16     async function events(ev){ try{ await fetch(getBase() + '/events', { method:'POST', headers: head(), body: JSON.stringify(ev||{}) }); }catch(_){ /* no-op */ } }
17     async function infer(state){ const r = await fetch(getBase() + '/infer', { method:'POST', headers: head(), body: JSON.stringify(state||{}) }); return r.json(); }
18   
19     // Public API
20     const SaaS = window.SaaS = {
21       base: getBase(),
22       token: getToken(),
23       setBase, setToken,
24       head,
25       health, config, skills, events, infer,
26       // aliases matching the issue description
27       async inferAssist(state){ return infer(state); },
28       async sendEvent(ev){ return events(ev); }
29     };
30   
31     // Helper to wire Remote/Skills once base is configured
32     SaaS.applyRemote = async function(){
33       try{
34         if (typeof window.Remote === 'object' && typeof window.Remote.setSource === 'function'){
35           window.Remote.setSource(getBase() + '/config');
36         }
37         if (typeof window.Skills === 'object' && typeof window.Skills.sync === 'function'){
38           await window.Skills.sync(getBase() + '/skillpack');
39         }
40       }catch(_){ }
41     };
42   
43     // If base already set, try to preload Remote/Skills non-blocking
44     if (getBase()) { SaaS.applyRemote().catch(()=>{}); }
45   })();
