73   (function(){
74     const API = location.origin + '/api/galaxy/generate';
75     const canvas = document.getElementById('galaxy');
76     const stage = document.getElementById('stage');
77     const msg = document.getElementById('msg');

79     // THREE Grundsetup
80     const renderer = new THREE.WebGLRenderer({ canvas, antialias:true, alpha:true });
81     renderer.setPixelRatio(Math.min(window.devicePixelRatio||1, 1.8));
82     const scene = new THREE.Scene();
83     const camera = new THREE.PerspectiveCamera(60, 1, 0.1, 5000);
84     camera.position.set(0,0,900);
85   
86     function resize(){ const w=stage.clientWidth, h=stage.clientHeight; renderer.setSize(w,h,false); camera.aspect=w/h; camera.updateProjectionMatrix(); }
87     window.addEventListener('resize', resize); resize();
88   
89     let galaxy=null; let currentParams=null;
90   
91     function clearGalaxy(){ if(!galaxy) return; try{ scene.remove(galaxy); if(galaxy.geometry) galaxy.geometry.dispose(); if(galaxy.material) galaxy.material.dispose(); galaxy=null; }catch(_){} }
92   
93     function lerpColor(a,b,t){ const ca=new THREE.Color(a), cb=new THREE.Color(b); return ca.lerp(cb, t).getStyle(); }
94   
95     function buildGalaxy(params){
96       clearGalaxy();
97       const arms = params.arms||4;
98       const count = Math.min(90000, Math.max(3000, params.stars||48000));
99       const radius = params.radius||360;
100      const twist = params.twist||1.0;
101      const mirror = !!params.mirror;
102      const palette = Array.isArray(params.palette)&&params.palette.length? params.palette : ['#66e0ff','#a8ffdf','#ffffff'];
103  
104      const positions = new Float32Array(count * (mirror? 6 : 3));
105      const colors = new Float32Array(count * (mirror? 6 : 3));
106  
107      function rand(){ return Math.random(); }
108      function gaussian(){ // Box-Muller
109        let u=0,v=0; while(u===0)u=Math.random(); while(v===0)v=Math.random(); return Math.sqrt(-2.0*Math.log(u))*Math.cos(2.0*Math.PI*v);
110      }
111  
112      for(let i=0;i<count;i++){
113        const r = Math.pow(rand(), 0.5) * radius; // dichter im Zentrum
114        const arm = i % arms;
115        const baseAngle = (arm/arms) * Math.PI*2 + r * (twist/ radius) * 4.0;
116        const jitter = gaussian()* (0.7 + 0.6 * (r/radius));
117        const x = Math.cos(baseAngle) * r + jitter*2;
118        const y = (gaussian()*0.6) * (1 - r/radius) * 40; // leichte Dicke
119        const z = Math.sin(baseAngle) * r + jitter*2;
120        const t = Math.min(1, r/radius);
121        const cA = new THREE.Color(palette[0]);
122        const cB = new THREE.Color(palette[palette.length-1]);
123        const c = cA.lerp(cB, t);
124  
125        let idx = i*3;
126        positions[idx+0]=x; positions[idx+1]=y; positions[idx+2]=z;
127        colors[idx+0]=c.r; colors[idx+1]=c.g; colors[idx+2]=c.b;
128  
129        if (mirror){
130          // Spiegeleintrag (x → -x)
131          const midx = count*3 + idx;
132          positions[midx+0] = -x; positions[midx+1] = y; positions[midx+2] = z;
133          colors[midx+0]=c.r; colors[midx+1]=c.g; colors[midx+2]=c.b;
134        }
135      }
136  
137      const geo = new THREE.BufferGeometry();
138      geo.setAttribute('position', new THREE.BufferAttribute(positions, 3));
139      geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
140  
141      const mat = new THREE.PointsMaterial({ size: 2.0, vertexColors: true, transparent:true, opacity:0.95, depthWrite:false, blending:THREE.AdditiveBlending });
142      galaxy = new THREE.Points(geo, mat);
143      scene.add(galaxy);
144      currentParams = params;
145    }

147    const clock = new THREE.Clock();
148    function animate(){
149      const dt = clock.getDelta();
150      if (galaxy){
151        galaxy.rotation.y += 0.06*dt;
152        scene.rotation.z += 0.01*dt;
153      }
154      renderer.render(scene, camera);
155      requestAnimationFrame(animate);
156    }
157    animate();

159    async function generate(){
160      const prompt = document.getElementById('prompt').value.trim();
161      const seed = document.getElementById('seed').value.trim();
162      const mirror = document.getElementById('mirror').checked;
163      msg.textContent = 'Erzeuge…';
164      try{
165        const res = await fetch(API, { method:'POST', headers:{ 'Content-Type':'application/json' }, body: JSON.stringify({ prompt, seed: seed||undefined, mirror }) });
166        const j = await res.json();
167        if (!res.ok || !j || !j.params){ throw new Error('API Fehler'); }
168        buildGalaxy(j.params);
169        msg.textContent = `OK • Arme=${j.params.arms} • Sterne≈${j.params.stars} • Spiegel=${j.params.mirror?'ja':'nein'}`;
170      }catch(e){
171        msg.textContent = '❌ Fehler: ' + (e.message||e);
172      }
173    }

181    });
