16   <script>
17   (async()=>{
18     try{
19       const v=document.getElementById('v'), c=document.getElementById('c'), x=c.getContext('2d');
20       const stream=await navigator.mediaDevices.getUserMedia({video:{width:{ideal:320},height:{ideal:240},facingMode:'environment'},audio:false});
21       v.srcObject=stream; await v.play();
22       function loop(){
23         const w=c.width=innerWidth, h=c.height=innerHeight, sw=v.videoWidth||320, sh=v.videoHeight||240;
24         const s=Math.min(w/sw,h/sh), dw=(sw*s)|0, dh=(sh*s)|0, ox=((w-dw)/2)|0, oy=((h-dh)/2)|0;
25         x.drawImage(v,ox,oy,dw,dh);
26         const img=x.getImageData(ox,oy,dw,dh), d=img.data;
27         let cnt=0, sx=0, sy=0;
28         for(let i=0;i<d.length;i+=4){
29           const r=d[i], g=d[i+1], b=d[i+2];
30           const max=Math.max(r,g,b), min=Math.min(r,g,b), C=max-min;
31           let H=0; if(C>0){ if(max===r) H=((g-b)/C)%6; else if(max===g) H=(b-r)/C+2; else H=(r-g)/C+4; H*=60; if(H<0)H+=360; }
32           const S=max?C/max:0;
33           if(H>340||H<20){ if(S>0.4 && max>80){
34             const idx=(i>>2); const px=idx%dw, py=(idx/dw)|0; cnt++; sx+=px; sy+=py;
35           }}
36         }
37         if(cnt>100){
38           const cx=sx/cnt, cy=sy/cnt, size=cnt/(dw*dh);
39           try{
40             parent.postMessage({ type:'vision:blob', payload:{ x:(cx/dw)*2-1, y:(cy/dh)*2-1, size } }, '*');
41           }catch(_){ }
42           // draw reticle
43           x.strokeStyle='#0f0'; x.beginPath(); x.arc(ox+cx,oy+cy,Math.max(8,Math.sqrt(cnt)),0,Math.PI*2); x.stroke();
44         }
45         requestAnimationFrame(loop);
46       }
47       loop();
48     }catch(e){
49       console.warn('Vision init failed', e);
50     }
51   })();
52   </script>
