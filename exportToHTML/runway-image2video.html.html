82   // Service Worker für Mock‑API registrieren
83   if ('serviceWorker' in navigator){ window.addEventListener('load', ()=> navigator.serviceWorker.register('sw-mock-api.js').catch(()=>{}) ); }
84   
85   (function(){
86     const API_CREATE = location.origin + '/api/runway/image-to-video';
87     const API_TASK   = location.origin + '/api/runway/tasks?id=';
88     const elUrl = document.getElementById('imgUrl');
89     const elModel = document.getElementById('model');
90     const elRatio = document.getElementById('ratio');
91     const elSeed = document.getElementById('seed');
92     const elMsg = document.getElementById('msg');
93     const elStatus = document.getElementById('status');
94     const v = document.getElementById('vid');
95     const i = document.getElementById('img');
96   
97     let taskId = null;
98     let pollIv = null;
99   
100    function sanitizeUrl(u){
101      if (!u) return '';
102      let x = String(u).trim();
103      x = x.replace(/^htttps:/,'https:').replace(/^htttp:/,'http:');
104      return x;
105    }
106  
107    async function create(){
108      try{
109        const payload = {
110          promptImage: sanitizeUrl(elUrl.value) || 'https://upload.wikimedia.org/wikipedia/commons/thumb/5/50/Googleplex_HQ_%28cropped%29.jpg/640px-Googleplex_HQ_%28cropped%29.jpg',
111          model: elModel.value||'gen4_turbo',
112          ratio: elRatio.value||'1280:768'
113        };
114        const s = (elSeed.value||'').trim(); if (s) payload.seed = s;
115        elMsg.textContent = 'Erzeuge Task…';
116        const r = await fetch(API_CREATE, { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify(payload) });
117        const j = await r.json();
118        if (!r.ok){ elMsg.textContent = '❌ Fehler: ' + (j && j.error || ('HTTP '+r.status)); return; }
119        taskId = j.id; elStatus.textContent = j.status || 'PENDING'; elMsg.textContent = 'Task '+taskId+' erstellt. Starte Polling…';
120        startPolling();
121      }catch(e){ elMsg.textContent = '❌ Fehler: ' + (e.message||e); }
122    }
123  
124    async function pollOnce(){
125      if (!taskId) return;
126      try{
127        const r = await fetch(API_TASK + encodeURIComponent(taskId), { cache: 'no-store' });
128        const j = await r.json();
129        if (!r.ok){ elMsg.textContent = '❌ Fehler: ' + (j && j.error || ('HTTP '+r.status)); clearInterval(pollIv); return; }
130        elStatus.textContent = j.status || '-';
131        if (j.status === 'SUCCEEDED' && j.result && j.result.videoUrl){
132          i.style.display='none'; v.style.display='block'; v.src = j.result.videoUrl; try{ await v.play(); }catch(_){}
133          elMsg.textContent = '✅ Fertig.'; clearInterval(pollIv);
134        } else if (j.status === 'FAILED'){
135          elMsg.textContent = '❌ Task fehlgeschlagen.'; clearInterval(pollIv);
136        } else {
137          elMsg.textContent = '⏳ '+(j.status||'Warte…');
138        }
139      }catch(e){ elMsg.textContent = '❌ Fehler beim Abrufen: ' + (e.message||e); clearInterval(pollIv); }
140    }
141  
142    function startPolling(){ if (pollIv) clearInterval(pollIv); pollIv = setInterval(pollOnce, 1200); }
143    function cancel(){ if (pollIv) clearInterval(pollIv); pollIv=null; taskId=null; elStatus.textContent='–'; elMsg.textContent='Abgebrochen.'; }
144  
145    document.getElementById('btnCreate').addEventListener('click', create);
146    document.getElementById('btnCancel').addEventListener('click', cancel);
147  })();
