12    * AgroKommerzOS_Pro (offline, single file) 
13    * ---------------------------------------- 
14    * - Biometrie (SHA-256 Hash) als Start-Gate – keine echte DNA 
15    * - Sensoren → KI → Bewässerung/Energie/Handel/Finanzen 
16    * - Social Equity + AidRouter (Wasserhilfe) 
17    * - Mehrere Wallets (Owner/Ops/Aid) + RevenueRouter (konfigurierbar) 
18    * - Feature-Flags: cloud / marketing / kopplung / walletbackfeed 
19    * - Compliance/Regulator: Whitelist, Preis-Floor, Tageslimit, KYC-Gate 
20    * - Entrepreneur-Verkaufsloop (Dauerschleife) zusätzlich zum Sensor-Tick 
21    * - Snapshots: ./data/state.json (inkl. Flags/Wallets/Compliance/Contacts) 
22    * 
23    * Build: 
24    *   javac AgroKommerzOS_Pro.java 
25    * 
26    * Biometrie einmalig: 
27    *   java AgroKommerzOS_Pro enroll <DEIN_SHA256_HEX> 
28    * 
29    * Start (Beispiel, komplett offline): 
30    *   java AgroKommerzOS_Pro run <DEIN_SHA256_HEX> zone=FARMLAND units=10000000000 \ 
31    *     flags=cloud:off,marketing:on,kopplung:on,walletback:on \ 
32    *     split=owner:0.6,ops:0.35,aid:0.05 phone="+43-000-0000000"

49       static String sha256(String s){
50           try{
51               MessageDigest md = MessageDigest.getInstance("SHA-256");
52               byte[] d = md.digest(s.getBytes(StandardCharsets.UTF_8));
53               StringBuilder sb=new StringBuilder(); for(byte b:d) sb.append(String.format("%02x", b));
54               return sb.toString();
55           }catch(Exception e){ throw new RuntimeException(e); }

120      static final class RevenueRouter {
121          final Wallet owner, ops, aid;
122          double pctOwner=0.60, pctOps=0.35, pctAid=0.05;
123          RevenueRouter(Wallet owner, Wallet ops, Wallet aid){ this.owner=owner; this.ops=ops; this.aid=aid; }
124          void setSplits(String s){ if(s==null) return;
125              for(String p:s.split(",")){
126                  String[] kv=p.split(":"); if(kv.length!=2) continue;
127                  String k=kv[0].trim().toLowerCase(); double v=Double.parseDouble(kv[1]);
128                  if("owner".equals(k)) pctOwner=v; else if("ops".equals(k)) pctOps=v; else if("aid".equals(k)) pctAid=v;
129              }
130              double sum=pctOwner+pctOps+pctAid; if(Math.abs(sum-1.0)>1e-6){ // normalisieren
131                  pctOwner/=sum; pctOps/=sum; pctAid/=sum;
132              }
133          }
134          void route(double revenue){
135              if(revenue<=0) return;
136              owner.credit(revenue*pctOwner, "revenue split");
137              ops.credit(revenue*pctOps, "revenue split");
138              aid.credit(revenue*pctAid, "revenue split");
139          }

177      enum EquityTier { NEEDY, STANDARD, WEALTHY }
178      static final class SocialEquity {
179          EquityTier classify(double incomeIdx){ if(incomeIdx>=0.75) return EquityTier.WEALTHY; if(incomeIdx<=0.35) return EquityTier.NEEDY; return EquityTier.STANDARD; }
180          double buyerPrice(EquityTier t,double p){ return switch(t){ case NEEDY->Math.max(0,p*0.9); case WEALTHY->p*1.2; default->p; }; }
181          double waterAidPrice(EquityTier t,double liters){ double base=liters*0.02; return (t==EquityTier.NEEDY)?0 : (t==EquityTier.WEALTHY? base*1.6 : base); }
182      }
183      static final class AidRouter {
184          final SocialEquity eq; final Silos silos; final Wallet opsWallet;
185          AidRouter(SocialEquity e, Silos s, Wallet ops){ this.eq=e; this.silos=s; this.opsWallet=ops; }
186          void waterAid(double liters, double incomeIdx){
187              EquityTier t = eq.classify(incomeIdx);
188              if(silos.waterLiters < liters){ LOG.info("[AID] nicht genug Wasser"); return; }
189              silos.waterLiters -= liters;
190              double charge = eq.waterAidPrice(t, liters);
191              if(charge>0) opsWallet.credit(charge, "AID water "+(int)liters+"L tier="+t);
192              LOG.info("[AID] "+(int)liters+"L geliefert tier="+t+" charge="+money(charge));
193          }
194      }

197      static final class Regulator {
198          final Set<String> whitelist = new HashSet<>(List.of("Wheat","Corn"));
199          final Map<String,Double> floor = Map.of("Wheat", 0.55, "Corn", 0.40);
200          final int dailyKgLimit = 4000;
201          final AtomicLong soldTodayKg = new AtomicLong(0);
202          LocalDate day = LocalDate.now();
203          boolean kycOk(double reputation){ return reputation>=0.5; }
204          synchronized boolean allowSell(String product, int kg, double price){
205              if(!LocalDate.now().equals(day)){ day=LocalDate.now(); soldTodayKg.set(0); }
206              if(!whitelist.contains(product)) return false;
207              if(price < floor.getOrDefault(product, 0.0)) return false;
208              if(soldTodayKg.get()+kg > dailyKgLimit) return false;
209              soldTodayKg.addAndGet(kg); return true;
210          }
211      }

214      static final class Offer { final long id; final String name; double base; int stockKg;
215          static final AtomicLong SEQ=new AtomicLong(1);
216          Offer(String n,double b,int s){ id=SEQ.getAndIncrement(); name=n;base=b; stockKg=s; } }
217      static final class Marketplace {
218          final Map<Long,Offer> offers = new LinkedHashMap<>();
219          final Map<Long,Integer> demand = new HashMap<>();
220          final Random r = new Random();
221          void add(Offer o){ offers.put(o.id, o); }
222          double demandIndex(long id){ int d=demand.getOrDefault(id,0)+1; demand.put(id,d); return 1.0 + Math.min(1.2, d*0.04); }
223          double trySell(String product, int kg, SocialEquity eq, Regulator reg, boolean marketingBoost,
224                         RevenueRouter router, Wallet opsWallet){
225              Offer o = offers.values().stream().filter(x->x.name.equals(product) && x.stockKg>0).findFirst().orElse(null);
226              if(o==null) return 0;
227              int sell = Math.min(kg, o.stockKg); if(sell<=0) return 0;
228  
229              // Käuferprofil (sim)
230              double incomeIdx = 0.30 + r.nextDouble()*0.70;
231              double reputation= 0.40 + r.nextDouble()*0.60; // für KYC
232              if(!reg.kycOk(reputation)){ LOG.info("[KYC] abgelehnt"); return 0; }
233  
234              double dyn = o.base * demandIndex(o.id);
235              if(marketingBoost) dyn *= 1.05; // kleines Plus
236              EquityTier tier = eq.classify(incomeIdx);
237              double pricePerKg = clamp(eq.buyerPrice(tier, dyn), o.base*0.7, o.base*1.9);
238  
239              if(!reg.allowSell(product, sell, pricePerKg)){
240                  LOG.info("[REG] Verkauf blockiert (Whitelist/Floor/Limit)"); return 0;
241              }
242  
243              double revenue = sell * pricePerKg;
244              o.stockKg -= sell;
245  
246              // Einnahmen verteilen über Router
247              router.route(revenue);
248              opsWallet.credit(0, "log"); // nudge ledger rotation
249              LOG.info("[SALE] "+sell+"kg "+product+" @"+money(pricePerKg)+"/kg → "+money(revenue)
250                       +" (tier="+tier+") stockLeft="+o.stockKg);
251              return revenue;
252          }

259      static final class KI {
260          Decision decide(List<SensorData> d, Silos s, boolean couplingOn){
261              double soil = val(d,"soil_moisture"), sun=val(d,"sun_intensity"), tank=val(d,"cistern_liters"), rain=val(d,"rain_forecast_mm");
262  
263              double need = 0;
264              if(soil<28 && rain<2) need=600;
265              else if(soil<35 && rain<3) need=320;
266              if(tank<800) need*=0.5;
267              if(sun>0.8) need*=1.1;
268              if(!couplingOn) need*=1.06; // ohne Kopplung etwas ineffizienter
269  
270              IrrigationPlan ip = (need>0)? new IrrigationPlan("Feld A", Math.round(need)) : null;
271  
272              double demand = 40 + (sun*30);
273              if(!couplingOn) demand*=0.95; // Kopplung aus → weniger Ambition
274              EnergyPlan ep = new EnergyPlan(demand, sun>0.75);
275  
276              boolean sell = s.grainKg>600;
277              double sellKg = sell ? Math.min(300, s.grainKg-500) : 0;
278              TradePlan tp = new TradePlan(sell, sellKg, "Wheat");
279  
280              FinancePlan fp = new FinancePlan(need>0 ? Math.max(2, need*0.01) : 0, 3);
281              return new Decision(ip, ep, tp, fp);
282          }
283          double val(List<SensorData> d, String k){ return d.stream().filter(x->x.type.equals(k)).mapToDouble(x->x.value).findFirst().orElse(0); }
284      }

339      static final class Engine implements AutoCloseable {
340          // Config/Flags
341          final FeatureFlags flags;
342          final RevenueRouter router;
343          final Contacts contacts;
344          final CloudSync cloud;
345  
346          // Core
347          final Sensors sensors = new Sensors();
348          final KI ki = new KI();
349          final Silos silos = new Silos();
350          final SocialEquity equity = new SocialEquity();
351          final Regulator reg = new Regulator();
352          final Marketplace market = new Marketplace();
353  
354          // Wallets
355          final Wallet wOwner = new Wallet("OWNER");
356          final Wallet wOps   = new Wallet("OPS");
357          final Wallet wAid   = new Wallet("AID");
358  
359          // Actuators
360          final IrrigationSystem irrigation = new IrrigationSystem(silos);
361          final SolarSystem solar = new SolarSystem(silos);
362  
363          // Schedulers
364          final ScheduledExecutorService ses = Executors.newScheduledThreadPool(2);
365  
366          Engine(FeatureFlags f, RevenueRouter router, Contacts c){
367              this.flags=f; this.router=router; this.contacts=c; this.cloud=new CloudSync(f.cloud);
368              // Seed-Angebote
369              market.add(new Offer("Wheat", 0.85, 3000));
370              market.add(new Offer("Corn",  0.65, 2200));
371              // Bootstrap
372              wOps.credit(120, "bootstrap");
373          }
374  
375          void start(String zoneArg, long units){
376              LOG.info("Deploy "+NumberFormat.getInstance(Locale.US).format(units)+" Units (agg)");
377              // Haupt-Tick (Sensors → KI → Aktoren → Aid → Snapshot)
378              ses.scheduleAtFixedRate(()->{
379                  try{
380                      List<SensorData> d = sensors.read();
381                      Decision dec = ki.decide(d, silos, flags.kopplung);
382  
383                      irrigation.execute(dec.irrigation);
384                      solar.execute(dec.energy, flags.kopplung);
385  
386                      // On-demand Trade aus KI (kleiner Batch)
387                      if(dec.trade.sellGrain && dec.trade.kg>0){
388                          boolean ok = silos.useGrain(dec.trade.kg);
389                          if(ok){
390                              market.trySell(dec.trade.product, (int)dec.trade.kg, equity, reg, flags.marketing, router, wOps);
391                              if(flags.walletBackfeed){
392                                  // Light Reinvest in Silos
393                                  double fund = dec.trade.kg * 0.10; // pseudo-Äquivalent
394                                  silos.addEnergy(fund*0.5); silos.addWater(fund*5);
395                              }
396                          }
397                      }
398  
399                      // Regelmäßige Wasserhilfe-Demo
400                      if ((System.currentTimeMillis()/1000)%7==0) {
401                          new AidRouter(equity, silos, wOps).waterAid(160, 0.28);
402                      }
403  
404                      // Cloud (inaktiv, wenn cloud=false)
405                      cloud.maybeSync();
406  
407                      // Snapshot
408                      Snapshot.write(this, dec);
409  
410                      LOG.info(String.format(Locale.US,"Silos water=%.0fL energy=%.0f grain=%.0fkg | wallets owner=%s ops=%s aid=%s",
411                          silos.waterLiters, silos.energyUnits, silos.grainKg,
412                          money(wOwner.bal()), money(wOps.bal()), money(wAid.bal())));
413                  }catch(Exception e){ LOG.log(Level.SEVERE,"tick error", e); }
414              }, 0, 1200, TimeUnit.MILLISECONDS);
415  
416              // Entrepreneur-Verkaufsloop (Dauerschleife): legt fortlaufend Angebote ab
417              ses.scheduleAtFixedRate(()->{
418                  try{
419                      // Nur wenn Lager > Puffer
420                      if(silos.grainKg > 700){
421                          int burst = (int)Math.min(500, silos.grainKg - 600);
422                          if(silos.useGrain(burst)){
423                              market.trySell("Wheat", burst, equity, reg, flags.marketing, router, wOps);
424                          }
425                      }
426                  }catch(Exception e){ LOG.log(Level.WARNING,"entrepreneur loop error", e); }
427              }, 1000, 2000, TimeUnit.MILLISECONDS);
428          }
429  
430          @Override public void close(){ ses.shutdownNow(); }
431      }

438          Map<String,String> cli = parseArgs(args);
439          String mode = cli.getOrDefault("mode","run");
440  
441          // Biometrie
442          Biometry bio = new Biometry();
443          if("enroll".equalsIgnoreCase(mode)){
444              if(args.length<2){ System.out.println("Usage: java AgroKommerzOS_Pro enroll <SHA256_HEX>"); return; }
445              bio.enroll(args[1].trim().toLowerCase());
446              System.out.println("Enrolled marker = "+args[1]); return;
447          }
448          if(!bio.load()){ System.out.println("Kein Marker registriert. Bitte zuerst:\n  java AgroKommerzOS_Pro enroll <dein_SHA256_hex>"); return; }
449          String provided = (args.length>1)? args[1].trim().toLowerCase() : "";
450          if(!bio.check(provided)){ System.out.println("Biometrie: Zugriff verweigert (falscher Marker-Hash)."); return; }
451  
452          // Flags & Splits
453          FeatureFlags flags = FeatureFlags.from(cli.get("flags"));
454          Wallet wOwner = new Wallet("OWNER"), wOps=new Wallet("OPS"), wAid=new Wallet("AID");
455          RevenueRouter router = new RevenueRouter(wOwner, wOps, wAid);
456          router.setSplits(cli.get("split"));
457  
458          // Kontakte
459          Contacts contacts = new Contacts();
460          if(cli.containsKey("phone")) contacts.setPhone(cli.get("phone"));
461          if(cli.containsKey("callback") && ("true".equalsIgnoreCase(cli.get("callback")))) contacts.requestCallback();
462          contacts.snapshot();
463  
464          // Route/Units
465          String zoneArg = cli.get("zone");
466          long units = parseLong(cli.getOrDefault("units","10000000000"), 10_000_000_000L);
467  
468          try(Engine engine = new Engine(flags, router, contacts)){
469              // Wallets vom Engine ersetzen (Router wurde mit lokalen Wallets erzeugt – hier angleichen)
470              engine.wOwner.credit(wOwner.bal(), "sync"); // 0 beim Start
471              engine.wOps.credit(wOps.bal(), "sync");
472              engine.wAid.credit(wAid.bal(), "sync");
473              // Router zeigt auf Engine-Wallets? (einfachheitshalber belassen – Router nutzt seine eigenen,
474              // für reale Nutzung könnte man den Router in Engine konstruieren.)
475  
476              engine.start(zoneArg, units);
477              LOG.info("Engine läuft. Ctrl+C zum Beenden. Snapshots in ./data/state.json");
478  
479              Runtime.getRuntime().addShutdownHook(new Thread(()->{
480                  try{ engine.close(); }catch(Exception ignored){}
481                  System.out.println("\nShutdown ok.");
482              }));
483              Thread.currentThread().join();
484          }
485      }
486  }
