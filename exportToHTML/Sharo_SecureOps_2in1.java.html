109      static final class PredictiveRiskEngine {
110          // Einfache EMA + Trend; rein informativ (Sharo Predictive)
111          private Double ema = null;
112          private final double alpha = 0.35;
113          private final Deque<Double> last = new ArrayDeque<>();
114  
115          public void feedCO2(double ppm){
116              if (ema == null) ema = ppm;
117              else ema = alpha*ppm + (1-alpha)*ema;
118              last.add(ema);
119              if (last.size() > 6) last.removeFirst();
120          }
121          public double trend(){ // einfacher Trend: letzte minus mittlere
122              if (last.isEmpty()) return 0;
123              double avg = last.stream().mapToDouble(d->d).average().orElse(0);
124              return last.getLast() - avg;
125          }
126          public String currentLevel(){
127              double t = trend();
128              if (t > 60) return "HIGH";
129              if (t > 20) return "ELEVATED";
130              return "NORMAL";
131          }
132      }

179          File exportPackage(String name){
180              try {
181                  List<String> out = new ArrayList<>();
182                  String prev = "GENESIS";
183                  for (String l : lines) {
184                      String h = sha256(prev + "|" + l);
185                      out.add(h + "  " + l);
186                      prev = h;
187                  }
188                  out.add("CHAIN_HEAD="+prev);
189                  try (Writer w = new OutputStreamWriter(new FileOutputStream(name), StandardCharsets.UTF_8)){
190                      w.write(String.join("\n", out));

193                  System.out.println("Evidence export fehlgeschlagen: "+e.getMessage());
194              }
195              return new File(name);

197          private static String sha256(String s) throws Exception{
198              MessageDigest md = MessageDigest.getInstance("SHA-256");
199              byte[] d = md.digest(s.getBytes(StandardCharsets.UTF_8));
200              StringBuilder sb = new StringBuilder();
201              for (byte b : d) sb.append(String.format("%02x", b));
202              return sb.toString();
203          }

312      static final class QrPngWriter {
313          @SuppressWarnings("unchecked")
314          static boolean writePngIfZXing(String payload, File out, int size){
315              try {
316                  Class<?> bfClass   = Class.forName("com.google.zxing.BarcodeFormat");
317                  Object qrFormat    = Enum.valueOf((Class<Enum>) bfClass.asSubclass(Enum.class), "QR_CODE");
318                  Class<?> writerCls = Class.forName("com.google.zxing.qrcode.QRCodeWriter");
319                  Object writer      = writerCls.getDeclaredConstructor().newInstance();
320                  Method encode      = writerCls.getMethod("encode", String.class, bfClass, int.class, int.class, Map.class);
321                  Object bitMatrix   = encode.invoke(writer, payload, qrFormat, size, size, null);
322                  Class<?> bmClass   = Class.forName("com.google.zxing.common.BitMatrix");
323                  int w = (int) bmClass.getMethod("getWidth").invoke(bitMatrix);
324                  int h = (int) bmClass.getMethod("getHeight").invoke(bitMatrix);
325                  BufferedImage img = new BufferedImage(w,h, BufferedImage.TYPE_INT_RGB);
326                  Method get = bmClass.getMethod("get", int.class, int.class);
327                  for (int y=0; y<h; y++) for (int x=0; x<w; x++) {
328                      boolean on = (boolean) get.invoke(bitMatrix, x, y);
329                      img.setRGB(x, y, on ? 0x000000 : 0xFFFFFF);
330                  }
331                  ImageIO.write(img, "png", out);
332                  System.out.println("[QR] PNG geschrieben → " + out.getAbsolutePath());
333                  return true;
334              } catch (ClassNotFoundException e) {
335                  System.out.println("[QR] ZXing nicht gefunden – PNG-Export übersprungen. Payload unten.");
336                  return false;
337              } catch (Throwable t) {
338                  System.out.println("[QR] Fehler beim PNG-Schreiben: " + t.getMessage());
339                  return false;
340              }
341          }

355          void route(double revenue, double ops){
356              revenue = Math.max(0,revenue); ops=Math.max(0,ops);
357              System.out.println(String.format(Locale.US,"[Route] Einnahmen=%.2f Kosten=%.2f Reserve=%.2f",revenue,ops,reserve));
358              double margin = revenue - ops;
359              if (margin < 0){
360                  double need = -margin;
361                  if (reserve >= need){ reserve -= need; System.out.println("[Route] Defizit "+euro(need)+" aus Reserve gedeckt → "+euro(reserve)); margin=0; }
362                  else { System.out.println("[Route] Defizit "+euro(need)+" NICHT gedeckt. Keine Ausschüttung."); return; }
363              }
364              double fill = Math.min(Math.max(0, p.reserveTarget - reserve), margin);
365              reserve += fill; double rest = margin - fill;
366              if (fill>0) System.out.println("[Route] Reserve +"+euro(fill)+" → "+euro(reserve));
367              if (rest<=0){ System.out.println("[Route] Kein Überschuss."); return; }
368              double toWallet = rest*p.wWallet, toTech = rest*p.wTech, toAid = rest*p.wAid, toCloud = rest*p.wCloud;
369              if (toWallet>0){ wallet.gutschreiben(toWallet); silos.wallet(toWallet); }
370              if (toCloud>0){ cloud.feed(toCloud); silos.cloud(toCloud); }
371              if (toTech>0) System.out.println("[R&D] Budget +"+euro(toTech));
372              if (toAid>0)  System.out.println("[Aid] Budget +"+euro(toAid));
373          }

430              double avg = responseMin.stream().mapToInt(i->i).average().orElse(0);
431              int p90 = percentile(responseMin, 90);
432              System.out.println("== OPS-STATISTIK ==");
433              System.out.println("Total: "+total+" • Avg Resp: "+avg+" min • P90: "+p90+" min");
434              for (Map.Entry<String,Integer> e : byType.entrySet()) System.out.println(" - "+e.getKey()+": "+e.getValue());
435          }
436          private static int percentile(List<Integer> data, int p){
437              if (data.isEmpty()) return 0;
438              List<Integer> s = new ArrayList<>(data); Collections.sort(s);
439              int idx = (int)Math.ceil(p/100.0*s.size())-1; idx=Math.max(0,Math.min(idx,s.size()-1));
440              return s.get(idx);
441          }
442      }
