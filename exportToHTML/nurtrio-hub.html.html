118  <script>
119  // ---------------------------------
120  // Renderer/Loop mit FPS‑Regel & Eco
121  // ---------------------------------
122  const canvas = document.getElementById('cv');
123  const ctx = canvas.getContext('2d');
124  function resize(){
125    const pr = renderer._pxr || 1;
126    const w = canvas.clientWidth, h = canvas.clientHeight;
127    canvas.width = Math.max(1, Math.floor(w*pr));
128    canvas.height = Math.max(1, Math.floor(h*pr));
129    ctx.setTransform(pr,0,0,pr,0,0); // logisch 1x zeichnen
130  }
131  const renderer = {
132    _pxr: Math.min(window.devicePixelRatio||1, 1.5),
133    setPixelRatio(v){ this._pxr = Math.max(0.5, Math.min(2, Number(v)||1)); resize(); }
134  };
135  let scene={}, camera={}; // Platzhalter
136  window.addEventListener('resize', resize); resize();
137  
138  let tPrev=performance.now();
139  let fCnt=0, t0=performance.now();
140  let anim=0;
141  function step(dt){
142    // einfache Visual: wandernde Partikel
143    const w=canvas.clientWidth, h=canvas.clientHeight;
144    ctx.fillStyle = '#05070c'; ctx.fillRect(0,0,w,h);
145    const n = S.eco ? 60 : 140;
146    for(let i=0;i<n;i++){
147      const x = (i*37 + (anim*60)) % w;
148      const y = (i*91 + (anim*30)) % h;
149      const a = 0.35 + 0.35*Math.sin((i+anim)*0.02);
150      ctx.fillStyle = `rgba(0,255,136,${a.toFixed(3)})`;
151      ctx.fillRect(x, y, 2, 2);
152    }
153    anim += (S.eco? 0.8 : 1.6);
154  }
155  
156  function tick(){
157    const t=performance.now(), dt=Math.min(0.05,(t-tPrev)/1000); tPrev=t;
158    // Eco: jeden 2. Frame überspringen
159    if (!(S.eco && (fCnt%2===0))) { step(dt); renderer.render && renderer.render(scene,camera); }
160    requestAnimationFrame(tick);
161  
162    // FPS messen & PixelRatio anpassen
163    fCnt++; if (t - t0 > 1200){
164      const fps = fCnt * 1000 / (t - t0);
165      if (fps < 24) renderer.setPixelRatio(1);
166      else if (fps > 40) renderer.setPixelRatio(Math.min(window.devicePixelRatio||1,1.5));
167      fCnt=0; t0=t;
168    }
169    // Watchdog beat
170    wdLast = Date.now();
171  }
172  requestAnimationFrame(tick);
173  </script>
