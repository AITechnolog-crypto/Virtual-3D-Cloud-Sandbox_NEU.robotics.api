3    public class AppManager {
4    
5        private static volatile boolean running = true;
6    
7        public static void main(String[] args) {
8            System.out.println("AppManager gestartet – Shopmanager ist bereit.");
9            Runtime.getRuntime().addShutdownHook(new Thread(() -> {
10               try {
11                   running = false;
12                   System.out.println("AppManager wird beendet …");
13               } catch (Exception ignored) {}
14           }, "shutdown-hook"));
15   
16           InputListener inputListener = new InputListener();
17           BackgroundScheduler scheduler = new BackgroundScheduler();
18   
19           inputListener.setDaemon(true);
20           scheduler.setDaemon(true);
21           inputListener.start();
22           scheduler.start();
23   
24           while (running) {
25               try {
26                   Thread.sleep(1000); // Hauptschleife hält alles am Leben
27               } catch (InterruptedException e) {
28                   running = false;
29               }
30           }
31           System.out.println("AppManager beendet.");
32       }
33   }

35   class InputListener extends Thread {
36       @Override
37       public void run() {
38           while (!isInterrupted()) {
39               // Simuliert Eingabeüberwachung – könnte an Touch‑Ereignisse gebunden werden
40               try {
41                   Thread.sleep(3000);
42                   System.out.println("[InputListener] Überwache Benutzerinteraktionen…");
43               } catch (InterruptedException e) {
44                   interrupt();
45               }
46           }
47       }
48   }

50   class BackgroundScheduler extends Thread {
51       @Override
52       public void run() {
53           while (!isInterrupted()) {
54               try {
55                   Thread.sleep(60_000); // alle 60 Sekunden Ping
56                   System.out.println("[BackgroundScheduler] Session wach gehalten.");
57                   // Hier könnte man API‑Ping oder Initialisierungsaufruf einbauen
58               } catch (InterruptedException e) {
59                   interrupt();
60               }
61           }
62       }
