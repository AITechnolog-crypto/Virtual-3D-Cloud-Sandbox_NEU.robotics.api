65     <script>
66       const chatButton = document.getElementById('chatButton');
67       const chatPanel = document.getElementById('chatPanel');
68       const chatMessages = document.getElementById('chatMessages');
69       const inputText = document.getElementById('inputText');
70       const sendBtn = document.getElementById('sendBtn');
71       const acceptBtn = document.getElementById('acceptBtn');
72       const consentRow = document.getElementById('consentRow');
73       const ctxInfo = document.getElementById('ctxInfo');
74       let WSCTX = null;
75       // signal readiness to parent (java-interface)
76       try { if (window.parent && window.parent !== window) { window.parent.postMessage({ ns:'JavaWorkspaceV1', type:'ready' }, '*'); } } catch(_) {}
77       // accept context from parent
78       window.addEventListener('message', (ev)=>{
79         try{
80           const d = ev.data||{};
81           if (d && d.ns==='JavaWorkspaceV1' && d.type==='context'){
82             WSCTX = d;
83             if (ctxInfo){
84               const n = Array.isArray(d.snippets)? d.snippets.length : 0;
85               const titles = (d.snippets||[]).slice(0,3).map(s=> s.title||'').filter(Boolean).join(' • ');
86               ctxInfo.textContent = `Workspace: Kontext=${d.context?'an':'aus'} · Auto‑Ingest=${d.autoIngest?'an':'aus'} · Snippets=${n}` + (titles? ` (${titles})` : '');
87               ctxInfo.style.display = 'block';
88             }
89           }
90         }catch(_){ }
91       });
92   
93       // einfache Session-ID lokal
94       let sessionId = localStorage.getItem('appmgr.session');
95       if (!sessionId) {
96         sessionId = 'sess-' + Math.random().toString(36).slice(2, 10);
97         localStorage.setItem('appmgr.session', sessionId);
98       }
99   
100      chatButton.addEventListener('click', () => {
101        const visible = chatPanel.style.display === 'flex';
102        chatPanel.style.display = visible ? 'none' : 'flex';
103        chatPanel.setAttribute('aria-hidden', visible ? 'true' : 'false');
104        if (!visible) {
105          inputText.focus();
106          // Warmup Ping (best-effort)
107          fetch('/api/warmup', { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ sessionId }) }).catch(()=>{});
108        }
109      });
110  
111      function appendMessage(text, who) {
112        const div = document.createElement('div');
113        div.className = who === 'user' ? 'msg-user' : 'msg-bot';
114        div.textContent = text;
115        chatMessages.appendChild(div);
116        chatMessages.scrollTop = chatMessages.scrollHeight;
117      }
118  
119      async function sendMessage() {
120        let text = inputText.value.trim();
121        if (!text) return;
122        // Workspace‑Kontext ggf. voranstellen
123        try{
124          if (WSCTX && WSCTX.context && Array.isArray(WSCTX.snippets) && WSCTX.snippets.length){
125            const preview = WSCTX.snippets.slice(0,3).map(s=> `• ${s.title||''} [${(s.tags||[]).join(', ')}]`).join('\n');
126            text = `Kontext:\n${preview}\n\n` + text;
127          }
128        }catch(_){ }
129        appendMessage(text, 'user');
130        inputText.value = '';
131        appendMessage('...', 'bot'); // placeholder while waiting
132  
133        try {
134          const resp = await fetch('/api/chat/assist', {
135            method: 'POST',
136            headers: { 'Content-Type': 'application/json' },
137            body: JSON.stringify({ channel: 'customer', role: 'user', sender: 'user', message: text })
138          });
139          const data = await resp.json();
140          const last = chatMessages.lastChild;
141          if (last && last.textContent === '...') last.textContent = (data && data.reply) || 'Keine Antwort';
142          else appendMessage((data && data.reply) || 'Keine Antwort', 'bot');
143        } catch (e) {
144          const last = chatMessages.lastChild;
145          const msg = 'Fehler: ' + (e.message || 'Netzwerk');
146          if (last && last.textContent === '...') last.textContent = msg; else appendMessage(msg, 'bot');
147        }
148      }
149  
150      sendBtn.addEventListener('click', sendMessage);
151      inputText.addEventListener('keydown', (e) => { if (e.key === 'Enter') sendMessage(); });
152  
153      // Consent speichern
154      acceptBtn.addEventListener('click', async () => {
155        try {
156          await fetch('/api/consent', { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ sessionId }) });
157          appendMessage('Konsent gespeichert. Ich bin bereit als Shop Manager.', 'bot');
158        } catch (_) {
159          appendMessage('Lokaler Modus aktiv – Konsent wird lokal angenommen.', 'bot');
160        } finally {
161          try{ consentRow.style.display = 'none'; }catch(_){ }
162        }
163      });
