11   /**
12    * Very small file-based append-only history store (JSONL files).
13    * - Writes telemetry snapshots to data/history/telemetry-YYYY-MM.jsonl
14    * - Provides simple iteration to compute summaries without loading entire file into memory at once.
15    */
16   @Component
17   public class HistoryStore {
18       private static final Path DATA_DIR = Paths.get("data", "history");
19   
20       public HistoryStore(){
21           try { Files.createDirectories(DATA_DIR); } catch (Exception ignored) {}
22       }
23   
24       public void append(Map<String,Object> payload){
25           try {
26               Path file = currentFile();
27               String line = toJsonLine(payload) + "\n";
28               Files.writeString(file, line, StandardCharsets.UTF_8,
29                       StandardOpenOption.CREATE, StandardOpenOption.APPEND);
30           } catch (IOException ignored) {}
31       }
32   
33       public Summary summarize(int maxFiles){
34           Summary s = new Summary();
35           try {
36               List<Path> files = listFilesSortedNewestFirst();
37               int count = 0;
38               for (Path f : files){
39                   if (maxFiles > 0 && count >= maxFiles) break;
40                   count++;
41                   summarizeFile(f, s);
42               }
43           } catch (Exception ignored) {}
44           return s;
45       }
46   
47       private List<Path> listFilesSortedNewestFirst() throws IOException {
48           List<Path> list = new ArrayList<>();
49           if (Files.exists(DATA_DIR)) {
50               try (DirectoryStream<Path> ds = Files.newDirectoryStream(DATA_DIR, "telemetry-*.jsonl")) {
51                   for (Path p : ds) list.add(p);
52               }
53           }
54           list.sort((a,b) -> {
55               try {
56                   long ta = Files.getLastModifiedTime(a).toMillis();
57                   long tb = Files.getLastModifiedTime(b).toMillis();
58                   return Long.compare(tb, ta);
59               } catch (IOException e) { return 0; }
60           });
61           return list;
62       }

64       public static final class Summary{
65           public long firstTs = Long.MAX_VALUE;
66           public long lastTs = 0L;
67           public long totalEvents = 0L;
68           // simple metrics aggregation for common fields
69           public double energyMin = Double.POSITIVE_INFINITY;
70           public double energyMax = Double.NEGATIVE_INFINITY;
71           public double energySum = 0.0;
72           public long energyCount = 0L;
73   
74           public Map<String,Object> toMap(){
75               Map<String,Object> m = new LinkedHashMap<>();
76               if (firstTs == Long.MAX_VALUE) firstTs = 0L;
77               m.put("firstTs", firstTs);
78               m.put("lastTs", lastTs);
79               m.put("totalEvents", totalEvents);
80               if (energyCount > 0){
81                   m.put("energy", Map.of(
82                           "min", energyMin == Double.POSITIVE_INFINITY ? null : energyMin,
83                           "max", energyMax == Double.NEGATIVE_INFINITY ? null : energyMax,
84                           "avg", energySum / Math.max(1, energyCount),
85                           "count", energyCount
86                   ));
87               } else {
88                   m.put("energy", Map.of("count", 0));
89               }
90               return m;
91           }
92       }

94       private void summarizeFile(Path f, Summary s){
95           try {
96               List<String> lines = Files.readAllLines(f, StandardCharsets.UTF_8);
97               for (String line : lines){
98                   if (line == null || line.isBlank()) continue;
99                   s.totalEvents++;
100                  // very naive parse: extract ts and metrics.energy if present
101                  try {
102                      int tsIdx = line.indexOf("\"ts\":");
103                      if (tsIdx >= 0){
104                          int start = tsIdx + 5;
105                          int end = findNumberEnd(line, start);
106                          long ts = Long.parseLong(line.substring(start, end).replaceAll("[^0-9]",""));
107                          if (s.firstTs == 0L || s.firstTs == Long.MAX_VALUE) s.firstTs = ts;
108                          if (ts < s.firstTs) s.firstTs = ts;
109                          if (ts > s.lastTs) s.lastTs = ts;
110                      }
111                      int mIdx = line.indexOf("\"metrics\":");
112                      if (mIdx >= 0){
113                          int eIdx = line.indexOf("\"energy\":", mIdx);
114                          if (eIdx >= 0){
115                              int eStart = eIdx + 9;
116                              int eEnd = findNumberEnd(line, eStart);
117                              String sub = line.substring(eStart, Math.min(line.length(), eEnd));
118                              double v = parseNumberSafe(sub);
119                              if (!Double.isNaN(v)) {
120                                  s.energyMin = Math.min(s.energyMin, v);
121                                  s.energyMax = Math.max(s.energyMax, v);
122                                  s.energySum += v;
123                                  s.energyCount++;
124                              }
125                          }
126                      }
127                  } catch (Exception ignored) {}
128              }
129          } catch (Exception ignored) {}
130      }

132      private static double parseNumberSafe(String s){
133          if (s == null) return Double.NaN;
134          StringBuilder b = new StringBuilder();
135          for (int i=0;i<s.length();i++){
136              char c = s.charAt(i);
137              if ((c>='0'&&c<='9') || c=='.' || c=='-') b.append(c);
138          }
139          try { return Double.parseDouble(b.toString()); } catch (Exception e){ return Double.NaN; }
140      }
141  
142      private static int findNumberEnd(String s, int start){
143          int i = start;
144          while (i < s.length()){
145              char c = s.charAt(i);
146              if ((c >= '0' && c <= '9') || c == '.' || c == '-') { i++; } else { break; }
147          }
148          return i;
149      }

151      private Path currentFile(){
152          String ym = new SimpleDateFormat("yyyy-MM").format(new Date());
153          return DATA_DIR.resolve("telemetry-"+ym+".jsonl");
154      }

156      @SuppressWarnings("unchecked")
157      private static String toJsonLine(Map<String,Object> map){
158          StringBuilder sb = new StringBuilder();
159          appendJson(sb, map);
160          return sb.toString();
161      }

163      private static void appendJson(StringBuilder sb, Object obj){
164          if (obj == null){ sb.append("null"); return; }
165          if (obj instanceof Number || obj instanceof Boolean){ sb.append(String.valueOf(obj)); return; }
166          if (obj instanceof String){ sb.append('"').append(escape((String)obj)).append('"'); return; }
167          if (obj instanceof Map){
168              sb.append('{'); boolean first = true;
169              for (Map.Entry<?,?> e : ((Map<?,?>)obj).entrySet()){
170                  if (!first) sb.append(','); first = false;
171                  sb.append('"').append(escape(String.valueOf(e.getKey()))).append('"').append(':');
172                  appendJson(sb, e.getValue());
173              }
174              sb.append('}'); return;
175          }
176          if (obj instanceof Iterable){
177              sb.append('['); boolean first = true;
178              for (Object v : (Iterable<?>)obj){
179                  if (!first) sb.append(','); first = false; appendJson(sb, v);
180              }
181              sb.append(']'); return;
182          }
183          sb.append('"').append(escape(String.valueOf(obj))).append('"');
184      }

186      private static String escape(String s){
187          return s.replace("\\","\\\\").replace("\"","\\\"").replace("\n","\\n");
188      }
