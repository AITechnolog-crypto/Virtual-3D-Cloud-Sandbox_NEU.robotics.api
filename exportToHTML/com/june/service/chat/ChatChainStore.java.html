11   /**
12    * ChatChainStore — sehr kleine, append-only Hash-Chain ("wie Blockchain", ohne Sidechains)
13    * - Speichert je Channel (admin, customer, ...) in data/chat/<channel>.jsonl
14    * - Jeder Eintrag enthält prevHash und hash (SHA-256), gebildet über prevHash+ts+sender+role+text
15    * - Kein Forking/Sidechain: Es wird immer der aktuelle Head als prevHash verwendet
16    */
17   @Component
18   public class ChatChainStore {
19       private static final Path CHAT_DIR = Paths.get("data", "chat");
20   
21       public ChatChainStore(){
22           try { Files.createDirectories(CHAT_DIR); } catch (Exception ignored) {}
23       }

25       public static final class ChatMessage {
26           public String id;           // monotonisch steigend als String
27           public long ts;             // epoch millis
28           public String channel;      // z.B. "admin" oder "customer"
29           public String sender;       // frei: "admin", "assistant", Benutzername, etc.
30           public String role;         // frei: "user" | "assistant" | "system"
31           public String text;         // Nachrichtentext
32           public String prevHash;     // Head vor Append
33           public String hash;         // SHA-256(prevHash+ts+sender+role+text)
34       }

36       public synchronized ChatMessage append(String channel, String sender, String role, String text){
37           Objects.requireNonNull(channel, "channel");
38           Objects.requireNonNull(sender, "sender");
39           Objects.requireNonNull(role, "role");
40           Objects.requireNonNull(text, "text");
41           try {
42               Path file = fileFor(channel);
43               Files.createDirectories(file.getParent());
44               String prev = headHash(file);
45               long ts = System.currentTimeMillis();
46               String id = String.valueOf(nextId(file));
47               String hash = sha256(prev + "|" + ts + "|" + sender + "|" + role + "|" + text);
48               String line = toJsonLine(id, ts, channel, sender, role, text, prev, hash) + "\n";
49               Files.writeString(file, line, StandardCharsets.UTF_8, StandardOpenOption.CREATE, StandardOpenOption.APPEND);
50               ChatMessage m = new ChatMessage();
51               m.id = id; m.ts = ts; m.channel = channel; m.sender = sender; m.role = role; m.text = text; m.prevHash = prev; m.hash = hash;
52               return m;
53           } catch (Exception e){
54               throw new RuntimeException(e);
55           }
56       }

58       public synchronized List<ChatMessage> list(String channel, int limit){
59           try {
60               Path file = fileFor(channel);
61               if (!Files.exists(file)) return Collections.emptyList();
62               List<String> lines = Files.readAllLines(file, StandardCharsets.UTF_8);
63               int start = Math.max(0, lines.size() - Math.max(1, limit));
64               List<ChatMessage> out = new ArrayList<>();
65               for (int i=start; i<lines.size(); i++){
66                   ChatMessage m = parse(lines.get(i));
67                   if (m != null) out.add(m);
68               }
69               return out;
70           } catch (Exception e){
71               return Collections.emptyList();
72           }
73       }

75       public synchronized Map<String,Object> validate(String channel){
76           Map<String,Object> res = new LinkedHashMap<>();
77           try{
78               Path file = fileFor(channel);
79               if (!Files.exists(file)){
80                   res.put("length", 0);
81                   res.put("valid", true);
82                   res.put("headHash", "");
83                   return res;
84               }
85               List<String> lines = Files.readAllLines(file, StandardCharsets.UTF_8);
86               String prev = "";
87               int i = 0;
88               for (String line : lines){
89                   i++;
90                   ChatMessage m = parse(line);
91                   if (m == null){ res.put("valid", false); res.put("errorAt", i); return res; }
92                   String expect = sha256(prev + "|" + m.ts + "|" + m.sender + "|" + m.role + "|" + m.text);
93                   if (!Objects.equals(expect, m.hash)){
94                       res.put("valid", false); res.put("errorAt", i); return res;
95                   }
96                   prev = m.hash;
97               }
98               res.put("length", lines.size());
99               res.put("valid", true);
100              res.put("headHash", prev);
101              return res;
102          }catch(Exception e){
103              res.put("valid", false);
104              res.put("error", e.getMessage());
105              return res;
106          }
107      }

109      private static Path fileFor(String channel){
110          String safe = channel == null ? "default" : channel.toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9_-]", "_");
111          return CHAT_DIR.resolve(safe + ".jsonl");
112      }
113  
114      private static String headHash(Path file){
115          try{
116              if (!Files.exists(file)) return "";
117              List<String> lines = Files.readAllLines(file, StandardCharsets.UTF_8);
118              for (int i=lines.size()-1; i>=0; i--){
119                  ChatMessage m = parse(lines.get(i));
120                  if (m != null) return m.hash == null ? "" : m.hash;
121              }
122              return "";
123          }catch(Exception e){ return ""; }
124      }
125  
126      private static long nextId(Path file){
127          try{
128              if (!Files.exists(file)) return 1L;
129              List<String> lines = Files.readAllLines(file, StandardCharsets.UTF_8);
130              for (int i=lines.size()-1; i>=0; i--){
131                  ChatMessage m = parse(lines.get(i));
132                  if (m != null){
133                      try { return Long.parseLong(m.id) + 1; } catch(Exception ignored) { return lines.size()+1L; }
134                  }
135              }
136              return lines.size()+1L;
137          }catch(Exception e){ return 1L; }
138      }

140      private static ChatMessage parse(String line){
141          if (line == null || line.isBlank()) return null;
142          try{
143              // sehr einfacher Parser: erwartet flache JSON-Schlüssel
144              ChatMessage m = new ChatMessage();
145              m.id = str(line, "id");
146              m.ts = longVal(line, "ts");
147              m.channel = str(line, "channel");
148              m.sender = str(line, "sender");
149              m.role = str(line, "role");
150              m.text = str(line, "text");
151              m.prevHash = str(line, "prevHash");
152              m.hash = str(line, "hash");
153              return m;
154          }catch(Exception e){ return null; }
155      }

157      private static String toJsonLine(String id, long ts, String channel, String sender, String role, String text, String prev, String hash){
158          return "{"+
159                  quote("id")+":"+quote(id)+","+
160                  quote("ts")+":"+ts+","+
161                  quote("channel")+":"+quote(channel)+","+
162                  quote("sender")+":"+quote(sender)+","+
163                  quote("role")+":"+quote(role)+","+
164                  quote("text")+":"+quote(text)+","+
165                  quote("prevHash")+":"+quote(prev)+","+
166                  quote("hash")+":"+quote(hash)+
167                  "}";
168      }

170      private static String quote(String s){
171          return '"' + escape(s) + '"';
172      }
173      private static String escape(String s){
174          if (s == null) return "";
175          return s.replace("\\", "\\\\").replace("\"","\\\"").replace("\n","\\n");
176      }

178      private static String str(String json, String key){
179          int i = json.indexOf('"'+key+'"'); if (i<0) return "";
180          int c = json.indexOf(':', i); if (c<0) return "";
181          int q1 = json.indexOf('"', c+1); if (q1<0) return "";
182          int q2 = json.indexOf('"', q1+1); if (q2<0) return "";
183          return json.substring(q1+1, q2).replace("\\\"","\"").replace("\\n","\n").replace("\\\\","\\");
184      }
185      private static long longVal(String json, String key){
186          int i = json.indexOf('"'+key+'"'); if (i<0) return 0L;
187          int c = json.indexOf(':', i); if (c<0) return 0L;
188          int j=c+1; StringBuilder b=new StringBuilder();
189          while (j<json.length()){
190              char ch=json.charAt(j++);
191              if ((ch>='0'&&ch<='9')) b.append(ch); else if (b.length()>0) break;
192          }
193          try { return Long.parseLong(b.toString()); } catch(Exception e){ return 0L; }
194      }

196      private static String sha256(String s){
197          try{
198              MessageDigest md = MessageDigest.getInstance("SHA-256");
199              byte[] d = md.digest(s.getBytes(StandardCharsets.UTF_8));
200              StringBuilder sb = new StringBuilder();
201              for (byte b : d) sb.append(String.format("%02x", b));
202              return sb.toString();
203          }catch(Exception e){ return ""; }
204      }
