1    package com.june.service.chat;
2    
3    import org.springframework.stereotype.Service;
4    
5    import java.time.Instant;
6    import java.util.*;
7    import java.util.concurrent.ConcurrentHashMap;
8    import java.util.concurrent.ConcurrentLinkedDeque;
9    
10   @Service
11   public class SessionService {
12       private static final int HISTORY_LIMIT = 50;
13   
14       private final Map<String, Boolean> consent = new ConcurrentHashMap<>();
15       private final Map<String, Deque<Map<String, String>>> history = new ConcurrentHashMap<>();
16   
17       // simple rate limit: max 10 messages per 30 seconds per session
18       private final Map<String, Deque<Long>> buckets = new ConcurrentHashMap<>();
19       private static final int RL_MAX = 10;
20       private static final long RL_WINDOW_MS = 30_000L;
21   
22       public void setConsent(String sessionId, boolean ok) {
23           consent.put(sessionId, ok);
24       }
25   
26       public boolean hasConsent(String sessionId) {
27           return consent.getOrDefault(sessionId, false);
28       }
29   
30       public void touch(String sessionId) {
31           consent.putIfAbsent(sessionId, false); // ensure key exists
32       }
33   
34       public boolean isRateLimited(String sessionId) {
35           final long now = System.currentTimeMillis();
36           buckets.putIfAbsent(sessionId, new ConcurrentLinkedDeque<>());
37           Deque<Long> q = buckets.get(sessionId);
38           synchronized (q) {
39               while (!q.isEmpty() && now - q.peekFirst() > RL_WINDOW_MS) {
40                   q.pollFirst();
41               }
42               if (q.size() >= RL_MAX) return true;
43               q.addLast(now);
44           }
45           return false;
46       }
47   
48       public void addMessage(String sessionId, String role, String content) {
49           history.putIfAbsent(sessionId, new ConcurrentLinkedDeque<>());
50           Deque<Map<String, String>> q = history.get(sessionId);
51           synchronized (q) {
52               Map<String, String> m = new HashMap<>();
53               m.put("role", role);
54               m.put("content", content);
55               m.put("ts", Instant.now().toString());
56               q.addLast(m);
57               while (q.size() > HISTORY_LIMIT) q.pollFirst();
58           }
59       }
60   
61       public List<Map<String, String>> getMessages(String sessionId) {
62           Deque<Map<String, String>> q = history.get(sessionId);
63           if (q == null) return Collections.emptyList();
64           return new ArrayList<>(q);
65       }
66   }
