56       public void appendDatasetRow(Map<String,Object> row) throws IOException {
57           if (row == null) return;
58           Map<String,Object> safe = new LinkedHashMap<>(row);
59           safe.putIfAbsent("ts", System.currentTimeMillis());
60           String line = toJson(safe) + "\n";
61           Files.writeString(DATASET_FILE, line, StandardCharsets.UTF_8,
62                   StandardOpenOption.CREATE, StandardOpenOption.APPEND);
63       }

65       // --- simple rate limiter for training ---
66       private final Map<String, Deque<Long>> trainWindows = new ConcurrentHashMap<>();
67   
68       public boolean canTrain(String clientId, int maxPerHour){
69           String key = clientId == null ? "default" : clientId;
70           long now = System.currentTimeMillis();
71           long cutoff = now - 3600_000L;
72           Deque<Long> q = trainWindows.computeIfAbsent(key, k -> new ArrayDeque<>());
73           synchronized (q){
74               while (!q.isEmpty() && q.peekFirst() < cutoff) q.pollFirst();
75               return q.size() < Math.max(1, maxPerHour);
76           }
77       }
78       public void recordTrain(String clientId){
79           String key = clientId == null ? "default" : clientId;
80           Deque<Long> q = trainWindows.computeIfAbsent(key, k -> new ArrayDeque<>());
81           synchronized (q){ q.addLast(System.currentTimeMillis()); }
82       }

85       private static Policy fromJson(String s){
86           Policy p = new Policy();
87           try {
88               // very small/naive parsing by key presence
89               p.trainingEnabled = getBool(s, "trainingEnabled", true);
90               p.ingestEnabled = getBool(s, "ingestEnabled", true);
91               p.mirrorTelemetryToDataset = getBool(s, "mirrorTelemetryToDataset", false);
92               p.retentionDays = (int)getLong(s, "retentionDays", 30);
93               p.maxTrainPerHour = (int)getLong(s, "maxTrainPerHour", 6);
94               p.allowExternalModels = getBool(s, "allowExternalModels", false);
95           } catch (Exception ignored) {}
96           return p;
97       }
