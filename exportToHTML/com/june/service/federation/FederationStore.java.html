11    * File-based federation store for sharing simple model parameters between instances.
12    * - Appends submitted models to data/federation/models.jsonl
13    * - Computes a simple average aggregate written to data/federation/aggregate-energy.json
14    */
15   @Component
16   public class FederationStore {
17       private static final Path FED_DIR = Paths.get("data", "federation");
18       private static final Path MODELS_FILE = FED_DIR.resolve("models.jsonl");
19       private static final Path AGG_FILE = FED_DIR.resolve("aggregate-energy.json");
20   
21       public FederationStore(){
22           try { Files.createDirectories(FED_DIR); } catch (Exception ignored) {}
23       }

25       public synchronized void submit(Map<String, Object> model){
26           Objects.requireNonNull(model, "model");
27           Map<String,Object> copy = new LinkedHashMap<>(model);
28           copy.putIfAbsent("ts", Instant.now().toEpochMilli());
29           try {
30               String json = toJson(copy) + "\n";
31               Files.writeString(MODELS_FILE, json, StandardCharsets.UTF_8,
32                       StandardOpenOption.CREATE, StandardOpenOption.APPEND);
33           } catch (Exception ignored) {}
34       }

36       public synchronized Map<String,Object> aggregate(int max){
37           List<Map<String,Object>> items = readLast(max);
38           if (items.isEmpty()) return Map.of("count", 0);
39           double slopeSum=0, interceptSum=0, min=Double.POSITIVE_INFINITY, maxV=Double.NEGATIVE_INFINITY, avgSum=0;
40           int n=0;
41           for (Map<String,Object> m : items){
42               Double slope = asD(m.get("slope"));
43               Double intercept = asD(m.get("intercept"));
44               Double aMin = asD(m.get("min"));
45               Double aMax = asD(m.get("max"));
46               Double aAvg = asD(m.get("avg"));
47               if (slope!=null && intercept!=null){
48                   slopeSum += slope; interceptSum += intercept; n++;
49               }
50               if (aMin!=null) min = Math.min(min, aMin);
51               if (aMax!=null) maxV = Math.max(maxV, aMax);
52               if (aAvg!=null) avgSum += aAvg;
53           }
54           Map<String,Object> agg = new LinkedHashMap<>();
55           agg.put("ts", Instant.now().toEpochMilli());
56           agg.put("count", items.size());
57           if (n>0){
58               agg.put("slope", slopeSum/n);
59               agg.put("intercept", interceptSum/n);
60           }
61           if (min!=Double.POSITIVE_INFINITY) agg.put("min", min);
62           if (maxV!=Double.NEGATIVE_INFINITY) agg.put("max", maxV);
63           if (items.size()>0) agg.put("avg", avgSum/Math.max(1, items.size()));
64           // persist
65           try { Files.writeString(AGG_FILE, toJson(agg), StandardCharsets.UTF_8, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);} catch(Exception ignored) {}
66           return agg;
67       }

69       public List<Map<String,Object>> list(int limit){
70           return readLast(limit);
71       }

76       private List<Map<String,Object>> readLast(int limit){
77           List<Map<String,Object>> out = new ArrayList<>();
78           try {
79               if (!Files.exists(MODELS_FILE)) return out;
80               List<String> lines = Files.readAllLines(MODELS_FILE, StandardCharsets.UTF_8);
81               int start = Math.max(0, lines.size() - Math.max(1, limit));
82               for (int i=start; i<lines.size(); i++){
83                   String line = lines.get(i);
84                   Map<String,Object> m = parseJsonLine(line);
85                   if (m!=null) out.add(m);
86               }
87           } catch (Exception ignored) {}
88           return out;
89       }

91       private static Double asD(Object o){
92           if (o == null) return null;
93           if (o instanceof Number) return ((Number)o).doubleValue();
94           try { return Double.parseDouble(String.valueOf(o)); } catch(Exception e){ return null; }
95       }

97       private static String toJson(Map<String,Object> map){
98           StringBuilder sb = new StringBuilder(); appendJson(sb, map); return sb.toString();
99       }
100      private static void appendJson(StringBuilder sb, Object obj){
101          if (obj == null){ sb.append("null"); return; }
102          if (obj instanceof Number || obj instanceof Boolean){ sb.append(String.valueOf(obj)); return; }
103          if (obj instanceof String){ sb.append('"').append(escape((String)obj)).append('"'); return; }
104          if (obj instanceof Map){ sb.append('{'); boolean first=true; for (Map.Entry<?,?> e: ((Map<?,?>)obj).entrySet()){ if(!first) sb.append(','); first=false; sb.append('"').append(escape(String.valueOf(e.getKey()))).append('"').append(':'); appendJson(sb, e.getValue()); } sb.append('}'); return; }
105          if (obj instanceof Iterable){ sb.append('['); boolean first=true; for (Object v: (Iterable<?>)obj){ if(!first) sb.append(','); first=false; appendJson(sb, v);} sb.append(']'); return; }
106          sb.append('"').append(escape(String.valueOf(obj))).append('"');
107      }
108      private static String escape(String s){ return s.replace("\\","\\\\").replace("\"","\\\"").replace("\n","\\n"); }

110      @SuppressWarnings("unchecked")
111      private static Map<String,Object> parseJsonLine(String line){
112          if (line == null || line.isBlank()) return null;
113          // very small tolerant parser: expect key:value pairs at top-level
114          Map<String,Object> m = new LinkedHashMap<>();
115          try {
116              String s = line.trim();
117              if (s.startsWith("{") && s.endsWith("}")) s = s.substring(1, s.length()-1);
118              String[] parts = s.split(",");
119              for (String p : parts){
120                  int i = p.indexOf(':'); if (i<=0) continue;
121                  String k = p.substring(0,i).trim().replaceAll("^\"|\"$", "");
122                  String v = p.substring(i+1).trim();
123                  if (v.startsWith("\"") && v.endsWith("\"")) { m.put(k, v.substring(1, v.length()-1)); }
124                  else if (v.equals("null")) { m.put(k, null); }
125                  else if (v.equals("true")||v.equals("false")) { m.put(k, Boolean.valueOf(v)); }
126                  else { try { m.put(k, Double.parseDouble(v.replaceAll("[^0-9eE+\\-\\.]",""))); } catch(Exception e){ m.put(k, v); } }
127              }
128          } catch (Exception ignored) {}
129          return m;
130      }
