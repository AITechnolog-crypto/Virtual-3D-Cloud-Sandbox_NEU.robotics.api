6    public class AutonomousResourceManagementAndFinanceSystem {

8        private final IResourceExtractionSystem resourceExtractionSystem;
9        private final IBlockchainFinancialSystem blockchainFinancialSystem;
10   
11       public AutonomousResourceManagementAndFinanceSystem(IResourceExtractionSystem resourceExtractionSystem,
12                                                           IBlockchainFinancialSystem blockchainFinancialSystem) {
13           this.resourceExtractionSystem = resourceExtractionSystem;
14           this.blockchainFinancialSystem = blockchainFinancialSystem;
15       }
16   
17       public void operateSystem() {
18           try {
19               int extractedResources = resourceExtractionSystem.extractAndProcessResources();
20               int requiredFunds = extractedResources * 10; // Beispielhafte Berechnung der benötigten Mittel
21               blockchainFinancialSystem.allocateFundsForDevelopment(requiredFunds);
22               logger.info("System erfolgreich betrieben.");
23           } catch (ResourceExtractionException | FundingAllocationException e) {
24               logger.log(Level.SEVERE, "Fehler im Systembetrieb: " + e.getMessage(), e);
25           }
26       }
27

32           AutonomousResourceManagementAndFinanceSystem system = new AutonomousResourceManagementAndFinanceSystem(extractionSystem, financialSystem);
