7    public class ResourceExtractionSystem implements IResourceExtractionSystem {
8        private final Logger logger = Logger.getLogger(ResourceExtractionSystem.class.getName());
9        private final String resourceType;

11       public ResourceExtractionSystem(String resourceType) {
12           this.resourceType = resourceType;
13       }
14   
15       @Override
16       public int extractAndProcessResources() throws ResourceExtractionException {
17           try {
18               int extractedAmount = new Random().nextInt(100);
19               logger.info(extractedAmount + " Einheiten von " + resourceType + " wurden extrahiert und verarbeitet.");
20               return extractedAmount;
21           } catch (Exception e) {
22               logger.log(Level.SEVERE, "Fehler bei der Ressourcenextraktion", e);
23               throw new ResourceExtractionException("Fehler bei der Extraktion von " + resourceType, e);
24               
25           }
26       }
27   }
