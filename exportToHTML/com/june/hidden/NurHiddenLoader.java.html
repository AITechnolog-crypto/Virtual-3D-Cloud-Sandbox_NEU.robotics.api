14   /**
15    * NurHiddenLoader — kompakter Loader/Packager für verschlüsselte Java-Payloads.
16    *
17    * Funktionen (Menü):
18    * 1) Quellcode eingeben (vollqualifizierter Klassenname + Source); kompiliert in-memory
19    *    und speichert die Klassenbytes AES‑GCM‑verschlüsselt in payload.bin.
20    * 2) Verschlüsselte payload.bin laden, entschlüsseln, Klasse definieren und instanziieren;
21    *    die Klasse muss das Interface {@link PayloadTask} implementieren; anschließend wird run(ctx) ausgeführt.
22    * 3) Vorliegende .class‑Datei verschlüsselt als payload.bin packen.
23    *
24    * Sicherheit:
25    * - Schlüsselableitung via PBKDF2WithHmacSHA256 aus Passphrase + Geräte‑Fingerprint.
26    * - AES/GCM/NoPadding mit 12‑Byte IV und 128‑Bit Tag. Salt/IV werden im Dateiformat gespeichert.
27    * - Keine Geheimnisse im Code hinterlegt. Dieses Tool ist für lokale Demo-/Entwicklungszwecke gedacht.
28    */
29   public class NurHiddenLoader {
30       static final String PAYLOAD_FILE = "payload.bin";
31       static final int SALT_LEN = 16;
32       static final int IV_LEN = 12;
33       static final int GCM_TAG_BITS = 128;
34       static final int PBKDF2_ITERS = 120_000;
35       static final int KEY_BITS = 256;
36   
37       public static void main(String[] args) throws Exception {
38           BufferedReader br = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.UTF_8));
39           println("NurHiddenLoader bereit");
40           println("1 Quellcode eingeben und als verschluesselte Payload speichern");
41           println("2 Verschluesselte Payload laden und ausfuehren");
42           println("3 Payload aus Klassenbytecode packen Dateiweg angeben");
43           print("Auswahl: ");
44           String choice = br.readLine();
45           if ("1".equals(choice)) {
46               handleSourceToPayload(br);
47           } else if ("2".equals(choice)) {
48               handleRunPayload(br);
49           } else if ("3".equals(choice)) {
50               handleClassFileToPayload(br);
51           } else {
52               println("Abbruch");
53           }
54       }

56       private static void handleSourceToPayload(BufferedReader br) throws Exception {
57           println("Vollqualifizierten Klassennamen angeben Beispiel: my.pkg.HiddenPayload");
58           print("ClassName: ");
59           String className = br.readLine().trim();
60           println("Source eingeben Ende mit Zeile ::END");
61           String source = readUntilEnd(br);
62   
63           println("Passphrase fuer Verschluesselung eingeben wird nicht angezeigt");
64           String pass = readPassword(br);
65   
66           Map<String, byte[]> compiled = SourceCompiler.compileInMemory(className, source);
67           if (!compiled.containsKey(className)) {
68               println("Fehler Kompilation lieferte keine Bytes fuer " + className);
69               return;
70           }
71           byte[] classBytes = compiled.get(className);
72           writeEncryptedPayload(PAYLOAD_FILE, className, classBytes, pass);
73           println("Payload gespeichert in " + PAYLOAD_FILE);
74       }

76       private static void handleClassFileToPayload(BufferedReader br) throws Exception {
77           println("Pfad zu .class angeben und vollqualifizierten Namen z.B. build/classes/java/main/my/pkg/HiddenPayload.class");
78           print("Pfad: ");
79           String path = br.readLine().trim();
80           print("ClassName: ");
81           String className = br.readLine().trim();
82           println("Passphrase fuer Verschluesselung eingeben wird nicht angezeigt");
83           String pass = readPassword(br);
84           byte[] classBytes;
85           try (FileInputStream fis = new FileInputStream(path)) {
86               classBytes = readAll(fis);
87           }
88           writeEncryptedPayload(PAYLOAD_FILE, className, classBytes, pass);
89           println("Payload gespeichert in " + PAYLOAD_FILE);
90       }

92       private static void handleRunPayload(BufferedReader br) throws Exception {
93           File f = new File(PAYLOAD_FILE);
94           if (!f.exists()) {
95               println("Keine payload.bin gefunden zuerst erstellen");
96               return;
97           }
98           println("Passphrase eingeben um Payload zu laden wird nicht angezeigt");
99           String pass = readPassword(br);
100          EncryptedClassLoader loader = new EncryptedClassLoader(pass);
101          Class<?> payloadClass = loader.loadEncrypted(PAYLOAD_FILE);
102          Object task = payloadClass.getDeclaredConstructor().newInstance();
103          if (!(task instanceof PayloadTask)) {
104              println("Geladene Klasse implementiert nicht PayloadTask");
105              return;
106          }
107          PayloadTask p = (PayloadTask) task;
108          Map<String, Object> ctx = new HashMap<>();
109          ctx.put("timestamp", System.currentTimeMillis());
110          ctx.put("device", DeviceFingerprint.fingerprint());
111          p.run(ctx);
112          println("Payload ausgefuehrt");
113      }

115      private static void writeEncryptedPayload(String outPath, String className, byte[] classBytes, String pass) throws Exception {
116          byte[] salt = new byte[SALT_LEN];
117          byte[] iv = new byte[IV_LEN];
118          SecureRandom sr = new SecureRandom();
119          sr.nextBytes(salt);
120          sr.nextBytes(iv);
121  
122          byte[] key = deriveKey(pass, salt);
123          byte[] ciphertext = encrypt(classBytes, key, iv);
124  
125          byte[] nameBytes = className.getBytes(StandardCharsets.UTF_8);
126          ByteArrayOutputStream bos = new ByteArrayOutputStream();
127          bos.write(new byte[]{'N','H','V','8'});
128          bos.write(salt);
129          bos.write(iv);
130          bos.write(shortToBytes((short) nameBytes.length));
131          bos.write(nameBytes);
132          bos.write(ciphertext);
133          try (FileOutputStream fos = new FileOutputStream(outPath)) {
134              fos.write(bos.toByteArray());
135          }
136          Arrays.fill(key, (byte) 0);
137      }

139      private static byte[] encrypt(byte[] plain, byte[] key, byte[] iv) throws Exception {
140          Cipher c = Cipher.getInstance("AES/GCM/NoPadding");
141          SecretKeySpec sk = new SecretKeySpec(key, "AES");
142          c.init(Cipher.ENCRYPT_MODE, sk, new GCMParameterSpec(GCM_TAG_BITS, iv));
143          return c.doFinal(plain);
144      }
145  
146      static byte[] decrypt(byte[] cipher, byte[] key, byte[] iv) throws Exception {
147          Cipher c = Cipher.getInstance("AES/GCM/NoPadding");
148          SecretKeySpec sk = new SecretKeySpec(key, "AES");
149          c.init(Cipher.DECRYPT_MODE, sk, new GCMParameterSpec(GCM_TAG_BITS, iv));
150          return c.doFinal(cipher);
151      }

153      private static byte[] deriveKey(String pass, byte[] salt) throws Exception {
154          String dev = DeviceFingerprint.fingerprint();
155          char[] chars = (pass + "|" + dev).toCharArray();
156          try {
157              SecretKeyFactory f = SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256");
158              KeySpec spec = new PBEKeySpec(chars, salt, PBKDF2_ITERS, KEY_BITS);
159              return f.generateSecret(spec).getEncoded();
160          } finally {
161              Arrays.fill(chars, '\0');
162          }
163      }

165      private static String readUntilEnd(BufferedReader br) throws IOException {
166          StringBuilder sb = new StringBuilder();
167          while (true) {
168              String line = br.readLine();
169              if (line == null) break;
170              if ("::END".equals(line)) break;
171              sb.append(line).append('\n');
172          }
173          return sb.toString();
174      }

176      private static String readPassword(BufferedReader br) throws IOException {
177          if (System.console() != null) {
178              char[] c = System.console().readPassword();
179              return new String(c);
180          }
181          return br.readLine();
182      }

184      private static byte[] shortToBytes(short s) {
185          return new byte[]{(byte)((s >> 8) & 0xFF), (byte)(s & 0xFF)};
186      }
187  
188      private static short bytesToShort(byte[] b, int off) {
189          return (short) (((b[off] & 0xFF) << 8) | (b[off+1] & 0xFF));
190      }

192      private static byte[] readAll(InputStream in) throws IOException {
193          ByteArrayOutputStream bos = new ByteArrayOutputStream();
194          byte[] buf = new byte[8192];
195          int r;
196          while ((r = in.read(buf)) != -1) bos.write(buf, 0, r);
197          return bos.toByteArray();
198      }

203      static class EncryptedClassLoader extends ClassLoader {
204          private final String pass;
205          EncryptedClassLoader(String pass) { super(NurHiddenLoader.class.getClassLoader()); this.pass = pass; }
206  
207          Class<?> loadEncrypted(String path) throws Exception {
208              byte[] file;
209              try (FileInputStream fis = new FileInputStream(path)) {
210                  file = readAll(fis);
211              }
212              if (file.length < 4 + SALT_LEN + IV_LEN + 2) throw new IOException("Payload zu klein");
213              if (!(file[0]=='N' && file[1]=='H' && file[2]=='V' && file[3]=='8')) throw new IOException("Header ungueltig");
214              int off = 4;
215              byte[] salt = Arrays.copyOfRange(file, off, off+SALT_LEN); off += SALT_LEN;
216              byte[] iv = Arrays.copyOfRange(file, off, off+IV_LEN); off += IV_LEN;
217              short nameLen = bytesToShort(file, off); off += 2;
218              byte[] nameBytes = Arrays.copyOfRange(file, off, off+nameLen); off += nameLen;
219              String className = new String(nameBytes, StandardCharsets.UTF_8);
220              byte[] cipher = Arrays.copyOfRange(file, off, file.length);
221              byte[] key = deriveKey(pass, salt);
222              byte[] plain = decrypt(cipher, key, iv);
223              Arrays.fill(key, (byte)0);
224              return defineClass(className, plain, 0, plain.length);
225          }
226      }
