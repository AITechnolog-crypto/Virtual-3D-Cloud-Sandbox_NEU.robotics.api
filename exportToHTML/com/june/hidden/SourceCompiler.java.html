9     * In‑Memory Java Source Compiler util.

11   public final class SourceCompiler {
12       private SourceCompiler() {}
13   
14       public static Map<String, byte[]> compileInMemory(String className, String source) throws Exception {
15           JavaCompiler compiler = ToolProvider.getSystemJavaCompiler();
16           if (compiler == null) throw new IllegalStateException("Kein InMemory Compiler verfuegbar JDK erforderlich");
17           StandardJavaFileManager std = compiler.getStandardFileManager(null, null, StandardCharsets.UTF_8);
18           InMemoryFileManager mem = new InMemoryFileManager(std);
19           List<JavaFileObject> units = new ArrayList<>();
20           units.add(new StringJavaFileObject(className, source));
21           List<String> opts = Arrays.asList("-encoding","UTF-8","-Xlint:deprecation","-Xlint:unchecked");
22           JavaCompiler.CompilationTask task = compiler.getTask(null, mem, null, opts, null, units);
23           boolean ok = task.call();
24           if (!ok) throw new IllegalStateException("Kompilierung fehlgeschlagen");
25           return mem.getClassBytes();
26       }
27   }

29   /** String‑backed JavaFileObject */
30   class StringJavaFileObject extends SimpleJavaFileObject {
31       private final String code;
32       StringJavaFileObject(String className, String code) {
33           super(uriForClass(className), Kind.SOURCE);
34           this.code = code;
35       }
36       @Override public CharSequence getCharContent(boolean ignoreEncodingErrors){ return code; }
37       private static URI uriForClass(String className) {
38           return URI.create("string:///" + className.replace('.','/') + Kind.SOURCE.extension);
39       }
40   }

42   /** Captures compiled class bytes */
43   class ByteArrayJavaFileObject extends SimpleJavaFileObject {
44       private final java.io.ByteArrayOutputStream bos = new java.io.ByteArrayOutputStream();
45       ByteArrayJavaFileObject(String name, Kind kind) {
46           super(URI.create("mem:///" + name.replace('.','/') + kind.extension), kind);
47       }
48       @Override public java.io.OutputStream openOutputStream() { return bos; }
49       byte[] toByteArray(){ return bos.toByteArray(); }
50   }

52   /** FileManager storing compiled classes in memory */
53   class InMemoryFileManager extends ForwardingJavaFileManager<JavaFileManager> {
54       private final Map<String, ByteArrayJavaFileObject> compiled = new HashMap<>();
55       InMemoryFileManager(JavaFileManager fileManager) { super(fileManager); }
56       @Override
57       public JavaFileObject getJavaFileForOutput(Location location, String className, JavaFileObject.Kind kind, FileObject sibling){
58           ByteArrayJavaFileObject out = new ByteArrayJavaFileObject(className, kind);
59           compiled.put(className, out);
60           return out;
61       }
62       Map<String, byte[]> getClassBytes(){
63           Map<String, byte[]> m = new HashMap<>();
64           for (Map.Entry<String, ByteArrayJavaFileObject> e : compiled.entrySet()) {
65               m.put(e.getKey(), e.getValue().toByteArray());
66           }
67           return m;
68       }
