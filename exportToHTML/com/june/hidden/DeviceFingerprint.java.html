3    /**
4     * Einfache Geräte-Fingerprint-Hilfsklasse.
5     * Nutzt nur harmlose System-Properties und bildet einen stabilen Hash ab.
6     */
7    public final class DeviceFingerprint {
8        private DeviceFingerprint() {}
9    
10       public static String fingerprint() {
11           String os = System.getProperty("os.name", "");
12           String arch = System.getProperty("os.arch", "");
13           String user = System.getProperty("user.name", "");
14           String home = System.getProperty("user.home", "");
15           String java = System.getProperty("java.version", "");
16           String t = os + "|" + arch + "|" + user + "|" + home + "|" + java;
17           return Integer.toHexString(t.hashCode());
18       }
19   }
