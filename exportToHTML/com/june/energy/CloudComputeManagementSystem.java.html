3    import java.util.ArrayList;
4    import java.util.List;
5    import java.util.Map;

7    /**
8     * CloudComputeManagementSystem
9     *
10    * Analyse & Verbesserungsvorschläge:
11    * - Entkopplung/Interfaces: Kollektoren (IEnergyCollector), Speicher (IEnergyStorage) und Compute-Center (IComputeCenter)
12    *   sind über Schnittstellen abstrahiert. So lassen sich Implementierungen austauschen (z. B. andere Energiequellen, verteilte Speicher,
13    *   echte Datacenter-Backends).
14    * - Datenmodell: Energy ist immutable, somit threadsicherer bei Weitergabe zwischen Komponenten.
15    * - Speicher: GlobalEnergyStorageSystem aggregiert Energiemengen pro Typ. Für höhere Lasten wäre eine
16    *   concurrent-Datenstruktur oder Actor/Message-Ansatz denkbar.
17    * - Fehlerbehandlung: collect/store/process sind aktuell „best-effort“. In der Praxis sollte Logging, Retry,
18    *   Telemetrie/Metriken (z. B. Micrometer), sowie Circuit-Breaker eingesetzt werden.
19    * - Konfiguration: Werte (z. B. Kollektorstärken) hart codiert – künftig extern konfigurierbar machen (Properties/DB/API).
20    * - Skalierung: manageSystem() läuft synchron. Bei realen Workloads: Scheduler/Executor, Backpressure, asynchrone Pipelines.
21    * - Testbarkeit: Jede Klasse ist klein und testbar. Unit-Tests könnten leicht hinzugefügt werden.
22    */
23   public class CloudComputeManagementSystem {
24       private final List<IEnergyCollector> energyCollectors;
25       private final IEnergyStorage energyStorage;
26       private final List<IComputeCenter> computeCenters;
27       private final SecretTransportRailway secretRailway;
28   
29       public CloudComputeManagementSystem() {
30           this.energyCollectors = new ArrayList<>();
31           this.computeCenters = new ArrayList<>();
32           // Initialisierung des Energiespeichers und der geheimen Transportbahn
33           this.energyStorage = new GlobalEnergyStorageSystem();
34           this.secretRailway = new SecretTransportRailway();
35       }
36   
37       public void addEnergyCollector(IEnergyCollector collector) {
38           if (collector != null) energyCollectors.add(collector);
39       }
40   
41       public void addComputeCenter(IComputeCenter computeCenter) {
42           if (computeCenter != null) computeCenters.add(computeCenter);
43       }
44   
45       public void manageSystem() {
46           // Sammeln und Speichern von Energie
47           for (IEnergyCollector collector : energyCollectors) {
48               try {
49                   Energy energy = collector.collectEnergy();
50                   energyStorage.storeEnergy(energy);
51               } catch (RuntimeException ex) {
52                   System.err.println("Fehler beim Sammeln/Speichern von Energie: " + ex.getMessage());
53               }
54           }
55   
56           // Speicherstände ausgeben (nur Demo)
57           if (energyStorage instanceof GlobalEnergyStorageSystem ges) {
58               Map<EnergyType, Double> snap = ges.snapshot();
59               System.out.println("Energie-Snapshot: " + snap + ", Total=" + ges.getTotal());
60           }
61   
62           // Betrieb der Rechenzentren
63           for (IComputeCenter center : computeCenters) {
64               try {
65                   center.processTasks();
66               } catch (RuntimeException ex) {
67                   System.err.println("Fehler bei der Auftragsverarbeitung: " + ex.getMessage());
68               }
69           }
70   
71           // Überprüfung auf Notfälle und Aktivierung der geheimen Transportbahn
72           try {
73               if (secretRailway.checkForEmergencies()) {
74                   secretRailway.activate();
75               }
76           } catch (RuntimeException ex) {
77               System.err.println("Fehler bei Notfallprüfung/Transportbahn: " + ex.getMessage());
78           }
79       }
80   
81       // Hauptprogramm (Demo)
82       public static void main(String[] args) {
83           CloudComputeManagementSystem managementSystem = new CloudComputeManagementSystem();
84   
85           // Hinzufügen von Energiekollektoren
86           managementSystem.addEnergyCollector(new SolarEnergyCollector());
87           managementSystem.addEnergyCollector(new CO2EnergyCollector());
88   
89           // Hinzufügen von Rechenzentren an verschiedenen Standorten
90           managementSystem.addComputeCenter(new ComputeCenter("Antarctica"));
91           managementSystem.addComputeCenter(new ComputeCenter("Sahara"));
92   
93           // Starten des Managementsystems
94           managementSystem.manageSystem();
95       }
96   }
