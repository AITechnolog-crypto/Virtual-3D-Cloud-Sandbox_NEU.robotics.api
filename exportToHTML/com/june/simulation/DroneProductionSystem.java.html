1    package com.june.simulation;
2    
3    import java.util.concurrent.ExecutorService;
4    import java.util.concurrent.Executors;
5    import java.util.concurrent.TimeUnit;
6    import java.util.concurrent.atomic.AtomicLong;
7    import java.util.logging.Level;
8    import java.util.logging.Logger;
9    
10   /**
11    * Multithreaded Drone Production demo.
12    *
13    * Behavior mirrors the provided issue description: uses a fixed thread pool sized to available
14    * processors, increments a global AtomicLong counter, simulates production latency (10ms),
15    * logs progress per production line, and shuts down gracefully.
16    *
17    * This class is self-contained and does not depend on Spring components.
18    */
19   public class DroneProductionSystem {
20       private final Logger logger = Logger.getLogger(DroneProductionSystem.class.getName());
21       private final ExecutorService productionService;
22       private final AtomicLong droneCount;
23       private final int numberOfCores;
24   
25       public DroneProductionSystem() {
26           this.numberOfCores = Runtime.getRuntime().availableProcessors();
27           this.productionService = Executors.newFixedThreadPool(numberOfCores);
28           this.droneCount = new AtomicLong(0);
29           logger.info("Starte Drohnenproduktion mit " + numberOfCores + " Produktionslinien.");
30       }
31   
32       public void startProduction(long targetDroneCount) {
33           for (int i = 0; i < numberOfCores; i++) {
34               productionService.execute(() -> {
35                   long localCount = 0;
36                   while (droneCount.get() < targetDroneCount) {
37                       droneCount.incrementAndGet();
38                       localCount++;
39                       try {
40                           // Simuliere die Produktion einer Drohne (10ms Pause)
41                           Thread.sleep(10);
42                       } catch (InterruptedException e) {
43                           logger.log(Level.WARNING, "Thread wurde unterbrochen", e);
44                           Thread.currentThread().interrupt();
45                           break;
46                       }
47                       if (localCount % 10000 == 0) {
48                           logger.info("Produktionslinie hat " + localCount + " Drohnen produziert.");
49                       }
50                   }
51                   logger.info("Produktionslinie beendet. Insgesamt " + localCount + " Drohnen produziert.");
52               });
53           }
54           try {
55               productionService.shutdown();
56               productionService.awaitTermination(Long.MAX_VALUE, TimeUnit.NANOSECONDS);
57           } catch (InterruptedException e) {
58               logger.log(Level.SEVERE, "Das Herunterfahren des ExecutorService wurde unterbrochen", e);
59               Thread.currentThread().interrupt();
60           }
61           logger.info("Produktion abgeschlossen. Insgesamt " + droneCount.get() + " Drohnen produziert.");
62       }
63   
64       public static void main(String[] args) {
65           DroneProductionSystem system = new DroneProductionSystem();
66           system.startProduction(1_000_000L); // Ziel von 1 Million Drohnen (angepasst für Testzwecke)
67       }
68   }
