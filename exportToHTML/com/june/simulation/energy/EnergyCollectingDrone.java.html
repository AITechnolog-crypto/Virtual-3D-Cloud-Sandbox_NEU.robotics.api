1    package com.june.simulation.energy;
2    
3    import java.util.logging.Level;
4    import java.util.logging.Logger;
5    
6    /**
7     * Energie-sammelnde Drohne: setzt Drohnen ein und speichert gesammelte Energie
8     * in einem bereitgestellten Energiespeicher.
9     */
10   public class EnergyCollectingDrone extends AutonomousDroneSystem {
11       private static final Logger logger = Logger.getLogger(EnergyCollectingDrone.class.getName());
12       private final IEnergyStorageV2 energyStorage;
13   
14       public EnergyCollectingDrone(IResourceManagement resourceManagement, IEnergyStorageV2 energyStorage) {
15           super(resourceManagement);
16           this.energyStorage = energyStorage;
17       }
18   
19       @Override
20       public void deployDronesForGood() {
21           super.deployDronesForGood();
22           collectAndStoreEnergy();
23       }
24   
25       void collectAndStoreEnergy() {
26           try {
27               // Beispielhaftes Sammeln von Energie (Dummy-Implementierung)
28               Energy collectedEnergy = new Energy(50, "kWh");
29               if (energyStorage.storeEnergy(collectedEnergy)) {
30                   logger.info("Energie erfolgreich gespeichert.");
31               }
32           } catch (Exception e) {
33               logger.log(Level.SEVERE, "Fehler beim Sammeln oder Speichern von Energie", e);
34           }
35       }
36   
37       public static void main(String[] args) {
38           IEnergyStorageV2 storage = new AntarcticEnergyStorage(1_000);
39           IResourceManagement resourceManagement = new ResourceManagement();
40           EnergyCollectingDrone drone = new EnergyCollectingDrone(resourceManagement, storage);
41           drone.deployDronesForGood();
42   
43           // Zusätzliche Sammlungen gemäß Beispiel
44           drone.collectAndStoreEnergy();
45           drone.collectAndStoreEnergy();
46           drone.collectAndStoreEnergy();
47       }
48   }
