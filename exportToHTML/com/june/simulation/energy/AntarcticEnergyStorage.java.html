1    package com.june.simulation.energy;
2    
3    /**
4     * Energiespeicher-Implementierung für die Antarktis mit Kapazitätsprüfung.
5     */
6    public class AntarcticEnergyStorage extends AbstractEnergyStorage {
7        public AntarcticEnergyStorage(double maxCapacity) {
8            super(maxCapacity);
9        }
10   
11       @Override
12       public boolean storeEnergy(Energy energy) {
13           if (energy == null || !Double.isFinite(energy.amount) || energy.amount <= 0) {
14               return false;
15           }
16           if (currentCapacity + energy.amount <= maxCapacity) {
17               currentCapacity += energy.amount;
18               logger.info("Energie in der Antarktis gespeichert: " + energy +
19                       ". Aktuelle Kapazität: " + currentCapacity + "/" + maxCapacity);
20               return true;
21           } else {
22               logger.warning("Nicht genügend Speicherkapazität in der Antarktis für: " + energy +
23                       ". Aktuell: " + currentCapacity + ", Max: " + maxCapacity);
24               return false;
25           }
26       }
27   }
