1    package com.june.simulation;
2    
3    import java.util.concurrent.*;
4    import java.util.concurrent.atomic.LongAdder;
5    import java.util.logging.Level;
6    import java.util.logging.Logger;
7    
8    public class DroneProductionSystemV2 {
9        private static final Logger logger = Logger.getLogger(DroneProductionSystemV2.class.getName());
10       private static final long TARGET_DRONE_COUNT = 1_000_000_000L;
11       private final LongAdder producedDrones = new LongAdder();
12       private final ExecutorService productionLine = Executors.newFixedThreadPool(10);
13       private volatile boolean productionRunning = true; // Flag zum Stoppen der Produktion
14   
15       public void startProduction() {
16           for (int i = 0; i < 10; i++) {
17               productionLine.execute(this::produceDrone);
18           }
19       }
20   
21       private void produceDrone() {
22           while (productionRunning && producedDrones.sum() < TARGET_DRONE_COUNT) {
23               try {
24                   // Simulierte Produktionszeit
25                   Thread.sleep(1);
26                   producedDrones.increment();
27               } catch (InterruptedException e) {
28                   logger.log(Level.WARNING, "Produktion unterbrochen.", e);
29                   Thread.currentThread().interrupt(); // Interrupt-Flag wieder setzen
30                   productionRunning = false; // Produktion stoppen
31               }
32           }
33       }
34   
35       public void stopProduction() {
36           productionRunning = false;
37           productionLine.shutdown(); // Initiates an orderly shutdown
38           try {
39               if (!productionLine.awaitTermination(60, TimeUnit.SECONDS)) {
40                   productionLine.shutdownNow(); // Forcefully shutdown
41                   if (!productionLine.awaitTermination(60, TimeUnit.SECONDS))
42                       System.err.println("Pool did not terminate");
43               }
44           } catch (InterruptedException ex) {
45               productionLine.shutdownNow();
46               Thread.currentThread().interrupt();
47           }
48       }
49   
50       public void printProductionStatus() {
51           while (producedDrones.sum() < TARGET_DRONE_COUNT && productionRunning) {
52               try {
53                   Thread.sleep(1000); // Warte 1 Sekunde
54                   long currentCount = producedDrones.sum();
55                   double progress = (double) currentCount / TARGET_DRONE_COUNT * 100;
56                   logger.info("Produktionsfortschritt: " + currentCount + " / " + TARGET_DRONE_COUNT + " (" + String.format("%.2f", progress) + "%)");
57               } catch (InterruptedException e) {
58                   logger.log(Level.WARNING, "Statusanzeige unterbrochen.", e);
59                   Thread.currentThread().interrupt();
60                   return;
61               }
62           }
63           logger.info("Produktion abgeschlossen. Insgesamt produzierte Drohnen: " + producedDrones.sum());
64       }
65   
66       public static void main(String[] args) throws InterruptedException {
67           DroneProductionSystemV2 productionSystem = new DroneProductionSystemV2();
68           productionSystem.startProduction();
69           productionSystem.printProductionStatus();
70           Thread.sleep(5000);
71           productionSystem.stopProduction();
72       }
73   }
