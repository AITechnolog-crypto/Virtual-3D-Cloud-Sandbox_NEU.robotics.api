1    package com.june.collection2;

3    import java.util.ArrayList;
4    import java.util.List;
5    import java.util.logging.Level;
6    import java.util.logging.Logger;
7    
8    public class AutonomousEnergyCollectionSystem {
9        private final Logger logger = Logger.getLogger(AutonomousEnergyCollectionSystem.class.getName());
10       private final List<IEnergyCollector> energyCollectors;
11       private final IEnergyStorage energyStorage;
12       private final ISatelliteCommunicator satelliteCommunicator;
13   
14       public AutonomousEnergyCollectionSystem(IEnergyStorage energyStorage, ISatelliteCommunicator satelliteCommunicator) {
15           this.energyCollectors = new ArrayList<>();
16           this.energyStorage = energyStorage;
17           this.satelliteCommunicator = satelliteCommunicator;
18       }
19   
20       public void addEnergyCollector(IEnergyCollector collector) {
21           energyCollectors.add(collector);
22       }
23   
24       public void collectAndStoreEnergy() {
25           try {
26               satelliteCommunicator.receiveCommands();
27               for (IEnergyCollector collector : energyCollectors) {
28                   Energy collectedEnergy = collector.collectEnergy();
29                   energyStorage.storeEnergy(collectedEnergy);
30               }
31           } catch (Exception e) {
32               logger.log(Level.SEVERE, "Fehler beim Sammeln und Speichern von Energie", e);
33           }
34       }
35   
36       public static void main(String[] args) {
37           IEnergyStorage energyStorage = new EnergyStorage();
38           ISatelliteCommunicator satelliteCommunicator = new SatelliteCommunicator();
39           AutonomousEnergyCollectionSystem energySystem = new AutonomousEnergyCollectionSystem(energyStorage, satelliteCommunicator);
40   
41           energySystem.addEnergyCollector(new SolarEnergyCollector(100));
42           energySystem.addEnergyCollector(new CO2EnergyCollector(50));
43   
44           energySystem.collectAndStoreEnergy();
45           energySystem.collectAndStoreEnergy();
46       }
47   }
