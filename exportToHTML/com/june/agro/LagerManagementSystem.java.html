1    package com.june.agro;

3    import java.util.*;
4    
5    /**
6     * Hauptklasse für das Management des Lagerungssystems (Demo/Main).
7     *
8     * Hinweis: Diese Demo ist unabhängig vom bestehenden Spring Boot Projekt
9     * und kann separat über die Main-Methode gestartet werden.
10    */
11   public class LagerManagementSystem {
12       private ILagerSystem<LandwirtschaftsProdukt> lagerSystem;
13   
14       public LagerManagementSystem() {
15           lagerSystem = new GlobalesLagerSystem<>();
16       }
17   
18       public void manageLager() {
19           List<LandwirtschaftsProdukt> produkte = Arrays.asList(
20                   new LandwirtschaftsProdukt("Korn", 10000),
21                   new LandwirtschaftsProdukt("Mais", 5000),
22                   new LandwirtschaftsProdukt("Korn", 2000) // Fügt weitere 2000 Einheiten Korn hinzu
23           );
24   
25           lagerSystem.lagereProdukte(produkte);
26   
27           try {
28               List<LandwirtschaftsProdukt> kornProdukte = lagerSystem.getProdukte("Korn");
29               System.out.println("Gelagerte Korn-Produkte:");
30               kornProdukte.forEach(System.out::println);
31   
32               System.out.println(lagerSystem.getLagerBestand());
33   
34               // Erwartet: wirft eine ProduktNichtGefundenException
35               lagerSystem.getProdukte("Weizen");
36           } catch (ProduktNichtGefundenException e) {
37               System.err.println("Fehler: " + e.getMessage());
38           }
39       }
40   
41       public static void main(String[] args) {
42           LagerManagementSystem managementSystem = new LagerManagementSystem();
43           managementSystem.manageLager();
44       }
45   }
