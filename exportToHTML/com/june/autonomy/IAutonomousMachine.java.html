1    package com.june.autonomy;
2    
3    import com.june.autonomy.action.IAction;
4    
5    /**
6     * Schnittstelle für eine autonome Maschine mit positiver/nicht-negativer Ausrichtung.
7     */
8    public interface IAutonomousMachine {
9        /** Führt positive Aktionen aus (z. B. Hilfs- oder Schutzfunktionen). */
10       void performPositiveActions();
11   
12       /** Verhindert negative Aktionen (z. B. Schadensvermeidung, Safeguards). */
13       void preventNegativeActions();
14   
15       /**
16        * Setzt den Aggressivitätsgrad im Bereich [0, 1].
17        * 0 = defensiv/risikoarm, 1 = sehr aggressiv/risikoreich.
18        */
19       void setAggressiveness(double aggressiveness);
20   
21       /**
22        * Neue überladene Variante: Führt genau eine Aktion aus, sofern sie nicht negativ ist.
23        * Default-Implementierung ruft bei nicht-negativer Aktion die Alt-API performPositiveActions() auf,
24        * um Rückwärtskompatibilität zu gewährleisten.
25        */
26       default void performPositiveActions(IAction action) {
27           if (action == null || action.isNegative()) {
28               return; // NOP bei fehlender oder negativer Aktion
29           }
30           performPositiveActions();
31       }
32   }
