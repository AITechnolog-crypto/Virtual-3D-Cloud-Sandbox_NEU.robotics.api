1    package com.june.integrated;
2    
3    import java.util.ArrayList;
4    import java.util.List;
5    import java.util.Objects;
6    import java.util.logging.Level;
7    import java.util.logging.Logger;
8    
9    /**
10    * Integriertes Technologiesystem: orchestriert alte und moderne Technologie
11    * und nutzt ein ML-Modell, um basierend auf Sensordaten Entscheidungen zu treffen.
12    */
13   public class IntegratedTechnologySystem {
14       private static final Logger logger = Logger.getLogger(IntegratedTechnologySystem.class.getName());
15   
16       private final IAncientTechnology ancientTech;
17       private final IModernTechnology modernTech;
18       private final IMachineLearningModel predictionModel;
19   
20       public IntegratedTechnologySystem(IAncientTechnology ancientTech,
21                                         IModernTechnology modernTech,
22                                         IMachineLearningModel predictionModel) {
23           this.ancientTech = Objects.requireNonNull(ancientTech, "ancientTech");
24           this.modernTech = Objects.requireNonNull(modernTech, "modernTech");
25           this.predictionModel = Objects.requireNonNull(predictionModel, "predictionModel");
26       }
27   
28       public void integrateAndOperate() {
29           try {
30               ancientTech.performAncientTasks();
31               modernTech.performModernTasks();
32   
33               List<SensorData> historicalData = getHistoricalSensorData();
34               predictionModel.train(historicalData);
35               SensorData currentData = getCurrentSensorData();
36               Prediction prediction = predictionModel.makePrediction(currentData);
37   
38               executeBasedOnPrediction(prediction);
39           } catch (Exception e) {
40               logger.log(Level.SEVERE, "Fehler im Systembetrieb: ", e);
41           }
42       }
43   
44       private List<SensorData> getHistoricalSensorData() {
45           List<SensorData> data = new ArrayList<>();
46           data.add(new SensorData(20, 1000));
47           data.add(new SensorData(22, 1010));
48           return data;
49       }
50   
51       private SensorData getCurrentSensorData() {
52           return new SensorData(21, 1005);
53       }
54   
55       private void executeBasedOnPrediction(Prediction prediction) {
56           logger.info("Ausführung von Aktionen basierend auf der Vorhersage: " + prediction);
57           if (prediction != null && prediction.getPredictedValue() > 25) {
58               logger.info("Warnung: Hoher Vorhersagewert!");
59           }
60       }
61   }
