1    package com.june.integrated;
2    
3    import java.util.List;
4    import java.util.logging.Level;
5    import java.util.logging.Logger;
6    
7    /**
8     * Dummy-ML-Modell: Trainiert einmal auf historischen Sensordaten und
9     * sagt als Vorhersage den Durchschnitt der Temperatur voraus.
10    * Entspricht der Spezifikation im Issue-Text.
11    */
12   public class AverageTemperatureModel implements IMachineLearningModel {
13       private static final Logger logger = Logger.getLogger(AverageTemperatureModel.class.getName());
14   
15       private boolean isTrained = false;
16       private double averageValue = 0.0;
17   
18       @Override
19       public void train(List<SensorData> trainingData) {
20           if (isTrained) {
21               logger.info("Modell ist bereits trainiert.");
22               return;
23           }
24           if (trainingData == null || trainingData.isEmpty()) {
25               logger.info("Keine Trainingsdaten vorhanden.");
26               return;
27           }
28           logger.info("Training des maschinellen Lernmodells mit Sensordaten.");
29           double sum = 0.0;
30           int n = 0;
31           for (SensorData d : trainingData) {
32               if (d == null) continue;
33               sum += d.getTemperature();
34               n++;
35           }
36           if (n > 0) {
37               averageValue = sum / n;
38               isTrained = true;
39               logger.log(Level.FINE, "Durchschnitt berechnet: {0}", averageValue);
40           } else {
41               logger.info("Keine gültigen Trainingsdatensätze.");
42           }
43       }
44   
45       @Override
46       public Prediction makePrediction(SensorData currentData) {
47           logger.info("Erstellung einer Vorhersage basierend auf aktuellen Sensordaten: " + currentData);
48           if (!isTrained) {
49               logger.warning("Modell ist nicht trainiert. Es wird ein Standardwert zurückgegeben.");
50               return new Prediction(0.0, 0.1);
51           }
52           // Dummy: gebe den gelernten Durchschnittswert zurück, Confidence moderat
53           return new Prediction(averageValue, 0.6);
54       }
55   }
