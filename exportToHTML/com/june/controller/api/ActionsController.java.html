1    package com.june.controller.api;
2    
3    import org.springframework.http.ResponseEntity;
4    import org.springframework.web.bind.annotation.*;
5    
6    import java.util.*;
7    
8    /**
9     * ActionsController – kleine Simulations-Endpoints für die UI-Buttons.
10    * Ziel: Keine 404 mehr, sichtbare Live-Aktualisierung via TelemetryController.
11    *
12    * Endpunkte (POST):
13    *  - /api/robots/deploy
14    *  - /api/energy/collect
15    *  - /api/threats/simulate
16    *  - /api/seed
17    */
18   @RestController
19   @RequestMapping("/api")
20   @CrossOrigin(origins = "*")
21   public class ActionsController {
22   
23       private final TelemetryController telemetry;
24   
25       public ActionsController(TelemetryController telemetry) {
26           this.telemetry = telemetry;
27       }
28   
29       @PostMapping("/robots/deploy")
30       public Map<String, Object> deployRobots() {
31           Map<String, Object> metrics = currentMetrics();
32           // kleiner positiver Effekt auf die Energie
33           double energy = asDouble(metrics.get("energy"));
34           metrics.put("energy", Math.max(0, energy + 50));
35           pushTelemetry("action:robots/deploy", metrics);
36           return Map.of("ok", true, "deployed", 100);
37       }
38   
39       @PostMapping("/energy/collect")
40       public Map<String, Object> collectEnergy() {
41           Map<String, Object> metrics = currentMetrics();
42           // deutlichere Energie-Erhöhung (simulativ)
43           double energy = asDouble(metrics.get("energy"));
44           double delta = 200 + Math.random() * 300; // 200..500
45           metrics.put("energy", Math.max(0, energy + Math.round(delta)));
46           pushTelemetry("action:energy/collect", metrics);
47           return Map.of("ok", true, "delta", Math.round(delta));
48       }
49   
50       @PostMapping("/threats/simulate")
51       public Map<String, Object> simulateThreat() {
52           Map<String, Object> metrics = currentMetrics();
53           long threats = asLong(metrics.get("threats"));
54           metrics.put("threats", threats + 1);
55           // leichte Temperaturänderung für Sichtbarkeit
56           double temp = asDouble(metrics.get("temperature"));
57           metrics.put("temperature", Math.max(0, Math.round((temp + (Math.random() > 0.5 ? 0.2 : -0.2)) * 10.0) / 10.0));
58           pushTelemetry("action:threats/simulate", metrics);
59           return Map.of("ok", true, "threats", threats + 1);
60       }
61   
62       @PostMapping("/seed")
63       public Map<String, Object> seed() {
64           // Nur Bestätigung + kleiner Impuls für Telemetrie
65           Map<String, Object> metrics = currentMetrics();
66           double energy = asDouble(metrics.get("energy"));
67           metrics.put("energy", Math.max(0, energy + 10));
68           pushTelemetry("action:seed", metrics);
69           return Map.of("ok", true);
70       }
71   
72       // === helpers ===
73       @SuppressWarnings("unchecked")
74       private Map<String, Object> currentMetrics() {
75           try {
76               ResponseEntity<Map<String, Object>> res = telemetry.status();
77               Map<String, Object> snap = res.getBody();
78               if (snap == null) snap = new LinkedHashMap<>();
79               Object m = snap.get("metrics");
80               if (m instanceof Map<?, ?>) {
81                   return new LinkedHashMap<>((Map<String, Object>) m);
82               }
83           } catch (Exception ignored) {}
84           // Fallback-Basiswerte
85           Map<String, Object> base = new LinkedHashMap<>();
86           base.put("energy", 3000);
87           base.put("threats", 0);
88           base.put("temperature", 23);
89           return base;
90       }
91   
92       private void pushTelemetry(String source, Map<String, Object> metrics) {
93           Map<String, Object> payload = new LinkedHashMap<>();
94           payload.put("source", source);
95           payload.put("metrics", metrics);
96           try {
97               telemetry.push(payload);
98           } catch (Exception ignored) { /* best-effort */ }
99       }
100  
101      private static double asDouble(Object o) {
102          if (o instanceof Number) return ((Number) o).doubleValue();
103          try { return Double.parseDouble(String.valueOf(o)); } catch (Exception e) { return 0.0; }
104      }
105      private static long asLong(Object o) {
106          if (o instanceof Number) return ((Number) o).longValue();
107          try { return Long.parseLong(String.valueOf(o)); } catch (Exception e) { return 0L; }
108      }
109  }
