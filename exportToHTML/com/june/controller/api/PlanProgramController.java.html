1    package com.june.controller.api;

3    import org.springframework.http.HttpHeaders;
4    import org.springframework.http.MediaType;
5    import org.springframework.http.ResponseEntity;
6    import org.springframework.web.bind.annotation.*;
7    
8    import java.nio.charset.StandardCharsets;
9    import java.nio.file.*;
10   import java.util.Map;
11   
12   @RestController
13   @RequestMapping("/api/ml/plan")
14   @CrossOrigin(origins = "*")
15   public class PlanProgramController {
16   
17       private static final Path PLAN_DIR = Paths.get("data", "ml");
18       private static final Path PLAN_FILE = PLAN_DIR.resolve("policy-plan.json");
19   
20       @GetMapping(produces = MediaType.APPLICATION_JSON_VALUE)
21       public ResponseEntity<String> get(){
22           try {
23               if (!Files.exists(PLAN_FILE)){
24                   String empty = "{\"planTitle\":\"\",\"objectives\":[],\"requiredSteps\":[]}";
25                   return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(empty);
26               }
27               String s = Files.readString(PLAN_FILE, StandardCharsets.UTF_8);
28               return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(s);
29           } catch (Exception e){
30               return ResponseEntity.status(500).contentType(MediaType.APPLICATION_JSON)
31                       .body("{\"error\":\"Plan lesen fehlgeschlagen\"}");
32           }
33       }
34   
35       @PostMapping(consumes = MediaType.APPLICATION_JSON_VALUE)
36       public ResponseEntity<Map<String,Object>> set(@RequestBody Map<String,Object> body){
37           try {
38               Files.createDirectories(PLAN_DIR);
39               String json = toJsonFiltered(body);
40               if (json.length() > 50_000){
41                   return ResponseEntity.badRequest().body(Map.of("error","Plan zu groß"));
42               }
43               Files.writeString(PLAN_FILE, json, StandardCharsets.UTF_8, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
44               return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(Map.of("ok",true));
45           } catch (Exception e){
46               return ResponseEntity.status(500).body(Map.of("error","Plan speichern fehlgeschlagen"));
47           }
48       }
49   
50       // keep only expected fields, simple JSON builder
51       private static String toJsonFiltered(Map<String,Object> in){
52           String title = String.valueOf(in.getOrDefault("planTitle", "")).trim();
53           if (title.length() > 200) title = title.substring(0,200);
54           Object objectives = in.get("objectives");
55           Object required = in.get("requiredSteps");
56           StringBuilder sb = new StringBuilder();
57           sb.append('{');
58           sb.append("\"planTitle\":\"").append(escape(title)).append("\"");
59           sb.append(",\"objectives\":").append(arrayOfStrings(objectives));
60           sb.append(",\"requiredSteps\":").append(arrayOfStrings(required));
61           sb.append('}');
62           return sb.toString();
63       }
64       private static String arrayOfStrings(Object o){
65           if (!(o instanceof Iterable<?>)) return "[]";
66           StringBuilder sb = new StringBuilder("["); boolean first=true;
67           for (Object v : (Iterable<?>)o){
68               String s = String.valueOf(v);
69               if (!first) sb.append(','); first=false;
70               sb.append('"').append(escape(s)).append('"');
71           }
72           sb.append(']'); return sb.toString();
73       }
74       private static String escape(String s){ return s.replace("\\","\\\\").replace("\"","\\\"").replace("\n","\\n"); }
75   }
