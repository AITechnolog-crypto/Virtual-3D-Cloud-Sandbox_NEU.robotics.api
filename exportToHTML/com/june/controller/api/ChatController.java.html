43       @GetMapping(value = "/history", produces = MediaType.APPLICATION_JSON_VALUE)
44       public ResponseEntity<List<Map<String,Object>>> history(@RequestParam(value = "channel", required = false) String channel,
45                                                               @RequestParam(value = "limit", required = false) Integer limit){
46           String ch = safeChannel(channel);
47           int lim = (limit == null || limit < 1 || limit > 500) ? 100 : limit;
48           List<ChatMessage> list = store.list(ch, lim);
49           List<Map<String,Object>> out = new ArrayList<>();
50           for (ChatMessage m : list){ out.add(toMap(m)); }
51           return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(out);
52       }

54       @PostMapping(value = "/send", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
55       public ResponseEntity<Map<String,Object>> send(@RequestBody SendReq body, @RequestHeader Map<String,String> headers){
56           String clientIp = headers.getOrDefault("x-forwarded-for", "direct");
57           if (!consumeRate(clientIp)) return err(429, "Rate Limit: max "+LIMIT_PER_MIN+"/Minute");
58           if (body == null) return err(400, "Body fehlt");
59           String ch = safeChannel(body.channel);
60           String role = safeRole(body.role);
61           String sender = safeSender(body.sender);
62           String text = safeText(body.text);
63           if (text.isBlank()) return err(400, "Text leer");
64           if (text.length() > MAX_TEXT) text = text.substring(0, MAX_TEXT);
65           String low = text.toLowerCase(Locale.ROOT);
66           for (String bad : BLOCKLIST){ if (low.contains(bad.toLowerCase(Locale.ROOT))) return err(400, "Blockiert durch Sicherheitsfilter"); }
67           ChatMessage m = store.append(ch, sender, role, text);
68           emit(ch, toMap(m));
69           return ok(toMap(m));
70       }

72       @GetMapping(path = "/stream", produces = MediaType.TEXT_EVENT_STREAM_VALUE)
73       public SseEmitter stream(@RequestParam(value = "channel", required = false) String channel){
74           String ch = safeChannel(channel);
75           SseEmitter em = new SseEmitter(0L);
76           channelListeners.computeIfAbsent(ch, k-> new CopyOnWriteArrayList<>()).add(em);
77           em.onCompletion(() -> channelListeners.getOrDefault(ch, List.of()).remove(em));
78           em.onTimeout(() -> channelListeners.getOrDefault(ch, List.of()).remove(em));
79           try { em.send(SseEmitter.event().name("ready").data(Map.of("ts", System.currentTimeMillis(), "channel", ch))); } catch (IOException ignored) {}
80           return em;
81       }

83       @PostMapping(value = "/assist", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
84       public ResponseEntity<Map<String,Object>> assist(@RequestBody AssistReq body, @RequestHeader Map<String,String> headers){
85           String clientIp = headers.getOrDefault("x-forwarded-for", "direct");
86           if (!consumeRate(clientIp)) return err(429, "Rate Limit: max "+LIMIT_PER_MIN+"/Minute");
87           if (body == null) return err(400, "Body fehlt");
88           String ch = safeChannel(body.channel);
89           String role = safeRole(body.role);
90           String sender = safeSender(body.sender);
91           String msg = safeText(body.message);
92           if (msg.isBlank()) return err(400, "message leer");
93           String sys = systemPromptFor(role);
94           String reply;
95           try {
96               DeepSeekClient client = new DeepSeekClient(env);
97               if (client.hasKey()){
98                   String model = (body.model == null || body.model.isBlank()) ? "deepseek-chat" : body.model;
99                   String prompt = buildPrompt(sys, ch, msg);
100                  String raw = client.chat(model, prompt, 600, 0.6);
101                  // sehr klein: versuche content zu extrahieren, sonst gib raw zurück
102                  String content = extractContent(raw);
103                  reply = (content == null || content.isBlank()) ? raw : content;
104              } else {
105                  reply = fallbackReply(role, msg);
106              }
107          } catch (Exception e){
108              reply = fallbackReply(role, msg) + "\n(Hinweis: Live-API nicht verfügbar: "+ e.getMessage() +")";
109          }
110          // schreibe Nutzernachricht+Antwort in Chain (ohne Sidechain)
111          ChatMessage userMsg = store.append(ch, sender, role == null?"user":role, msg);
112          emit(ch, toMap(userMsg));
113          ChatMessage botMsg = store.append(ch, "assistant", "assistant", reply);
114          Map<String,Object> out = toMap(botMsg);
115          emit(ch, out);
116          return ok(out);
117      }

127      private static boolean consumeRate(String ip){
128          long now = Instant.now().toEpochMilli();
129          Deque<Long> q = RATE.computeIfAbsent(ip, k-> new ArrayDeque<>());
130          synchronized (q){
131              long cutoff = now - 60_000L;
132              while (!q.isEmpty() && q.peekFirst() < cutoff) q.pollFirst();
133              if (q.size() >= LIMIT_PER_MIN) return false;
134              q.addLast(now); return true;
135          }
136      }

138      private void emit(String channel, Object payload){
139          List<SseEmitter> list = channelListeners.getOrDefault(channel, List.of());
140          for (SseEmitter e : list){
141              try { e.send(SseEmitter.event().name("msg").data(payload)); }
142              catch(Exception ex){ try{ e.complete(); }catch(Exception ignore){} }
143          }
144      }

146      private static String safeChannel(String ch){
147          if (ch == null || ch.isBlank()) return "customer";
148          String s = ch.toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9_-]", "_");
149          if (s.isBlank()) s = "customer";
150          return s;
151      }
152      private static String safeRole(String r){ return (r==null||r.isBlank())?"user":r.trim(); }
153      private static String safeSender(String s){ return (s==null||s.isBlank())?"user":s.trim(); }
154      private static String safeText(String t){ return t==null?"":t.trim(); }

156      private static Map<String,Object> toMap(ChatMessage m){
157          Map<String,Object> map = new LinkedHashMap<>();
158          map.put("id", m.id); map.put("ts", m.ts); map.put("channel", m.channel);
159          map.put("sender", m.sender); map.put("role", m.role); map.put("text", m.text);
160          map.put("prevHash", m.prevHash); map.put("hash", m.hash);
161          return map;
162      }

172          return base + " Antworte als Kunden-Chatbot (hilfsbereit, freundlich, einfache Sprache).";

175          return sys + "\nKanal: "+channel+"\nFrage: "+ msg + "\nAntwort:";

178      private static String extractContent(String raw){
179          if (raw == null) return null;
180          // sehr einfach: suche nach "message":{"content":"..."}
181          int i = raw.indexOf("\"message\""); if (i<0) return null;
182          int c = raw.indexOf("\"content\"", i); if (c<0) return null;
183          int q1 = raw.indexOf('"', c+9); if (q1<0) return null;
184          int q2 = raw.indexOf('"', q1+1); if (q2<0) return null;
185          String s = raw.substring(q1+1, q2);
186          return s.replace("\\n", "\n").replace("\\\"", "\"");
187      }
