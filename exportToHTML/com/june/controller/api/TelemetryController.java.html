1    package com.june.controller.api;

3    import com.june.service.history.HistoryStore;
4    import org.springframework.http.HttpHeaders;
5    import org.springframework.http.MediaType;
6    import org.springframework.http.ResponseEntity;
7    import org.springframework.web.bind.annotation.*;
8    import org.springframework.web.servlet.mvc.method.annotation.SseEmitter;
9    
10   import java.io.IOException;
11   import java.time.Instant;
12   import java.util.*;
13   import java.util.concurrent.CopyOnWriteArrayList;
14   
15   /**
16    * Lightweight telemetry endpoints to integrate future Java simulations in real time.
17    * - POST /api/telemetry/push  → accepts a small JSON payload and stores it in-memory
18    * - GET  /api/telemetry/status → returns latest snapshot
19    * - GET  /api/telemetry/stream → Server-Sent Events emitting on updates
20    */
21   @RestController
22   @RequestMapping("/api/telemetry")
23   @CrossOrigin(origins = "*")
24   public class TelemetryController {
25    
26       private static final int MAX_JSON_CHARS = 20_000;
27       private static final Set<String> BLOCKLIST = Set.of("<script>", "</script>", "\u0000");
28    
29       private volatile Map<String, Object> latest = baseSnapshot();
30       private final List<SseEmitter> listeners = new CopyOnWriteArrayList<>();
31       private final HistoryStore history;
32    
33       public TelemetryController(HistoryStore history){
34           this.history = history;
35       }
36   
37       @GetMapping("/status")
38       public ResponseEntity<Map<String, Object>> status(){
39           return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(latest);
40       }
41   
42       @PostMapping(value = "/push", consumes = MediaType.APPLICATION_JSON_VALUE)
43       public ResponseEntity<Map<String, Object>> push(@RequestBody Map<String, Object> payload){
44           String jsonLenCalc = String.valueOf(payload);
45           if (jsonLenCalc.length() > MAX_JSON_CHARS){
46               return ResponseEntity.badRequest().body(Map.of("error", "Payload zu groß"));
47           }
48           String asString = jsonLenCalc.toLowerCase(Locale.ROOT);
49           for (String bad : BLOCKLIST){
50               if (asString.contains(bad.toLowerCase(Locale.ROOT))){
51                   return ResponseEntity.badRequest().body(Map.of("error", "Payload durch Sicherheitsfilter blockiert"));
52               }
53           }
54           Map<String, Object> snap = new LinkedHashMap<>(baseSnapshot());
55           snap.put("ts", System.currentTimeMillis());
56           snap.put("source", payload.getOrDefault("source", "unknown"));
57           Object metrics = payload.get("metrics");
58           if (metrics instanceof Map){
59               // merge selected metrics into a flat view for convenience
60               @SuppressWarnings("unchecked") Map<String,Object> mm = (Map<String,Object>) metrics;
61               snap.put("metrics", new LinkedHashMap<>(mm));
62           } else {
63               snap.put("metrics", Map.of());
64           }
65           this.latest = snap;
66           // Persist to history (best-effort)
67           try { if (history != null) history.append(snap); } catch (Exception ignored) {}
68           emitSse(snap);
69           return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(snap);
70       }
71   
72       @GetMapping(path = "/stream", produces = MediaType.TEXT_EVENT_STREAM_VALUE)
73       public SseEmitter stream(){
74           SseEmitter emitter = new SseEmitter(0L); // no timeout
75           listeners.add(emitter);
76           emitter.onCompletion(() -> listeners.remove(emitter));
77           emitter.onTimeout(() -> listeners.remove(emitter));
78           // send initial snapshot
79           try {
80               emitter.send(SseEmitter.event().name("snapshot").data(latest));
81           } catch (IOException ignored) {}
82           return emitter;
83       }
84   
85       private void emitSse(Map<String, Object> data){
86           for (SseEmitter e : listeners){
87               try {
88                   e.send(SseEmitter.event().name("update").data(data));
89               } catch (Exception ex){
90                   try { e.complete(); } catch (Exception ignore) {}
91                   listeners.remove(e);
92               }
93           }
94       }
95   
96       private static Map<String, Object> baseSnapshot(){
97           Map<String,Object> m = new LinkedHashMap<>();
98           m.put("ts", Instant.now().toEpochMilli());
99           m.put("source", "init");
100          m.put("metrics", Map.of(
101                  "energy", 2847,
102                  "threats", 0,
103                  "temperature", 23
104          ));
105          return m;
106      }
107  }
