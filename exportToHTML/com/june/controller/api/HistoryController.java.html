1    package com.june.controller.api;
2    
3    import com.june.service.history.HistoryStore;
4    import org.springframework.http.HttpHeaders;
5    import org.springframework.http.MediaType;
6    import org.springframework.http.ResponseEntity;
7    import org.springframework.web.bind.annotation.*;
8    
9    import java.util.Map;
10   
11   @RestController
12   @RequestMapping("/api/history")
13   @CrossOrigin(origins = "*")
14   public class HistoryController {
15   
16       private final HistoryStore history;
17   
18       public HistoryController(HistoryStore history){
19           this.history = history;
20       }
21   
22       @GetMapping(value = "/summary", produces = MediaType.APPLICATION_JSON_VALUE)
23       public ResponseEntity<Map<String,Object>> summary(@RequestParam(value = "months", required = false) Integer months){
24           int maxFiles = (months == null || months <= 0 || months > 24) ? 12 : months;
25           HistoryStore.Summary s = history.summarize(maxFiles);
26           return ResponseEntity.ok()
27                   .header(HttpHeaders.CACHE_CONTROL, "no-store")
28                   .body(s.toMap());
29       }
30   }
