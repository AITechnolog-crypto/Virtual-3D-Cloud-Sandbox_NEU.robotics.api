8    
9    import java.nio.charset.StandardCharsets;
10   import java.nio.file.*;
11   import java.util.*;

13   @RestController
14   @RequestMapping("/api/federation")
15   @CrossOrigin(origins = "*")
16   public class FederationController {
17   
18       private final FederationStore store;
19       public FederationController(FederationStore store){ this.store = store; }
20   
21       @PostMapping(value = "/submitModel", consumes = MediaType.APPLICATION_JSON_VALUE)
22       public ResponseEntity<Map<String,Object>> submit(@RequestBody Map<String,Object> model){
23           if (model == null || model.isEmpty()) return ResponseEntity.badRequest().body(Map.of("error","leerer Body"));
24           // keep only known fields to reduce risk
25           Map<String,Object> m = new LinkedHashMap<>();
26           copyIfPresent(model, m, "slope", "intercept", "min", "max", "avg", "n", "source", "instanceId");
27           if (!m.containsKey("slope") && !m.containsKey("intercept")){
28               return ResponseEntity.badRequest().body(Map.of("error","Modellparameter fehlen"));
29           }
30           store.submit(m);
31           return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(Map.of("ok", true));
32       }
33   
34       @GetMapping(value = "/aggregate", produces = MediaType.APPLICATION_JSON_VALUE)
35       public ResponseEntity<Map<String,Object>> aggregate(@RequestParam(value = "limit", required = false) Integer limit){
36           int lim = (limit == null || limit < 1 || limit > 5000) ? 2000 : limit;
37           Map<String,Object> agg = store.aggregate(lim);
38           return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(agg);
39       }
40   
41       @GetMapping(value = "/models", produces = MediaType.APPLICATION_JSON_VALUE)
42       public ResponseEntity<List<Map<String,Object>>> models(@RequestParam(value = "limit", required = false) Integer limit){
43           int lim = (limit == null || limit < 1 || limit > 2000) ? 100 : limit;
44           List<Map<String,Object>> list = store.list(lim);
45           return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(list);
46       }
47   
48       @PostMapping(value = "/adoptAggregate")
49       public ResponseEntity<Map<String,Object>> adopt(){
50           try {
51               Path agg = FederationStore.aggregateFile();
52               if (!Files.exists(agg)) return ResponseEntity.status(409).body(Map.of("error","Kein Aggregat vorhanden"));
53               String json = Files.readString(agg, StandardCharsets.UTF_8);
54               // write into LearningController's model file path
55               Path learnDir = Paths.get("data", "learn");
56               Files.createDirectories(learnDir);
57               Path modelFile = learnDir.resolve("energy-model.json");
58               Files.writeString(modelFile, json, StandardCharsets.UTF_8, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
59               return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(Map.of("ok",true));
60           } catch (Exception e){
61               return ResponseEntity.status(500).body(Map.of("error","Adopt‑Fehler: "+e.getMessage()));
62           }
63       }
64   
65       private static void copyIfPresent(Map<String,Object> from, Map<String,Object> to, String... keys){
66           for (String k : keys){ if (from.containsKey(k)) to.put(k, from.get(k)); }
67       }
68   }
