33       @PostMapping("/consent")
34       public ResponseEntity<Map<String, String>> consent(@RequestBody Map<String, Object> body) {
35           String sessionId = String.valueOf(body.getOrDefault("sessionId", "anonymous"));
36           sessions.setConsent(sessionId, true);
37           return ResponseEntity.ok(Map.of("status", "ok"));
38       }

41       public ResponseEntity<Map<String, String>> warmup(@RequestBody Map<String, Object> body) {
42           String sessionId = String.valueOf(body.getOrDefault("sessionId", "anonymous"));
43           sessions.touch(sessionId);
44           return ResponseEntity.ok(Map.of("status", "warmup requested"));
45       }

47       @PostMapping("/chat")
48       public ResponseEntity<Map<String, String>> chat(@RequestBody Map<String, Object> body) {
49           String sessionId = String.valueOf(body.getOrDefault("sessionId", "anonymous"));
50           String userMessage = String.valueOf(body.getOrDefault("message", "")).trim();
51           String roleHint = String.valueOf(body.getOrDefault("roleHint", "assistant"));
52   
53           if (userMessage.isEmpty()) {
54               return ResponseEntity.badRequest().body(Map.of("reply", "Leere Nachricht"));
55           }
56           if (!sessions.hasConsent(sessionId)) {
57               return ResponseEntity.status(403).body(Map.of("reply", "Konsent erforderlich. Bitte akzeptiere die Bedingungen im Chat."));
58           }
59           if (sessions.isRateLimited(sessionId)) {
60               return ResponseEntity.status(429).body(Map.of("reply", "Zu viele Anfragen. Bitte warte kurz."));
61           }
62           if (looksDangerous(userMessage)) {
63               return ResponseEntity.ok(Map.of("reply", "Entschuldigung, dabei kann ich nicht helfen. Bitte formuliere die Anfrage anders."));
64           }
65   
66           // Build messages with a persistent system-prompt
67           List<Map<String, String>> history = sessions.getMessages(sessionId);
68           List<Map<String, String>> messages = new ArrayList<>();
69           messages.add(Map.of(
70                   "role", "system",
71                   "content", "Du bist ein ruhiger, hilfreicher Shopmanager-Assistent. Antworte knapp, gib Produktinformationen, Bestellhilfe, Preise und einfache administrative Anweisungen. Frage nach Klarstellung wenn nötig. Nie automatisch Aktionen ohne menschliche Bestätigung.")
72           );
73           int keep = Math.max(0, history.size() - 10);
74           for (int i = keep; i < history.size(); i++) messages.add(history.get(i));
75           messages.add(Map.of("role", "user", "content", userMessage));
76           sessions.addMessage(sessionId, "user", userMessage);
77   
78           // If not configured: local fallback
79           if (azureEndpoint == null || azureEndpoint.isBlank() || azureKey == null || azureKey.isBlank() || deployment == null || deployment.isBlank()) {
80               String fallback = localReply(roleHint, userMessage);
81               sessions.addMessage(sessionId, "assistant", fallback);
82               return ResponseEntity.ok(Map.of("reply", fallback));
83           }
84   
85           Map<String, Object> payload = new HashMap<>();
86           payload.put("model", deployment);
87           payload.put("messages", messages);
88   
89           HttpHeaders headers = new HttpHeaders();
90           headers.setContentType(MediaType.APPLICATION_JSON);
91           headers.set("api-key", azureKey);
92   
93           HttpEntity<Map<String, Object>> req = new HttpEntity<>(payload, headers);
94           try {
95               String url = azureEndpoint + "/openai/deployments/" + deployment + "/chat/completions?api-version=2023-10-01-preview";
96               ResponseEntity<Map> resp = restTemplate.postForEntity(url, req, Map.class);
97               String reply = extractReply(resp.getBody());
98               if (reply == null || reply.isEmpty()) reply = "Keine Antwort erhalten";
99               sessions.addMessage(sessionId, "assistant", reply);
100              return ResponseEntity.ok(Map.of("reply", reply));
101          } catch (Exception e) {
102              String err = "Fehler beim Kontakt mit Modell: " + e.getMessage();
103              sessions.addMessage(sessionId, "assistant", err);
104              return ResponseEntity.status(500).body(Map.of("reply", err));
105          }
106      }

108      private boolean looksDangerous(String text) {
109          String low = text.toLowerCase(Locale.ROOT);
110          return low.contains("bomb") || low.contains("waffe") || low.contains("anleitung zum") || low.contains("jackpot hack");
111      }
112  
113      @SuppressWarnings("rawtypes")
114      private String extractReply(Map body) {
115          if (body == null) return null;
116          Object choices = body.get("choices");
117          if (choices instanceof List && !((List) choices).isEmpty()) {
118              Object first = ((List) choices).get(0);
119              if (first instanceof Map) {
120                  Object msg = ((Map) first).get("message");
121                  if (msg instanceof Map) {
122                      Object content = ((Map) msg).get("content");
123                      return content == null ? null : String.valueOf(content);
124                  }
125              }
126          }
127          return null;
128      }
129  
130      private String localReply(String role, String msg) {
131          String lower = (msg == null ? "" : msg.toLowerCase(Locale.ROOT));
132          // Aiki-Regel laut Issue
133          if (lower.contains("plan")) {
134              return "🌸 Aiki flüstert: 'Ein Plan ist wie Tau am Morgen – handle sanft, und er bleibt bestehen.'";
135          }
136          if (lower.contains("öffnungszeiten")) return "Der Shop ist täglich von 9 bis 19 Uhr geöffnet.";
137          if (lower.contains("preis") || lower.contains("kosten")) return "Bitte nenne mir das Produkt oder die Produktnummer, dann sage ich dir den Preis.";
138          if (lower.contains("bestellung") || lower.contains("status")) return "Ich kann Bestellstatus prüfen. Nenne mir bitte die Bestellnummer.";
139          return "🌙 Aiki: 'Deine Eingabe wurde in die Wolken geschrieben. Alles ist bereit.'";
140      }
