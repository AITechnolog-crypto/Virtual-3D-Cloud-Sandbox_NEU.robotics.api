49       @GetMapping(value = "/predict", produces = MediaType.APPLICATION_JSON_VALUE)
50       public ResponseEntity<Map<String,Object>> predict(@RequestParam(value = "steps", required = false) Integer steps){
51           int n = (steps == null || steps < 1 || steps > 200) ? 10 : steps;
52           try {
53               Model m = readModel();
54               if (m == null) return err(409, "Kein Modell trainiert");
55               List<Double> preds = new ArrayList<>();
56               for (int i=1;i<=n;i++) preds.add(m.predictNext(i));
57               return okJson(Map.of(
58                       "trainedTs", m.trainedTs,
59                       "steps", n,
60                       "predictions", preds
61               ));
62           } catch (Exception e){
63               return err(500, "Vorhersage-Fehler: "+e.getMessage());
64           }
65       }

67       // --- minimal linear model over energy values ---
68       private Model fitModelFromHistory() throws Exception {
69           List<Path> files = listHistoryFiles();
70           long total = 0L;
71           List<Double> y = new ArrayList<>();
72           for (Path f : files){
73               List<String> lines = Files.readAllLines(f, StandardCharsets.UTF_8);
74               for (String line : lines){
75                   if (y.size() >= MAX_LINES) break;
76                   double v = extractEnergy(line);
77                   if (!Double.isNaN(v)) {
78                       y.add(v);
79                   }
80               }
81               if (y.size() >= MAX_LINES) break;
82           }
83           total = y.size();
84           if (total < 4) throw new IllegalStateException("Zu wenige Daten");
85           // simple linear regression y ~ a + b*t
86           double sumX=0, sumY=0, sumXX=0, sumXY=0;
87           for (int i=0;i<y.size();i++){
88               double xi = i+1; double yi = y.get(i);
89               sumX += xi; sumY += yi; sumXX += xi*xi; sumXY += xi*yi;
90           }
91           double n = y.size();
92           double denom = (n*sumXX - sumX*sumX);
93           double b = denom != 0 ? (n*sumXY - sumX*sumY)/denom : 0.0; // slope
94           double a = (sumY - b*sumX)/n; // intercept
95           double min = y.stream().mapToDouble(d->d).min().orElse(0);
96           double max = y.stream().mapToDouble(d->d).max().orElse(0);
97           double avg = y.stream().mapToDouble(d->d).average().orElse(0);
98           Model m = new Model();
99           m.trainedTs = System.currentTimeMillis();
100          m.n = (long) n;
101          m.slope = b;
102          m.intercept = a;
103          m.min = min; m.max = max; m.avg = avg;
104          return m;
105      }

107      private static List<Path> listHistoryFiles() throws Exception {
108          List<Path> list = new ArrayList<>();
109          if (Files.exists(HIST_DIR)){
110              try (DirectoryStream<Path> ds = Files.newDirectoryStream(HIST_DIR, "telemetry-*.jsonl")){
111                  for (Path p : ds) list.add(p);
112              }
113          }
114          // newest first by lastModified
115          list.sort((a,b)->{
116              try{
117                  long ta = Files.getLastModifiedTime(a).toMillis();
118                  long tb = Files.getLastModifiedTime(b).toMillis();
119                  return Long.compare(tb, ta);
120              }catch(Exception e){return 0;}
121          });
122          return list;
123      }

125      private static double extractEnergy(String line){
126          if (line == null) return Double.NaN;
127          int mIdx = line.indexOf("\"metrics\":");
128          if (mIdx < 0) return Double.NaN;
129          int eIdx = line.indexOf("\"energy\":", mIdx);
130          if (eIdx < 0) return Double.NaN;
131          int eStart = eIdx + 9;
132          int i = eStart;
133          StringBuilder num = new StringBuilder();
134          while (i < line.length()){
135              char c = line.charAt(i);
136              if ((c>='0'&&c<='9') || c=='.' || c=='-') num.append(c); else break;
137              i++;
138          }
139          try { return Double.parseDouble(num.toString()); } catch(Exception e){ return Double.NaN; }
140      }

142      private void writeModel(Model m) throws Exception {
143          String json = toJson(m.toMap());
144          Files.writeString(MODEL_FILE, json, StandardCharsets.UTF_8, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
145      }

147      private Model readModel() throws Exception {
148          if (!Files.exists(MODEL_FILE)) return null;
149          // very small parser: we only need fields back; use Properties-like parse
150          String s = Files.readString(MODEL_FILE, StandardCharsets.UTF_8);
151          Model m = new Model();
152          // naive extractions
153          m.trainedTs = extractLong(s, "trainedTs");
154          m.n = extractLong(s, "n");
155          m.slope = extractDouble(s, "slope");
156          m.intercept = extractDouble(s, "intercept");
157          m.min = extractDouble(s, "min");
158          m.max = extractDouble(s, "max");
159          m.avg = extractDouble(s, "avg");
160          return m;
161      }

163      private static long extractLong(String s, String key){
164          int i = s.indexOf('"'+key+'"'); if (i<0) return 0;
165          int c = s.indexOf(':', i); if (c<0) return 0;
166          int j = c+1; StringBuilder b = new StringBuilder();
167          while (j<s.length()){
168              char ch = s.charAt(j++);
169              if ((ch>='0'&&ch<='9')) b.append(ch); else if (b.length()>0) break;
170          }
171          try { return Long.parseLong(b.toString()); } catch(Exception e){ return 0; }
172      }
173      private static double extractDouble(String s, String key){
174          int i = s.indexOf('"'+key+'"'); if (i<0) return 0;
175          int c = s.indexOf(':', i); if (c<0) return 0;
176          int j = c+1; StringBuilder b = new StringBuilder();
177          while (j<s.length()){
178              char ch = s.charAt(j++);
179              if ((ch>='0'&&ch<='9')||ch=='.'||ch=='-') b.append(ch); else if (b.length()>0) break;
180          }
181          try { return Double.parseDouble(b.toString()); } catch(Exception e){ return 0; }
182      }

185          StringBuilder sb = new StringBuilder();
186          appendJson(sb, m);
187          return sb.toString();
188      }
189      private static void appendJson(StringBuilder sb, Object obj){
190          if (obj == null){ sb.append("null"); return; }
191          if (obj instanceof Number || obj instanceof Boolean){ sb.append(String.valueOf(obj)); return; }
192          if (obj instanceof String){ sb.append('"').append(escape((String)obj)).append('"'); return; }
193          if (obj instanceof Map){ sb.append('{'); boolean first=true; for (Map.Entry<?,?> e: ((Map<?,?>)obj).entrySet()){ if(!first) sb.append(','); first=false; sb.append('"').append(escape(String.valueOf(e.getKey()))).append('"').append(':'); appendJson(sb,e.getValue()); } sb.append('}'); return; }
194          if (obj instanceof Iterable){ sb.append('['); boolean first=true; for (Object v: (Iterable<?>)obj){ if(!first) sb.append(','); first=false; appendJson(sb,v);} sb.append(']'); return; }
195          sb.append('"').append(escape(String.valueOf(obj))).append('"');
196      }
197      private static String escape(String s){ return s.replace("\\","\\\\").replace("\"","\\\"").replace("\n","\\n"); }

206      private static final class Model{
207          long trainedTs;
208          long n;
209          double slope;
210          double intercept;
211          double min, max, avg;
212          Map<String,Object> toMap(){
213              return new LinkedHashMap<>(Map.of(
214                      "trained", true,
215                      "trainedTs", trainedTs,
216                      "n", n,
217                      "slope", slope,
218                      "intercept", intercept,
219                      "min", min,
220                      "max", max,
221                      "avg", avg
222              ));
223          }
224          double predictNext(int step){
225              double x = (n + step);
226              return intercept + slope * x;
227          }
228      }
