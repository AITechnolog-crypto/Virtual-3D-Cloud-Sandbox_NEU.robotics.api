3    import com.june.service.external.AzurePredictionClient;
4    import org.springframework.core.env.Environment;
5    import org.springframework.http.HttpHeaders;
6    import org.springframework.http.MediaType;
7    import org.springframework.http.ResponseEntity;
8    import org.springframework.web.bind.annotation.*;
9    
10   import java.time.Instant;
11   import java.util.*;
12   import java.util.concurrent.ConcurrentHashMap;
13   
14   @RestController
15   @RequestMapping("/api/azure")
16   @CrossOrigin(origins = "*")
17   public class AzurePredictionController {
18   
19       private final Environment env;
20       private static final int MAX_JSON_CHARS = 30_000;
21       private static final Set<String> BLOCKLIST = Set.of("<script>", "</script>", "..\\", "../", "\u0000");
22   
23       // naive per-IP rate limit
24       private static final Map<String, Deque<Long>> RATE = new ConcurrentHashMap<>();
25       private static final int LIMIT_PER_MIN = 30;
26   
27       public AzurePredictionController(Environment env){
28           this.env = env;
29       }
30   
31       @GetMapping(value = "/status", produces = MediaType.APPLICATION_JSON_VALUE)
32       public ResponseEntity<Map<String,Object>> status(){
33           AzurePredictionClient client = new AzurePredictionClient(env);
34           Map<String,Object> out = new LinkedHashMap<>();
35           out.put("configured", client.configured());
36           try {
37               java.net.URI u = java.net.URI.create(getEndpointSafe());
38               String host = Optional.ofNullable(u.getHost()).orElse("");
39               out.put("endpointHost", host);
40           } catch (Exception e){
41               out.put("endpointHost", "");
42           }
43           out.put("region", getRegionSafe());
44           out.put("time", Instant.now().toString());
45           return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(out);
46       }
47   
48       @PostMapping(value = "/predict", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
49       public ResponseEntity<String> predict(@RequestBody(required = false) String json,
50                                             @RequestHeader Map<String,String> headers,
51                                             @RequestParam(value = "path", required = false) String relativePath){
52           String clientIp = Optional.ofNullable(headers.get("x-forwarded-for")).orElse("direct");
53           if (!consumeRate(clientIp)){
54               return ResponseEntity.status(429).contentType(MediaType.APPLICATION_JSON)
55                       .body("{\"error\":\"Rate Limit: max "+LIMIT_PER_MIN+"/Minute\"}");
56           }
57           if (json == null) json = "{}";
58           if (json.length() > MAX_JSON_CHARS){
59               return ResponseEntity.badRequest().contentType(MediaType.APPLICATION_JSON)
60                       .body("{\"error\":\"Payload zu groß\"}");
61           }
62           String lower = json.toLowerCase(Locale.ROOT);
63           for (String bad : BLOCKLIST){
64               if (lower.contains(bad.toLowerCase(Locale.ROOT))){
65                   return ResponseEntity.badRequest().contentType(MediaType.APPLICATION_JSON)
66                           .body("{\"error\":\"Payload durch Sicherheitsfilter blockiert\"}");
67               }
68           }
69           AzurePredictionClient client = new AzurePredictionClient(env);
70           if (!client.configured()){
71               return ResponseEntity.status(503)
72                       .header(HttpHeaders.CACHE_CONTROL, "no-store")
73                       .contentType(MediaType.APPLICATION_JSON)
74                       .body("{\"error\":\"Azure Prediction nicht konfiguriert oder unsicherer Endpoint. Bitte AZURE_PREDICTION_ENDPOINT (*.azure.com/.net), AZURE_PREDICTION_KEY setzen.\"}");
75           }
76           try{
77               String body = client.postJson(relativePath, json);
78               String wrapped = "{"+
79                       "\"ts\":"+ System.currentTimeMillis() +","+
80                       "\"ok\":true,"+
81                       "\"response\":" + safeEmbed(body) +
82                       "}";
83               return ResponseEntity.ok()
84                       .header(HttpHeaders.CACHE_CONTROL, "no-store")
85                       .contentType(MediaType.APPLICATION_JSON)
86                       .body(wrapped);
87           }catch(Exception e){
88               String msg = e.getMessage();
89               if (msg == null) msg = "Fehler";
90               return ResponseEntity.status(502).contentType(MediaType.APPLICATION_JSON)
91                       .body("{\"error\":\"Azure Anfrage fehlgeschlagen: "+ escape(msg) +"\"}");
92           }
93       }
94   
95       private static boolean consumeRate(String ip){
96           long now = Instant.now().toEpochMilli();
97           Deque<Long> q = RATE.computeIfAbsent(ip, k -> new ArrayDeque<>());
98           synchronized (q){
99               long cutoff = now - 60_000L;
100              while (!q.isEmpty() && q.peekFirst() < cutoff) q.pollFirst();
101              if (q.size() >= LIMIT_PER_MIN) return false;
102              q.addLast(now);
103              return true;
104          }
105      }
106  
107      private static String escape(String s){
108          return s.replace("\\", "\\\\").replace("\"","\\\"").replace("\n","\\n");
109      }
110      // embed external JSON as string value safely (best-effort)
111      private static String safeEmbed(String raw){
112          if (raw == null) return "{}";
113          return '"' + escape(raw) + '"';
114      }
115  
116      private String getEndpointSafe(){
117          try{ String val = Optional.ofNullable(System.getenv("AZURE_PREDICTION_ENDPOINT")).orElse(""); if (val==null||val.isBlank()) val = env!=null? env.getProperty("azure.prediction.endpoint", ""):""; return val; }catch(Exception e){ return ""; }
118      }
119      private String getRegionSafe(){
120          try{ String val = Optional.ofNullable(System.getenv("AZURE_REGION")).orElse(""); if (val==null||val.isBlank()) val = env!=null? env.getProperty("azure.region", "eastus"):"eastus"; return val; }catch(Exception e){ return "eastus"; }
121      }
