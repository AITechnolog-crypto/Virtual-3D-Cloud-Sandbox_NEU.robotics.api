19    * SERP Lexikon Controller – blockweises Paging, Schnittstellen‑Sicherung & Prüfung.
20    *
21    * Endpunkte:
22    *  - GET /api/serp/ping      → { ok, hasKey }
23    *  - GET /api/serp/lexicon   → gruppierte Ergebnisse (A–Z), blockweise
24    */
25   @RestController
26   @RequestMapping("/api/serp")
27   @CrossOrigin(origins = "*")
28   public class SerpLexiconController {

30       private final Environment env;
31       private final ObjectMapper om = new ObjectMapper();
32       private final RateLimiter rl = new RateLimiter(90); // 90 req/min pro IP
33   
34       public SerpLexiconController(Environment env) {
35           this.env = env;
36       }

38       @GetMapping(value = "/ping", produces = MediaType.APPLICATION_JSON_VALUE)
39       public Map<String, Object> ping() {
40           SerpApiClient client = new SerpApiClient(env);
41           return Map.of("ok", true, "hasKey", client.hasKey(), "time", Instant.now().toString());
42       }

44       @GetMapping(value = "/lexicon", produces = MediaType.APPLICATION_JSON_VALUE)
45       public ResponseEntity<Map<String, Object>> lexicon(
46               @RequestParam("q") String q,
47               @RequestParam(value = "block", required = false, defaultValue = "0") int block,
48               @RequestParam(value = "block_size", required = false, defaultValue = "10") int blockSize,
49               @RequestParam Map<String, String> allParams,
50               HttpServletRequest req
51       ) {
52           String ip = clientIp(req);
53           if (!rl.allow("lex:" + ip)) {
54               return ResponseEntity.status(429).body(Map.of("error", "Zu viele Anfragen – bitte warten"));
55           }
56           // Validation
57           q = q == null ? "" : q.trim();
58           if (q.isEmpty() || q.length() > 200) {
59               return ResponseEntity.badRequest().body(Map.of("error", "Ungültige Anfrage: q leer/zu lang"));
60           }
61           if (!isPrintable(q)) {
62               return ResponseEntity.badRequest().body(Map.of("error", "Ungültige Zeichen in q"));
63           }
64           block = Math.max(0, block);
65           blockSize = Math.max(1, Math.min(50, blockSize));
66   
67           SerpApiClient client = new SerpApiClient(env);
68           if (!client.hasKey()) {
69               // Graceful fallback: provide mock, grouped results so UI remains functional without a key
70               String term = q;
71               String letter = term.isEmpty() ? "#" : term.substring(0,1).toUpperCase(Locale.ROOT);
72               if (!letter.matches("[A-ZÄÖÜ#]")) letter = "#";
73               List<Map<String, Object>> list = new ArrayList<>();
74               list.add(Map.of(
75                       "term", term + " – Überblick",
76                       "title", term + " – Überblick",
77                       "url", "https://example.com/lex/" + term + "/overview",
78                       "domain", "example.com",
79                       "snippet", "Mock (Demo) – keine Live‑Daten."
80               ));
81               list.add(Map.of(
82                       "term", term + " – Grundlagen",
83                       "title", term + " – Grundlagen",
84                       "url", "https://example.org/lex/" + term + "/basics",
85                       "domain", "example.org",
86                       "snippet", "Mock (Demo) – keine Live‑Daten."
87               ));
88               Map<String, List<Map<String, Object>>> groups = new LinkedHashMap<>();
89               groups.put(letter, list);
90               Map<String, Object> out = new LinkedHashMap<>();
91               out.put("q", q);
92               out.put("engine", Optional.ofNullable(allParams.get("engine")).orElse("google"));
93               out.put("block", block);
94               out.put("blockSize", blockSize);
95               out.put("estimatedTotal", 2);
96               out.put("groups", groups);
97               out.put("count", list.size());
98               out.put("mock", true);
99               out.put("note", "SERPAPI_KEY fehlt – Demo-Modus aktiv");
100              return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(out);
101          }
102  
103          try {
104              // Build options: pass-through a safe subset, map block→start/num
105              Map<String, String> opts = new HashMap<>();
106              String[] pass = new String[]{
107                      "engine","location","hl","gl","tbm","google_domain","safe","tbs","time_period",
108                      "device","as_sitesearch","include_domains","exclude_domains","filter","lr","cr","uule"
109              };
110              for (String k : pass) {
111                  String v = allParams.get(k);
112                  if (v != null && !v.isBlank()) opts.put(k, v);
113              }
114              opts.put("num", String.valueOf(blockSize));
115              opts.put("start", String.valueOf(block * blockSize));
116  
117              String raw = client.search(q, opts);
118              Map<String, Object> json = om.readValue(raw, new TypeReference<>() {});
119  
120              // Extract items (organic_results preferred)
121              List<Map<String, Object>> items = new ArrayList<>();
122              Object org = json.get("organic_results");
123              if (org instanceof List<?>) {
124                  for (Object o : (List<?>) org) if (o instanceof Map<?, ?> m) items.add(cast(m));
125              } else if (json.get("results") instanceof List<?>) {
126                  for (Object o : (List<?>) json.get("results")) if (o instanceof Map<?, ?> m) items.add(cast(m));
127              }
128  
129              // Normalize entries
130              List<Map<String, Object>> entries = new ArrayList<>();
131              for (Map<String, Object> it : items) {
132                  String title = str(it.getOrDefault("title", it.get("name")));
133                  String link  = str(it.getOrDefault("link", it.get("url")));
134                  String snippet = str(it.getOrDefault("snippet", it.get("description")));
135                  if (title.isBlank() && snippet.isBlank()) continue;
136                  String domain = link.replaceFirst("(?i)^https?://", "");
137                  int slash = domain.indexOf('/'); if (slash>0) domain = domain.substring(0, slash);
138                  Map<String, Object> e = new LinkedHashMap<>();
139                  e.put("term", title.isBlank() ? snippet : title);
140                  e.put("title", title);
141                  e.put("url", link);
142                  e.put("domain", domain);
143                  e.put("snippet", snippet);
144                  entries.add(e);
145              }
146  
147              // Sort by title and group A–Z
148              entries.sort(Comparator.comparing(a -> str(a.get("term")).toLowerCase(Locale.ROOT)));
149              Map<String, List<Map<String, Object>>> groups = new LinkedHashMap<>();
150              for (Map<String, Object> e : entries) {
151                  String term = str(e.get("term")).trim();
152                  String letter = term.isEmpty() ? "#" : term.substring(0,1).toUpperCase(Locale.ROOT);
153                  if (!letter.matches("[A-ZÄÖÜ#]")) letter = "#";
154                  groups.computeIfAbsent(letter, k -> new ArrayList<>()).add(e);
155              }
156  
157              // Pager info: echo block & size; try to forward total results if present
158              long total = 0;
159              try {
160                  Object si = json.get("search_information");
161                  if (si instanceof Map<?,?> m) {
162                      Object tr = m.get("total_results");
163                      if (tr instanceof Number n) total = n.longValue();
164                      else if (tr instanceof String s) total = Long.parseLong(s.replaceAll("[^0-9]", ""));
165                  }
166              } catch (Exception ignore) {}
167  
168              Map<String, Object> out = new LinkedHashMap<>();
169              out.put("q", q);
170              out.put("engine", opts.getOrDefault("engine", "google"));
171              out.put("block", block);
172              out.put("blockSize", blockSize);
173              out.put("estimatedTotal", total);
174              out.put("groups", groups);
175              out.put("count", entries.size());
176              return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(out);
177          } catch (Exception e) {
178              return ResponseEntity.status(502).body(Map.of("error", "Lexikon-Query fehlgeschlagen: " + e.getMessage()));
179          }
180      }

183      private static boolean isPrintable(String s) {
184          for (int i=0;i<s.length();i++) {
185              char c = s.charAt(i);
186              if (Character.isISOControl(c)) return false;
187          }
188          return true;
189      }
190      @SuppressWarnings("unchecked")
191      private static Map<String, Object> cast(Map<?,?> m){ return (Map<String, Object>) m; }
192      private static String str(Object o){ return o==null?"":String.valueOf(o); }
193      private static String clientIp(HttpServletRequest req){
194          try {
195              String xf = req.getHeader("X-Forwarded-For");
196              if (xf != null && !xf.isBlank()) return xf.split(",")[0].trim();
197              return Optional.ofNullable(req.getRemoteAddr()).orElse("public");
198          } catch (Exception e) { return "public"; }
199      }

201      static class RateLimiter {
202          private final int maxPerMin;
203          private final Map<String, Deque<Long>> map = new ConcurrentHashMap<>();
204          RateLimiter(int maxPerMin){ this.maxPerMin = maxPerMin; }
205          synchronized boolean allow(String key){
206              long now = System.currentTimeMillis(), win = now - 60_000L;
207              Deque<Long> q = map.computeIfAbsent(key, k -> new ArrayDeque<>());
208              while (!q.isEmpty() && q.peekFirst() < win) q.pollFirst();
209              if (q.size() >= maxPerMin) return false;
210              q.addLast(now);
211              return true;
212          }
213      }
