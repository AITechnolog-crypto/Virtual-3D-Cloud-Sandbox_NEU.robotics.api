39       @PostMapping(value = "/policy", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
40       public ResponseEntity<Map<String,Object>> updatePolicy(@RequestBody Map<String,Object> body){
41           Policy p = policySvc.load();
42           try{
43               if (body.containsKey("trainingEnabled")) p.trainingEnabled = asBool(body.get("trainingEnabled"), p.trainingEnabled);
44               if (body.containsKey("ingestEnabled")) p.ingestEnabled = asBool(body.get("ingestEnabled"), p.ingestEnabled);
45               if (body.containsKey("mirrorTelemetryToDataset")) p.mirrorTelemetryToDataset = asBool(body.get("mirrorTelemetryToDataset"), p.mirrorTelemetryToDataset);
46               if (body.containsKey("retentionDays")) p.retentionDays = clampInt(body.get("retentionDays"), 1, 365, p.retentionDays);
47               if (body.containsKey("maxTrainPerHour")) p.maxTrainPerHour = clampInt(body.get("maxTrainPerHour"), 1, 60, p.maxTrainPerHour);
48               if (body.containsKey("allowExternalModels")) p.allowExternalModels = asBool(body.get("allowExternalModels"), p.allowExternalModels);
49               policySvc.save(p);
50               return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(toMap(p));
51           }catch(Exception e){
52               return ResponseEntity.badRequest().contentType(MediaType.APPLICATION_JSON).body(Map.of("error", "Ungültige Policy: "+e.getMessage()));
53           }
54       }

56       @GetMapping(value = "/status", produces = MediaType.APPLICATION_JSON_VALUE)
57       public ResponseEntity<Map<String,Object>> getStatus(){
58           Map<String,Object> m = new LinkedHashMap<>();
59           Policy p = policySvc.load();
60           m.put("policy", toMap(p));
61           // Include model status if exists
62           try{
63               Path model = Paths.get("data", "learn", "energy-model.json");
64               if (Files.exists(model)){
65                   String s = Files.readString(model, StandardCharsets.UTF_8);
66                   long ts = extractLong(s, "trainedTs");
67                   long n = extractLong(s, "n");
68                   m.put("model", Map.of("trained", true, "trainedTs", ts, "n", n));
69               } else {
70                   m.put("model", Map.of("trained", false));
71               }
72           }catch(Exception ignored){}
73           return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(m);
74       }

76       public static class IngestRow {
77           public Map<String,Object> features;
78           public String label;
79           public String source;
80       }
81   
82       @PostMapping(value = "/ingest", consumes = MediaType.APPLICATION_JSON_VALUE, produces = MediaType.APPLICATION_JSON_VALUE)
83       public ResponseEntity<Map<String,Object>> ingest(@RequestBody IngestRow row, @RequestHeader Map<String,String> headers){
84           Policy p = policySvc.load();
85           if (!p.ingestEnabled){
86               return ResponseEntity.status(409).contentType(MediaType.APPLICATION_JSON).body(Map.of("error", "Ingest deaktiviert durch Policy"));
87           }
88           String raw = String.valueOf(row);
89           if (raw.length() > MAX_PAYLOAD_CHARS){
90               return ResponseEntity.badRequest().body(Map.of("error", "Payload zu groß"));
91           }
92           String s = raw.toLowerCase(Locale.ROOT);
93           for (String bad : BLOCKLIST){
94               if (s.contains(bad.toLowerCase(Locale.ROOT))){
95                   return ResponseEntity.badRequest().body(Map.of("error", "Payload durch Sicherheitsfilter blockiert"));
96               }
97           }
98           try{
99               Map<String,Object> payload = new LinkedHashMap<>();
100              payload.put("features", row.features == null ? Map.of() : row.features);
101              if (row.label != null) payload.put("label", row.label);
102              if (row.source != null) payload.put("source", row.source);
103              payload.put("ts", System.currentTimeMillis());
104              policySvc.appendDatasetRow(payload);
105              return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(Map.of("ok", true, "rows", 1));
106          }catch(Exception e){
107              return ResponseEntity.status(500).contentType(MediaType.APPLICATION_JSON).body(Map.of("error", "Ingest-Fehler: "+e.getMessage()));
108          }
109      }

111      // Delegation with policy checks
112      @PostMapping(value = "/train", produces = MediaType.APPLICATION_JSON_VALUE)
113      public ResponseEntity<Map<String,Object>> train(@RequestHeader(value = "X-ORG-ID", required = false) String org){
114          Policy p = policySvc.load();
115          if (!p.trainingEnabled){
116              return ResponseEntity.status(409).contentType(MediaType.APPLICATION_JSON).body(Map.of("error", "Training deaktiviert durch Policy"));
117          }
118          if (!policySvc.canTrain(org, p.maxTrainPerHour)){
119              return ResponseEntity.status(429).contentType(MediaType.APPLICATION_JSON).body(Map.of("error", "Rate Limit: max "+p.maxTrainPerHour+"/h"));
120          }
121          // delegate to LearningController
122          ResponseEntity<Map<String,Object>> res = learning.train();
123          if (res.getStatusCode().is2xxSuccessful()) policySvc.recordTrain(org);
124          return res;
125      }

133      private static Map<String,Object> toMap(Policy p){
134          Map<String,Object> m = new LinkedHashMap<>();
135          m.put("trainingEnabled", p.trainingEnabled);
136          m.put("ingestEnabled", p.ingestEnabled);
137          m.put("mirrorTelemetryToDataset", p.mirrorTelemetryToDataset);
138          m.put("retentionDays", p.retentionDays);
139          m.put("maxTrainPerHour", p.maxTrainPerHour);
140          m.put("allowExternalModels", p.allowExternalModels);
141          return m;
142      }
143      private static boolean asBool(Object o, boolean def){ if (o==null) return def; if (o instanceof Boolean) return (Boolean)o; return "true".equalsIgnoreCase(String.valueOf(o)); }
144      private static int clampInt(Object o, int min, int max, int def){
145          try{ int v = Integer.parseInt(String.valueOf(o)); return Math.max(min, Math.min(max, v)); }catch(Exception e){ return def; }
146      }
147      private static long extractLong(String s, String key){
148          int i = s.indexOf('"'+key+'"'); if (i<0) return 0; int c = s.indexOf(':', i); if (c<0) return 0; int j=c+1; StringBuilder b=new StringBuilder();
149          while (j<s.length()){ char ch = s.charAt(j++); if ((ch>='0'&&ch<='9')) b.append(ch); else if (b.length()>0) break; }
150          try { return Long.parseLong(b.toString()); } catch(Exception e){ return 0; }
151      }
