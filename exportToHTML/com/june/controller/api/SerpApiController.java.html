30       @GetMapping(value = "/search", produces = MediaType.APPLICATION_JSON_VALUE)
31       public ResponseEntity<String> search(
32               @RequestParam("q") String q,
33               @RequestParam Map<String, String> allParams
34       ) {
35           SerpApiClient client = new SerpApiClient(env);
36           if (!client.hasKey()) {
37               // Graceful fallback: return deterministic mock payload so the UI remains usable without a key
38               String engine = Optional.ofNullable(allParams.get("engine")).filter(s->!s.isBlank()).orElse("google");
39               String escQ = q.replace("\\", "\\\\").replace("\"", "\\\"");
40               String mock = "{"+
41                       "\"mock\":true,"+
42                       "\"engine\":\""+engine+"\","+
43                       "\"query\":\""+escQ+"\","+
44                       "\"note\":\"SERPAPI_KEY fehlt – Demo-Modus aktiv\","+
45                       "\"organic_results\":["+
46                       "{\"title\":\"Demo: "+escQ+" – Ergebnis 1\",\"link\":\"https://example.com/"+escQ+"/1\",\"snippet\":\"Mock-Ergebnis (keine Live-Daten).\"},"+
47                       "{\"title\":\"Demo: "+escQ+" – Ergebnis 2\",\"link\":\"https://example.org/"+escQ+"/2\",\"snippet\":\"Mock-Ergebnis (keine Live-Daten).\"},"+
48                       "{\"title\":\"Demo: "+escQ+" – Ergebnis 3\",\"link\":\"https://example.net/"+escQ+"/3\",\"snippet\":\"Mock-Ergebnis (keine Live-Daten).\"}"
49                       + "]}";
50               return ResponseEntity.ok()
51                       .header(HttpHeaders.CACHE_CONTROL, "no-store")
52                       .contentType(MediaType.APPLICATION_JSON)
53                       .body(mock);
54           }
55           try {
56               Map<String, String> opts = new HashMap<>();
57               // Default broader result count if none provided
58               if (!allParams.containsKey("num")) {
59                   opts.put("num", "20");
60               }
61               // Pass-through allowed parameters
62               String[] pass = new String[]{
63                       "engine","location","num","hl","gl","tbm","start","page",
64                       "google_domain","safe","tbs","time_period","device","as_sitesearch",
65                       "include_domains","exclude_domains","filter","lr","cr","uule"
66               };
67               for (String k : pass) {
68                   String v = allParams.get(k);
69                   if (v != null && !v.isBlank()) opts.put(k, v);
70               }
71               String json = client.search(q, opts);
72               return ResponseEntity.ok()
73                       .header(HttpHeaders.CACHE_CONTROL, "no-store")
74                       .contentType(MediaType.APPLICATION_JSON)
75                       .body(json);
76           } catch (IllegalArgumentException iae) {
77               String msg = "{\"error\":\"" + iae.getMessage().replace("\"", "'") + "\"}";
78               return ResponseEntity.badRequest().contentType(MediaType.APPLICATION_JSON).body(msg);
79           } catch (Exception e) {
80               String msg = "{\"error\":\"SERP Anfrage fehlgeschlagen: " + e.getMessage().replace("\"", "'") + "\"}";
81               return ResponseEntity.status(502).contentType(MediaType.APPLICATION_JSON).body(msg);
82           }
83       }

85       /**
86        * Safe page fetch proxy – returns full HTML for a given URL with basic SSRF/XSS safeguards.
87        */
88       @GetMapping(value = "/fetch", produces = MediaType.APPLICATION_JSON_VALUE)
89       public ResponseEntity<Map<String, Object>> fetch(@RequestParam("url") String url,
90                                                        @RequestParam(value = "max", required = false, defaultValue = "2000000") int maxBytes) {
91           try {
92               URI uri = URI.create(url);
93               String scheme = uri.getScheme();
94               if (scheme == null || !(scheme.equalsIgnoreCase("http") || scheme.equalsIgnoreCase("https"))) {
95                   return ResponseEntity.badRequest().body(Map.of("error", "Nur http/https erlaubt"));
96               }
97               String host = uri.getHost();
98               if (host == null) return ResponseEntity.badRequest().body(Map.of("error", "Ungültige URL"));
99               InetAddress addr = InetAddress.getByName(host);
100              if (addr.isAnyLocalAddress() || addr.isLoopbackAddress() || addr.isLinkLocalAddress() || addr.isSiteLocalAddress()) {
101                  return ResponseEntity.status(403).body(Map.of("error", "Interne/Private Hosts sind blockiert"));
102              }
103              HttpClient http = HttpClient.newBuilder().followRedirects(HttpClient.Redirect.NORMAL).connectTimeout(Duration.ofSeconds(10)).build();
104              HttpRequest req = HttpRequest.newBuilder(uri)
105                      .timeout(Duration.ofSeconds(20))
106                      .header("User-Agent", "Mozilla/5.0 (SERP Preview Proxy)")
107                      .header("Accept", "text/html,application/xhtml+xml,text/plain;q=0.9,*/*;q=0.8")
108                      .GET().build();
109              HttpResponse<byte[]> res = http.send(req, HttpResponse.BodyHandlers.ofByteArray());
110              String ct = Optional.ofNullable(res.headers().firstValue("Content-Type").orElse("text/html")).orElse("text/html");
111              long len = res.body() == null ? 0 : res.body().length;
112              if (len > maxBytes) {
113                  return ResponseEntity.status(413).body(Map.of("error", "Seite zu groß", "contentType", ct, "length", len));
114              }
115              String html = new String(res.body(), java.nio.charset.Charset.forName("UTF-8"));
116              // Basic script stripping
117              html = html.replaceAll("(?is)<script[^>]*>.*?</script>", "");
118              Map<String, Object> out = new LinkedHashMap<>();
119              out.put("url", url);
120              out.put("status", res.statusCode());
121              out.put("contentType", ct);
122              out.put("length", len);
123              out.put("html", html);
124              return ResponseEntity.ok().header(HttpHeaders.CACHE_CONTROL, "no-store").body(out);
125          } catch (IllegalArgumentException iae) {
126              return ResponseEntity.badRequest().body(Map.of("error", iae.getMessage()));
127          } catch (IOException | InterruptedException e) {
128              return ResponseEntity.status(502).body(Map.of("error", "Fetch fehlgeschlagen: "+ e.getMessage()));
129          }
130      }
