62   class ApiConnector {
63       private final HttpClient http = HttpClient.newBuilder()
64               .connectTimeout(Duration.ofSeconds(12))
65               .build();
66   
67       public String query(String userPrompt) {
68           String provider = getEnvOrProp("AI_PROVIDER", "openai").trim().toLowerCase();
69           try {
70               if ("azure".equals(provider)) {
71                   return doAzureOpenAI(userPrompt);
72               } else {
73                   return doOpenAI(userPrompt);
74               }
75           } catch (HttpTimeoutException te) {
76               return "⚠️ API Timeout: " + te.getMessage();
77           } catch (Exception e) {
78               return "⚠️ API Fehler: " + e.getMessage();
79           }
80       }

82       // -------- OpenAI (public endpoint) --------
83       private String doOpenAI(String prompt) throws Exception {
84           String apiKey = getEnvOrProp("OPENAI_API_KEY", null);
85           if (isBlank(apiKey)) {
86               return missingConfig("OPENAI_API_KEY");
87           }
88           String model = getEnvOrProp("OPENAI_MODEL", "gpt-4o-mini");
89           String body = "{" +
90                   quote("model") + ":" + quote(model) + "," +
91                   quote("messages") + ":[{" + quote("role") + ":" + quote("user") + "," + quote("content") + ":" + quote(escape(prompt)) + "}]" +
92                   "}";
93   
94           HttpRequest req = HttpRequest.newBuilder()
95                   .uri(URI.create("https://api.openai.com/v1/chat/completions"))
96                   .timeout(Duration.ofSeconds(20))
97                   .header("Content-Type", "application/json")
98                   .header("Authorization", "Bearer " + apiKey)
99                   .header("User-Agent", "NurSystemGlobalV8/1.0")
100                  .POST(HttpRequest.BodyPublishers.ofString(body, StandardCharsets.UTF_8))
101                  .build();
102          HttpResponse<String> resp = http.send(req, HttpResponse.BodyHandlers.ofString(StandardCharsets.UTF_8));
103          if (resp.statusCode() / 100 != 2) {
104              return "⚠️ OpenAI HTTP " + resp.statusCode() + ": " + safeTrim(resp.body());
105          }
106          return extractAssistantText(resp.body());
107      }

109      // -------- Azure OpenAI --------
110      private String doAzureOpenAI(String prompt) throws Exception {
111          String endpoint = trimTrailingSlash(getEnvOrProp("AZURE_OPENAI_ENDPOINT", null));
112          String deployment = getEnvOrProp("AZURE_OPENAI_DEPLOYMENT", null);
113          String apiKey = getEnvOrProp("AZURE_OPENAI_API_KEY", null);
114          String apiVersion = getEnvOrProp("AZURE_OPENAI_API_VERSION", "2024-08-01-preview");
115          if (isBlank(endpoint) || isBlank(deployment) || isBlank(apiKey)) {
116              return missingConfig("AZURE_OPENAI_ENDPOINT / AZURE_OPENAI_DEPLOYMENT / AZURE_OPENAI_API_KEY");
117          }
118          String url = endpoint + "/openai/deployments/" + urlEncode(deployment) + "/chat/completions?api-version=" + urlEncode(apiVersion);
119          String body = "{" +
120                  quote("messages") + ":[{" + quote("role") + ":" + quote("user") + "," + quote("content") + ":" + quote(escape(prompt)) + "}]" +
121                  "}";
122  
123          HttpRequest req = HttpRequest.newBuilder()
124                  .uri(URI.create(url))
125                  .timeout(Duration.ofSeconds(20))
126                  .header("Content-Type", "application/json")
127                  .header("api-key", apiKey)
128                  .header("User-Agent", "NurSystemGlobalV8/1.0")
129                  .POST(HttpRequest.BodyPublishers.ofString(body, StandardCharsets.UTF_8))
130                  .build();
131          HttpResponse<String> resp = http.send(req, HttpResponse.BodyHandlers.ofString(StandardCharsets.UTF_8));
132          if (resp.statusCode() / 100 != 2) {
133              return "⚠️ Azure OpenAI HTTP " + resp.statusCode() + ": " + safeTrim(resp.body());
134          }
135          return extractAssistantText(resp.body());
136      }

138      // -------- Helpers --------
139      private static String extractAssistantText(String json) {
140          // Defensive, dependency‑free mini‑parser: versucht content‑Feld zu finden.
141          if (json == null || json.isEmpty()) return "(leere Antwort)";
142          // 1) Versuche OpenAI‑ähnliche Struktur: choices[0].message.content
143          int idx = json.indexOf("\"message\"");
144          if (idx >= 0) {
145              int cidx = json.indexOf("\"content\"", idx);
146              if (cidx >= 0) {
147                  int colon = json.indexOf(':', cidx);
148                  if (colon >= 0) {
149                      String tail = json.substring(colon + 1).trim();
150                      String val = readJsonStringValue(tail);
151                      if (!isBlank(val)) return val;
152                  }
153              }
154          }
155          // 2) Fallback: "content":"..." erstes Auftreten
156          int c = json.indexOf("\"content\"");
157          if (c >= 0) {
158              int colon = json.indexOf(':', c);
159              if (colon >= 0) {
160                  String tail = json.substring(colon + 1).trim();
161                  String val = readJsonStringValue(tail);
162                  if (!isBlank(val)) return val;
163              }
164          }
165          // 3) Letzter Fallback: gib gekürzten JSON‑Body zurück
166          return safeTrim(json);
167      }

169      private static String readJsonStringValue(String s) {
170          // Erwartet einen JSON‑String (beginnend mit \"), retourniert entschärften Inhalt.
171          if (s == null) return null;
172          int firstQuote = s.indexOf('"');
173          if (firstQuote < 0) return null;
174          String rest = s.substring(firstQuote + 1);
175          StringBuilder out = new StringBuilder();
176          boolean esc = false;
177          for (int i = 0; i < rest.length(); i++) {
178              char ch = rest.charAt(i);
179              if (esc) {
180                  // einfache Escape‑Behandlung
181                  if (ch == '"' || ch == '\\' || ch == '/') out.append(ch);
182                  else if (ch == 'b') out.append('\b');
183                  else if (ch == 'f') out.append('\f');
184                  else if (ch == 'n') out.append('\n');
185                  else if (ch == 'r') out.append('\r');
186                  else if (ch == 't') out.append('\t');
187                  else out.append(ch);
188                  esc = false;
189              } else {
190                  if (ch == '\\') { esc = true; }
191                  else if (ch == '"') { break; }
192                  else { out.append(ch); }
193              }
194          }
195          return out.toString().trim();
196      }

198      private static String quote(String s) { return '"' + s + '"'; }
199      private static String escape(String s) { return s == null ? "" : s.replace("\\", "\\\\").replace("\"", "\\\""); }
200      private static String urlEncode(String s) { try { return java.net.URLEncoder.encode(s, StandardCharsets.UTF_8); } catch (Exception e) { return s; } }
201      private static String safeTrim(String s) { if (s == null) return ""; s = s.trim(); return s.length() > 600 ? s.substring(0, 600) + "…" : s; }
202      private static String trimTrailingSlash(String s){ if (s==null) return null; while (s.endsWith("/")) s=s.substring(0, s.length()-1); return s; }
203      private static boolean isBlank(String s){ return s == null || s.trim().isEmpty(); }
204      private static String getEnvOrProp(String key, String def){
205          String v = System.getenv(key);
206          if (v == null) v = System.getProperty(key);
207          return v == null ? def : v;
208      }
209      private static String missingConfig(String keys){
210          return "⚠️ API nicht konfiguriert. Bitte setzen: " + keys + ". (AI_PROVIDER=openai|azure)";
211      }
