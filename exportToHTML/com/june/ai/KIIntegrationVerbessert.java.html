1    package com.june.ai;
2    
3    import org.slf4j.Logger;
4    import org.slf4j.LoggerFactory;
5    
6    import java.util.concurrent.CompletableFuture;
7    
8    /**
9     * Beispiel-Entry-Point, der eine sichere, fehlertolerante KI-Integration demonstriert.
10    *
11    * Liest Konfiguration aus Umgebungsvariablen:
12    *  - CLOUD_ENDPOINT
13    *  - API_KEY
14    */
15   public class KIIntegrationVerbessert {
16       private static final Logger logger = LoggerFactory.getLogger(KIIntegrationVerbessert.class);
17   
18       public static void main(String[] args) {
19           // 1. Sicheres Laden der Konfiguration aus Umgebungsvariablen
20           String cloudEndpoint = System.getenv("CLOUD_ENDPOINT");
21           String apiKey = System.getenv("API_KEY");
22   
23           if (cloudEndpoint == null || apiKey == null) {
24               logger.error("Endpoint oder API-Schlüssel nicht gefunden. Bitte Umgebungsvariablen festlegen.");
25               logger.error("Beispiel: set CLOUD_ENDPOINT=https://example.ai && set API_KEY=xyz123 (Windows Powershell/CMD)");
26               return;
27           }
28   
29           CloudKonfiguration konfig = new CloudKonfiguration();
30           konfig.setEndpoint(cloudEndpoint);
31           konfig.setApiKey(apiKey);
32   
33           KIBibliothek ki = new KIBibliothek();
34           try {
35               // 2. Fehlerbehandlung für Verbindung und Initialisierung
36               logger.info("Versuche Verbindung zur Cloud herzustellen...");
37               ki.verbindeMitCloud(konfig);
38               logger.info("Verbindung erfolgreich. Initialisiere KI-Modell...");
39               ki.initialisiereModell();
40               logger.info("KI-Modell initialisiert.");
41   
42               // 3. Asynchrone Ausführung der Aufgabe
43               String aufgabe = args != null && args.length > 0 ? String.join(" ", args) : "Ihre Aufgabe";
44               logger.info("Sende Aufgabe an KI: {}", aufgabe);
45   
46               CompletableFuture.supplyAsync(() -> ki.fuehreAufgabeAus(aufgabe))
47                   .thenAccept(ergebnis -> logger.info("Ergebnis der KI-Aufgabe: {}", ergebnis))
48                   .exceptionally(e -> {
49                       logger.error("Fehler bei der Ausführung der KI-Aufgabe: {}", e.getMessage());
50                       return null;
51                   })
52                   // Warten bis die async Pipeline abgeschlossen wurde (kleiner Demo-Blocker)
53                   .join();
54   
55           } catch (KIException e) {
56               logger.error("Kritischer Fehler bei der KI-Integration: {}", e.getMessage());
57               // Optional: Retry-Logik oder sauberes Beenden
58           }
59       }
60   }
