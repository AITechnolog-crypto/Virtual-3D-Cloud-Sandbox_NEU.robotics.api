106      static class Energy {
107          double solar=0, plasma=0, lev=0;  // 0..100
108          static double clamp(double v){ return Math.max(0, Math.min(100, v)); }
109          void addSolar(double f){ solar=clamp(solar+3.5*f); }
110          void addPlasma(double f){ plasma=clamp(plasma+5.0*f); }
111          void addLev(double f){ lev=clamp(lev+2.5*f); }
112          boolean spend(double need){ // nimmt anteilig aus solar/plasma
113              double pool = solar + plasma;
114              if (pool < need) return false;
115              double ratioSolar = solar / pool;
116              double useSolar = need * ratioSolar, usePlasma = need - useSolar;
117              solar = clamp(solar - useSolar); plasma = clamp(plasma - usePlasma);
118              return true;
119          }
120          String json(){ return "{\"solar\":"+fmt(solar)+",\"plasma\":"+fmt(plasma)+",\"levitation\":"+fmt(lev)+"}"; }
121          private String fmt(double v){ return String.format(Locale.US,"%.1f",v); }
122      }

156      static class VCPU {
157          int cores=64; double baseGHz=3.2, turboGHz=6.4; boolean turbo=false; double tempC=38.0; double load=0.0;
158          long ticks=0; double lastOps=0; // MOp/s
159          final Metrics m; final Energy e;
160          VCPU(Metrics m, Energy e){ this.m=m; this.e=e; }
161          synchronized void setTurbo(boolean on){ turbo=on; }
162          synchronized void setLoad(double l){ load=Math.max(0,Math.min(1,l)); }
163          synchronized void tick(long ms){
164              double ghz = turbo? turboGHz : baseGHz;
165              double watts = (20 + cores*ghz*0.9) * (0.4 + 0.6*load);
166              double need = (watts * ms/1000.0) / 25.0;
167              boolean ok = e.spend(need);
168              if(!ok){ turbo=false; ghz = baseGHz*0.7; load=Math.min(load,0.6); need=need*0.5; e.spend(need); }
169              tempC = Math.max(28.0, tempC + (watts*0.02) - (e.lev*0.01));
170              if (tempC>82.0){ turbo=false; load=Math.min(load,0.7); }
171              double ops = cores*ghz*1e3*load; // (Pseudo) MOp/s
172              lastOps = ops;
173              ticks += ms;
174              m.gauge("vcpu_temp_c", tempC);
175              m.gauge("vcpu_ops_mops", ops);
176              m.gauge("vcpu_load", load);
177              m.gauge("vcpu_turbo", turbo?1:0);
178          }
179          synchronized String json(){
180              return "{\"cores\":"+cores+",\"baseGHz\":"+fmt(baseGHz)+",\"turboGHz\":"+fmt(turboGHz)+",\"turbo\":"+turbo+
181                     ",\"tempC\":"+fmt(tempC)+",\"load\":"+fmt(load)+",\"opsMops\":"+fmt(lastOps)+"}";
182          }
183          private String fmt(double v){ return String.format(Locale.US,"%.2f", v); }
184      }

347      private static Map<String,String> query(URI u){
348          Map<String,String> m=new HashMap<>(); String q=u.getRawQuery(); if(q==null) return m;
349          for(String p:q.split("&")){ if(p.isEmpty()) continue; String[] kv=p.split("=",2);
350              String k=url(kv[0]); String v=kv.length>1?url(kv[1]):""; m.put(k,v); }
351          return m;
352      }
353      private static String url(String s){ try{ return URLDecoder.decode(s, StandardCharsets.UTF_8);}catch(Exception e){return s;} }
354      private static String readBody(HttpExchange ex) throws IOException{ try(InputStream in=ex.getRequestBody()){ return new String(in.readAllBytes(), StandardCharsets.UTF_8);} }
355      private static void json(HttpExchange ex, int code, String j){
356          try{ byte[] b=j.getBytes(StandardCharsets.UTF_8); ex.getResponseHeaders().set("Content-Type","application/json; charset=utf-8");
357              ex.sendResponseHeaders(code,b.length); try(OutputStream os=ex.getResponseBody()){ os.write(b);} }catch(IOException ignored){}

363      private boolean auth(HttpExchange ex){
364          if (!"POST".equalsIgnoreCase(ex.getRequestMethod())) return true; // GET offen (außer sensible)
365          String u=ex.getRequestHeaders().getFirst("X-User");
366          String a=ex.getRequestHeaders().getFirst("X-Auth"); // Hash
367          String v=ex.getRequestHeaders().getFirst("X-Voice"); // Voice-Hash optional
368          boolean ok = users.check(u, a, v);
369          if(!ok) json(ex,401,"{\"error\":\"unauthorized\"}");
370          return ok;
371      }

380      private void handle(HttpExchange ex) throws IOException {
381          String p = ex.getRequestURI().getPath();
382          Map<String,String> q = query(ex.getRequestURI());
383  
384          if ("/metrics".equals(p)){ text(ex,200,metrics.prom(),"text/plain; version=0.0.4"); return; }
385          if ("/healthz".equals(p)){ text(ex,200,"ok","text/plain"); return; }
386  
387          switch (p){
388              /* ---- Sessions / State ---- */
389              case "/api/session/start" -> {
390                  if(!auth(ex)) return;
391                  Session s = session(q.get("session"), ex.getRequestHeaders().getFirst("X-User"));
392                  s.note("session:start device="+q.getOrDefault("device","?"));
393                  metrics.gauge("windows_open", s.desk.windowCount());
394                  json(ex,200,s.json());
395              }
396              case "/api/state" -> { Session s=session(q.get("session"), null); json(ex,200,s.json()); }
397              case "/api/snapshot/save" -> {
398                  if(!auth(ex)) return;
399                  Session s = session(q.get("session"), null);
400                  Path f = Env.DATA.resolve("snapshot_"+s.id+".json");
401                  Files.writeString(f, s.json(), StandardCharsets.UTF_8, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
402                  json(ex,200,"{\"ok\":true,\"file\":\""+f.toString().replace("\\","/")+"\"}");
403              }
404  
405              /* ---- Biometry ---- */
406              case "/api/biometry/enroll" -> { // POST body = raw secret -> SHA-256
407                  if(!auth(ex)) return;
408                  String user = q.getOrDefault("user","Sanel Crnkic");
409                  String type = q.getOrDefault("type","key"); // key|marker|voice
410                  String body = readBody(ex);
411                  String h = Hash.sha256(body.getBytes(StandardCharsets.UTF_8)).toLowerCase();
412                  // Einfachheit: bestehendes Entry ersetzen (Demo)
413                  String curMarker = "b8d0d1ff1e0d0a207ed2f3eb0f1f2af33138997934796c6850c650f8d4339526";
414                  String key="", marker=curMarker, voice="";
415                  if ("key".equals(type)) key=h; else if("marker".equals(type)) marker=h; else if("voice".equals(type)) voice=h;
416                  users.register(user, key, marker, voice);
417                  json(ex,200,"{\"user\":\""+user+"\",\"type\":\""+type+"\",\"hash\":\""+h+"\"}");
418              }
419  
420              /* ---- Ritual & AirPods ---- */
421              case "/api/ritual/wind" -> {
422                  if(!auth(ex)) return;
423                  Session s=session(q.get("session"), null);
424                  int loops=Integer.parseInt(q.getOrDefault("loops","36"));
425                  double c=s.ritual.wind(loops); s.note("wind:"+loops);
426                  metrics.gauge("energy_solar", s.energy.solar);
427                  json(ex,200,"{\"charge\":"+String.format(Locale.US,"%.1f",c)+"}");
428              }
429              case "/api/ritual/inschallah" -> {

575      /* ======================= SERVER LOOP ======================= */
576      public void start(int port) throws IOException {
577          HttpServer server = HttpServer.create(new InetSocketAddress(port), 0);
578          server.createContext("/", this::handle);
579          server.setExecutor(Executors.newCachedThreadPool());
580          // kleiner VCPU-Scheduler
581          ScheduledExecutorService ses = Executors.newSingleThreadScheduledExecutor();
582          ses.scheduleAtFixedRate(()->{
583              for(Session s: sessions.values()) s.vcpu.tick(200);
584          }, 200, 200, TimeUnit.MILLISECONDS);
585          server.start();
586          System.out.println("HoloGlobePro läuft lokal: http://localhost:"+port);
587      }
