45       static class Hash {
46           static String hexSha256(byte[] data){
47               try {
48                   MessageDigest md = MessageDigest.getInstance("SHA-256");
49                   byte[] d = md.digest(data);
50                   StringBuilder sb=new StringBuilder();
51                   for(byte b:d) sb.append(String.format("%02x", b));
52                   return sb.toString();
53               } catch(Exception e){ throw new RuntimeException(e); }
54           }
55           static String hexSha256OfFile(Path p){
56               try { return hexSha256(Files.readAllBytes(p)); }
57               catch(IOException e){ throw new UncheckedIOException(e); }
58           }

62       static class MetricsSink {
63           private final Map<String, Double> gauges = new ConcurrentHashMap<>();
64           private final Map<String, Long> counters = new ConcurrentHashMap<>();
65           void gauge(String name, double val){ gauges.put(name, val); }
66           void add(String name, double delta){ gauges.merge(name, delta, Double::sum); }
67           void count(String name){ counters.merge(name, 1L, Long::sum); }
68           String asPrometheus() {
69               StringBuilder sb=new StringBuilder();
70               for (var e : counters.entrySet()) {
71                   sb.append("# TYPE ").append(e.getKey()).append(" counter\n")
72                     .append(e.getKey()).append(" ").append(e.getValue()).append("\n");
73               }
74               for (var e : gauges.entrySet()) {
75                   sb.append("# TYPE ").append(e.getKey()).append(" gauge\n")
76                     .append(e.getKey()).append(" ").append(String.format(Locale.US,"%.3f", e.getValue())).append("\n");
77               }
78               return sb.toString();
79           }
80       }

83       static class UserRegistry {
84           static class Entry { final String keyHash; final String markerHash;
85               Entry(String k, String m){ keyHash=k==null?"":k.toLowerCase(); markerHash=m==null?"":m.toLowerCase(); } }
86           private final Map<String, Entry> map = new ConcurrentHashMap<>();
87           void register(String user, String keyHash, String markerHash){ map.put(user, new Entry(keyHash, markerHash)); }
88           boolean check(String user, String provided){
89               if (provided==null || user==null) return false;
90               Entry e = map.get(user);
91               if (e==null) return false;
92               String p = provided.toLowerCase();
93               return (!e.keyHash.isEmpty() && p.equals(e.keyHash)) || (!e.markerHash.isEmpty() && p.equals(e.markerHash));
94           }
95       }

98       static class Wallet {
99           private double bal=0;
100          private final Path ledger = Env.DATA_DIR.resolve("wallet_ledger.csv");
101          synchronized void credit(double amt, String src){ if (amt<=0) return; bal+=amt; log("CREDIT",amt,src); }
102          synchronized boolean debit(double amt, String why){ if (amt<=0 || bal<amt) return false; bal-=amt; log("DEBIT",amt,why); return true; }
103          synchronized double balance(){ return bal; }
104          private void log(String t, double a, String m){
105              try {
106                  Files.writeString(ledger, Env.nowIso()+";"+t+";"+String.format(Locale.US,"%.2f", a)+";"+m+"\n",
107                          StandardCharsets.UTF_8, StandardOpenOption.CREATE, StandardOpenOption.APPEND);
108              } catch (IOException ignored){}
109          }
110          synchronized String asJson(){
111              return "{\"balance\":"+String.format(Locale.US,"%.2f", bal)+"}";
112          }
113      }

153      static class EnergyPack {
154          double solar=0, plasma=0, levitation=0; // 0..100
155          void harvestSolar(double f){ solar = clamp(solar + 3.5*f, 0, 100); }
156          void pulsePlasma(double f){ plasma = clamp(plasma + 5.0*f, 0, 100); }
157          void liftTick(double f){ levitation = clamp(levitation + 2.5*f, 0, 100); }
158          static double clamp(double v,double a,double b){ return Math.max(a, Math.min(b, v)); }
159          String json(){ return "{\"solar\":"+round(solar)+",\"plasma\":"+round(plasma)+",\"levitation\":"+round(levitation)+"}"; }
160          private String round(double v){ return String.format(Locale.US,"%.1f", v); }
161      }
162      static class RitualEngine {
163          double charge=0;         // 0..100
164          boolean blessed=false;   // Inschallah gesprochen
165          String last="";
166          final MetricsSink metrics;
167          final EnergyPack energy;
168          RitualEngine(MetricsSink m, EnergyPack e){ metrics=m; energy=e; }
169          synchronized double wind(int cycles){
170              // wickelnde Kreisbewegung: Solar wird gekoppelt, Ladung steigt
171              cycles = Math.max(1, Math.min(1000, cycles));
172              for(int i=0;i<cycles;i++){
173                  energy.harvestSolar(1.0);
174                  charge = EnergyPack.clamp(charge + 0.8, 0, 100);
175              }
176              last="wind:"+cycles;
177              metrics.count("ritual_wind_calls_total");
178              metrics.gauge("ritual_charge", charge);
179              return charge;
180          }
181          synchronized void inschallah(){
182              blessed=true;
183              energy.pulsePlasma(1.0);
184              last="inschallah";
185              metrics.count("ritual_inschallah_calls_total");
186          }
187          synchronized boolean openGlobe(DreamGlobe globe){
188              boolean ok = globe.openIfAllowed(blessed, charge);
189              last = ok? "globe_opened" : "globe_denied";
190              if (ok) metrics.count("globe_open_total"); else metrics.count("globe_denied_total");
191              return ok;
192          }
193          synchronized String json(){
194              return "{\"charge\":"+String.format(Locale.US,"%.1f",charge)+",\"blessed\":"+blessed+",\"last\":\""+last+"\"}";
195          }

200      static class AirPodsBridge {
201          final RitualEngine ritual;
202          final EnergyPack energy;
203          final MetricsSink metrics;
204          AirPodsBridge(RitualEngine r, EnergyPack e, MetricsSink m){ ritual=r; energy=e; metrics=m; }
205          synchronized String handle(Gesture g){
206              switch (g){
207                  case doubleTap -> {
208                      double c = ritual.wind(12);
209                      metrics.count("airpods_doubletap_total");
210                      return "wind:charge="+String.format(Locale.US,"%.1f", c);
211                  }
212                  case longPress -> {
213                      ritual.inschallah();
214                      metrics.count("airpods_longpress_total");
215                      return "inschallah:ok";
216                  }
217                  case tripleTap -> {
218                      metrics.count("airpods_tripletap_total");
219                      return "tripleTap:ready";
220                  }
221                  case tilt -> {
222                      energy.liftTick(1.0);
223                      metrics.count("airpods_tilt_total");
224                      return "levitation:"+String.format(Locale.US,"%.1f", energy.levitation);
225                  }
226              }
227              return "noop";
228          }

282      private static Map<String,String> query(URI uri){
283          Map<String,String> m=new HashMap<>();
284          String q=uri.getRawQuery(); if(q==null) return m;
285          for(String p:q.split("&")){
286              if(p.isEmpty()) continue;
287              String[] kv=p.split("=",2);
288              String k = urlDecode(kv[0]);
289              String v = kv.length>1 ? urlDecode(kv[1]) : "";
290              m.put(k, v);
291          }
292          return m;
293      }
294      private static String urlDecode(String s){
295          try { return URLDecoder.decode(s, StandardCharsets.UTF_8); }
296          catch (Exception e){ return s; }
297      }
298      private static String body(HttpExchange ex) throws IOException {
299          try (InputStream in = ex.getRequestBody()){
300              return new String(in.readAllBytes(), StandardCharsets.UTF_8);
301          }
302      }
303      private boolean requireAuth(HttpExchange ex){
304          // Für schreibende Endpunkte nötig (POST). GET /metrics ist offen.
305          if (!"POST".equalsIgnoreCase(ex.getRequestMethod())) return true;
306          String user = ex.getRequestHeaders().getFirst("X-User");
307          String auth = ex.getRequestHeaders().getFirst("X-Auth");
308          boolean ok = users.check(user, auth);
309          if(!ok){
310              reply(ex, 401, "{\"error\":\"unauthorized\"}");
311              return false;
312          }
313          return true;
314      }
315      private static void reply(HttpExchange ex, int code, String json){
316          try {
317              byte[] b = json.getBytes(StandardCharsets.UTF_8);
318              ex.getResponseHeaders().set("Content-Type", "application/json; charset=utf-8");
319              ex.sendResponseHeaders(code, b.length);
320              try(OutputStream os = ex.getResponseBody()){ os.write(b); }
321          } catch (IOException ignored){}
322      }
323      private static void replyText(HttpExchange ex, int code, String text, String ctype){
324          try {
325              byte[] b = text.getBytes(StandardCharsets.UTF_8);
326              ex.getResponseHeaders().set("Content-Type", ctype);
327              ex.sendResponseHeaders(code, b.length);
328              try(OutputStream os = ex.getResponseBody()){ os.write(b); }
329          } catch (IOException ignored){}
330      }

339      /* ============================== HTTP ROUTER ============================== */
340      private void handle(HttpExchange ex) throws IOException {
341          String path = ex.getRequestURI().getPath();
342          if ("/metrics".equals(path)) { replyText(ex, 200, metrics.asPrometheus(), "text/plain; version=0.0.4"); return; }
343          if ("/healthz".equals(path)) { replyText(ex, 200, "ok", "text/plain"); return; }
344  
345          Map<String,String> q = query(ex.getRequestURI());
346          switch (path){
347              case "/api/session/start" -> {
348                  if (!requireAuth(ex)) return;
349                  Session s = getOrCreateSession(q.get("session"));
350                  s.note("session:start device="+q.getOrDefault("device","?"));
351                  reply(ex, 200, s.json());
352              }
353              case "/api/state" -> {
354                  Session s = getOrCreateSession(q.get("session"));
355                  reply(ex, 200, s.json());
356              }
357              case "/api/ritual/wind" -> {
358                  if (!requireAuth(ex)) return;
359                  Session s = getOrCreateSession(q.get("session"));
360                  int cycles = Integer.parseInt(q.getOrDefault("cycles","12"));
361                  double c = s.ritual.wind(cycles);
362                  s.note("ritual:wind cycles="+cycles);
363                  metrics.gauge("energy_solar", s.energy.solar);
364                  reply(ex, 200, "{\"charge\":"+String.format(Locale.US,"%.1f", c)+"}");
365              }
366              case "/api/ritual/inschallah" -> {
367                  if (!requireAuth(ex)) return;
368                  Session s = getOrCreateSession(q.get("session"));
369                  s.ritual.inschallah();
370                  s.note("ritual:inschallah");
371                  metrics.gauge("energy_plasma", s.energy.plasma);
372                  reply(ex, 200, "{\"blessed\":true}");
373              }
374              case "/api/ritual/openGlobe" -> {
375                  if (!requireAuth(ex)) return;
376                  Session s = getOrCreateSession(q.get("session"));
377                  boolean ok = s.ritual.openGlobe(s.globe);
378                  s.note("ritual:openGlobe ok="+ok);
379                  reply(ex, ok?200:409, "{\"open\":"+ok+",\"apps\":"+s.globe.listAppsJson()+"}");
380              }
381              case "/api/airpods/gesture" -> {
382                  if (!requireAuth(ex)) return;
383                  Session s = getOrCreateSession(q.get("session"));
384                  String g = q.getOrDefault("g","doubleTap");
385                  Gesture gg = Gesture.valueOf(g);
386                  String res = s.air.handle(gg);
387                  s.lastGesture=g; s.note("airpods:"+g);
388                  // tripleTap → automatischer Versuch Globe öffnen
389                  if (gg==Gesture.tripleTap){
390                      boolean ok = s.ritual.openGlobe(s.globe);
391                      if (ok) s.note("globe:auto-open");
392                  }
393                  metrics.gauge("energy_levitation", s.energy.levitation);
394                  reply(ex, 200, "{\"result\":\""+res+"\",\"state\":"+s.json()+"}");
395              }
396              case "/api/contacts/get" -> {
397                  reply(ex, 200, contacts.dump());
398              }
399              case "/api/contacts/set" -> {
400                  if (!requireAuth(ex)) return;
401                  contacts.set(q.get("name"), q.get("email"), q.get("phone"));
402                  reply(ex, 200, contacts.dump());
403              }
404              case "/api/wallet/balance" -> {
405                  reply(ex, 200, wallet.asJson());
406              }
407              case "/api/wallet/credit" -> {
408                  if (!requireAuth(ex)) return;
409                  double amount = Double.parseDouble(q.getOrDefault("amount","0"));
410                  wallet.credit(amount, "api");
411                  reply(ex, 200, wallet.asJson());
412              }
413              case "/api/wallet/debit" -> {
414                  if (!requireAuth(ex)) return;
415                  double amount = Double.parseDouble(q.getOrDefault("amount","0"));
416                  boolean ok = wallet.debit(amount, "api");
417                  reply(ex, ok?200:409, "{\"ok\":"+ok+",\"balance\":"+String.format(Locale.US,"%.2f", wallet.balance())+"}");
418              }
419              case "/api/notes/add" -> {
420                  if (!requireAuth(ex)) return;
421                  String who = ex.getRequestHeaders().getFirst("X-User");
422                  String text = q.getOrDefault("text", body(ex));
423                  notes.add(who==null?"anon":who, text==null?"":text);
424                  reply(ex, 200, "{\"ok\":true}");
425              }
426              case "/api/users/register" -> {
427                  // POST /api/users/register?user=Name  (Body = Rohdaten/Key, z.B. Datei-Inhalt)
428                  if (!requireAuth(ex)) return; // nur bestehender Admin kann neue hinzufügen – simpel
429                  String user = q.get("user");
430                  if (user==null || user.isBlank()){ reply(ex, 400, "{\"error\":\"user required\"}"); return; }
431                  String b = body(ex);
432                  String keyHash = Hash.hexSha256(b.getBytes(StandardCharsets.UTF_8));
433                  users.register(user, keyHash, "");
434                  reply(ex, 200, "{\"user\":\""+user+"\",\"keyHash\":\""+keyHash+"\"}");
435              }
436              default -> reply(ex, 404, "{\"error\":\"not found\"}");
437          }
438      }

441      public void startServer(int port) throws IOException {
442          HttpServer server = HttpServer.create(new InetSocketAddress(port), 0);
443          server.createContext("/", this::handle);
444          server.setExecutor(Executors.newFixedThreadPool(8));
445          server.start();
446          System.out.println("HoloAirPodsSuite läuft auf http://localhost:"+port);
447      }
