1    package com.june.production.cloud;
2    
3    import java.util.logging.Level;
4    import java.util.logging.Logger;
5    
6    /**
7     * Simpler Ressourcenmanager: weist eine feste Anzahl Ressourcen zu und führt Buch.
8     */
9    public class CloudResourceManager implements ICloudResourceManager {
10       private static final Logger logger = Logger.getLogger(CloudResourceManager.class.getName());
11       private int allocatedResources = 0;
12   
13       @Override
14       public void allocateResources() throws ResourceAllocationException {
15           try {
16               int requestedResources = 10; // Beispielwert
17               allocatedResources += requestedResources;
18               logger.info(requestedResources + " Ressourcen zugewiesen. Aktuell zugewiesene Ressourcen: " + allocatedResources);
19           } catch (Exception e) {
20               logger.log(Level.SEVERE, "Fehler bei der Ressourcenzuweisung: " + e.getMessage(), e);
21               throw new ResourceAllocationException("Ressourcenzuweisung fehlgeschlagen.", e);
22           }
23       }
24   
25       @Override
26       public void deallocateResources() {
27           if (allocatedResources > 0) {
28               allocatedResources = 0;
29               logger.info("Alle Ressourcen freigegeben.");
30           } else {
31               logger.info("Keine Ressourcen zum Freigeben vorhanden.");
32           }
33       }
34   }
