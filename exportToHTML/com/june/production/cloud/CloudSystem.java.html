1    package com.june.production.cloud;
2    
3    import java.util.logging.Level;
4    import java.util.logging.Logger;
5    
6    /**
7     * Orchestrator für Cloudbetrieb: allokiert Ressourcen, integriert Technologie, startet.
8     */
9    public class CloudSystem {
10       private static final Logger logger = Logger.getLogger(CloudSystem.class.getName());
11       private final ICloudResourceManager resourceManager;
12       private final ITechnologyIntegrator technologyIntegrator;
13   
14       public CloudSystem(ICloudResourceManager resourceManager, ITechnologyIntegrator technologyIntegrator) {
15           this.resourceManager = resourceManager;
16           this.technologyIntegrator = technologyIntegrator;
17       }
18   
19       public void operateSystem() {
20           try {
21               resourceManager.allocateResources();
22               technologyIntegrator.integrateTechnology();
23               logger.info("Cloud-System erfolgreich gestartet.");
24           } catch (ResourceAllocationException | IntegrationFailedException e) {
25               logger.log(Level.SEVERE, "Fehler im Cloud-Systembetrieb: " + e.getMessage(), e);
26           }
27       }
28   
29       /** Öffentliche Hilfsmethode entsprechend der Issue-Vorlage. */
30       public void deallocateAll() {
31           resourceManager.deallocateResources();
32       }
33   
34       // Demo main
35       public static void main(String[] args) {
36           ICloudResourceManager resourceManager = new CloudResourceManager();
37           ITechnologyIntegrator technologyIntegrator = new TechnologyIntegrator();
38           CloudSystem cloudSystem = new CloudSystem(resourceManager, technologyIntegrator);
39           cloudSystem.operateSystem();
40           cloudSystem.operateSystem();
41           cloudSystem.deallocateAll();
42       }
43   }
