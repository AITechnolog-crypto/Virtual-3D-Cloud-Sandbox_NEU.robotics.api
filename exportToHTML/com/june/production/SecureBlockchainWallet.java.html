1    package com.june.production;

3    import java.util.logging.Logger;

5    /**
6     * Sichere Blockchain-Wallet-Attrappe mit einfachem Kontostand.
7     */
8    public class SecureBlockchainWallet implements IBlockchainWallet {
9        private static final Logger logger = Logger.getLogger(SecureBlockchainWallet.class.getName());
10   
11       private double balance;
12   
13       public SecureBlockchainWallet() {
14           this(10_000.0); // Default-Budget
15       }
16   
17       public SecureBlockchainWallet(double initialBalance) {
18           this.balance = initialBalance;
19       }
20   
21       @Override
22       public double getBalance() {
23           return balance;
24       }
25   
26       @Override
27       public void processTransaction(Transaction transaction) {
28           logger.info("Blockchain Transaktion durchgeführt: " + transaction);
29           // Beispiel: Ziehe symbolisch einen kleinen Betrag ab
30           this.balance = Math.max(0.0, this.balance - 100.0);
31       }
32   }
