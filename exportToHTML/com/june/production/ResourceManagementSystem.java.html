1    package com.june.production;
2    
3    import java.util.ArrayList;
4    import java.util.List;
5    import java.util.logging.Level;
6    import java.util.logging.Logger;
7    
8    /**
9     * Ressourcenverwaltung: Extraktion → ML-Training → Vorhersage → Transaktion.
10    */
11   public class ResourceManagementSystem {
12       private static final Logger logger = Logger.getLogger(ResourceManagementSystem.class.getName());
13   
14       private final IResourceExtractor extractor;
15       private final IMachineLearningModel mlModel;
16       private final IBlockchainWallet wallet;
17       private final IErrorHandler errorHandler;
18   
19       public ResourceManagementSystem(IResourceExtractor extractor,
20                                       IMachineLearningModel mlModel,
21                                       IBlockchainWallet wallet,
22                                       IErrorHandler errorHandler) {
23           this.extractor = extractor;
24           this.mlModel = mlModel;
25           this.wallet = wallet;
26           this.errorHandler = errorHandler;
27       }
28   
29       public void manageResources() {
30           try {
31               List<Resource> resources = extractor.extractResources();
32               List<ResourceData> data = convertToData(resources);
33               mlModel.train(data);
34               ResourcePrediction prediction = mlModel.predict(new ResourceData("NEU", 1234.0));
35               wallet.processTransaction(new Transaction(prediction));
36               logger.info("Ressourcenverteilung erfolgreich durchgeführt.");
37           } catch (Exception e) {
38               if (errorHandler != null) {
39                   errorHandler.handleException(e, "Fehler im Ressourcenverwaltungssystem.");
40               } else {
41                   logger.log(Level.SEVERE, "Fehler im Ressourcenverwaltungssystem.", e);
42               }
43           }
44       }
45   
46       private static List<ResourceData> convertToData(List<Resource> resources) {
47           List<ResourceData> out = new ArrayList<>();
48           if (resources == null) return out;
49           for (Resource r : resources) {
50               if (r == null) continue;
51               out.add(new ResourceData(r.getType(), r.getAmount()));
52           }
53           return out;
54       }
55   
56       // Demo main
57       public static void main(String[] args) {
58           ResourceManagementSystem rms = new ResourceManagementSystem(
59                   new AdvancedResourceExtractor(),
60                   new ResourceOptimizationModel(),
61                   new SecureBlockchainWallet(5000.0),
62                   new ErrorHandler()
63           );
64           rms.manageResources();
65       }
66   }
