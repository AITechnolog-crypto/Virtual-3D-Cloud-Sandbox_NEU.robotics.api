1    package com.june.production;
2    
3    import java.util.logging.Level;
4    import java.util.logging.Logger;
5    
6    /**
7     * Orchestriert Wallet, Ressourcenbeschaffung und Produktionsstart.
8     */
9    public class AIProductionManager {
10       private static final Logger logger = Logger.getLogger(AIProductionManager.class.getName());
11   
12       private final IBlockchainWallet blockchainWallet;
13       private final IResourceAcquisition resourceAcquisition;
14       private final IProductionControl productionControl;
15   
16       public AIProductionManager(IBlockchainWallet blockchainWallet,
17                                  IResourceAcquisition resourceAcquisition,
18                                  IProductionControl productionControl) {
19           this.blockchainWallet = blockchainWallet;
20           this.resourceAcquisition = resourceAcquisition;
21           this.productionControl = productionControl;
22       }
23   
24       public void manageProduction() {
25           try {
26               double budget = blockchainWallet.getBalance();
27               if (budget < 0) {
28                   throw new InsufficientFundsException("Negativer Kontostand.");
29               }
30               resourceAcquisition.acquireResources(budget);
31               productionControl.startProduction();
32               logger.info("Produktion erfolgreich abgeschlossen.");
33           } catch (InsufficientFundsException | ResourceAcquisitionException | ProductionStartException e) {
34               logger.log(Level.SEVERE, "Fehler im Produktionsmanagement: " + e.getMessage(), e);
35           }
36       }
37   
38       // Kleiner Demo-Entry-Point
39       public static void main(String[] args) {
40           AIProductionManager mgr = new AIProductionManager(
41                   new SecureBlockchainWallet(6_000.0),
42                   new ResourceAcquisitionImplementation(),
43                   new ProductionControlImplementation()
44           );
45           mgr.manageProduction();
46       }
47   }
