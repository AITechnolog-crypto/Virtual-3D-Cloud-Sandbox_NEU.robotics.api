1    package com.june.production;
2    
3    import java.util.List;
4    import java.util.logging.Logger;
5    
6    /**
7     * Dummy-Optimierungsmodell: trainiert auf Ressourcendaten und gibt eine einfache Vorhersage zurück.
8     */
9    public class ResourceOptimizationModel implements IMachineLearningModel {
10       private static final Logger logger = Logger.getLogger(ResourceOptimizationModel.class.getName());
11   
12       private boolean trained;
13       private double avgAmount;
14   
15       @Override
16       public void train(List<ResourceData> data) {
17           if (data == null || data.isEmpty()) {
18               logger.info("Keine Trainingsdaten vorhanden.");
19               trained = false;
20               avgAmount = 0.0;
21               return;
22           }
23           double sum = 0.0; int n = 0;
24           for (ResourceData d : data) { if (d != null) { sum += d.getAmount(); n++; } }
25           if (n > 0) { avgAmount = sum / n; trained = true; }
26           logger.info("Training abgeschlossen. Durchschnittsmenge=" + avgAmount);
27       }
28   
29       @Override
30       public ResourcePrediction predict(ResourceData newData) {
31           if (!trained) {
32               return new ResourcePrediction(0.0, 0.1);
33           }
34           double base = newData != null ? newData.getAmount() : avgAmount;
35           return new ResourcePrediction(base * 0.9 + avgAmount * 0.1, 0.7);
36       }
37   }
