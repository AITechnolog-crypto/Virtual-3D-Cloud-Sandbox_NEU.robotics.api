1    package com.june.factory;
2    
3    import java.util.logging.Level;
4    import java.util.logging.Logger;
5    
6    /**
7     * Konkrete Fabrik, die Ressourcen allokiert, eine Blockchain-Transaktion ausführt
8     * und anschließend einen Energiespeicher erzeugt.
9     */
10   public class EnergyStorageFactory implements IEnergyStorageFactory {
11       private final Logger logger = Logger.getLogger(EnergyStorageFactory.class.getName());
12       private final IResourceManagement resourceManagement;
13       private final IBlockchainWallet blockchainWallet;
14   
15       public EnergyStorageFactory(IResourceManagement resourceManagement, IBlockchainWallet blockchainWallet) {
16           this.resourceManagement = resourceManagement;
17           this.blockchainWallet = blockchainWallet;
18       }
19   
20       @Override
21       public EnergyStorage produceEnergyStorage() throws ResourceAllocationException, BlockchainTransactionException {
22           try {
23               resourceManagement.allocateResourcesForGood();
24               blockchainWallet.processTransaction(new Transaction("Produktion", "EnergyStorage"));
25               EnergyStorage storage = new EnergyStorage("AntarcticModule", 1000);
26               logger.info("Energiespeicher produziert: " + storage);
27               return storage;
28           } catch (ResourceAllocationException | BlockchainTransactionException e) {
29               logger.log(Level.SEVERE, "Fehler bei der Produktion des Energiespeichers: " + e.getMessage(), e);
30               throw e; // Exception weiterwerfen
31           }
32       }
33   }
