1    package com.june.resources;
2    
3    import java.util.EnumMap;
4    import java.util.Map;
5    
6    /**
7     * Einfacher Ressourcen-Speicher mit Bewertung je Einheit.
8     * Entspricht der Beschreibung: Zuweisung, Abfrage, Freigabe mit Exception und Gesamtwert.
9     */
10   public class ResourceStorage {
11   
12       private final Map<ResourceType, Integer> resources;
13   
14       public ResourceStorage() {
15           this.resources = new EnumMap<>(ResourceType.class);
16           for (ResourceType type : ResourceType.values()) {
17               resources.put(type, 0);
18           }
19       }
20   
21       /**
22        * Fügt eine Anzahl Einheiten eines Ressourcentyps hinzu. Negative/0 wird abgewiesen.
23        * @return true, falls erfolgreich zugewiesen wurde; false bei ungültiger Menge
24        */
25       public boolean allocateResources(ResourceType resourceType, int quantity) {
26           if (resourceType == null || quantity <= 0) {
27               return false;
28           }
29           resources.put(resourceType, resources.get(resourceType) + quantity);
30           return true;
31       }
32   
33       /**
34        * Liefert gespeicherte Einheitenzahl des Typs.
35        */
36       public int getResourceUnits(ResourceType resourceType) {
37           if (resourceType == null) return 0;
38           return resources.getOrDefault(resourceType, 0);
39       }
40   
41       /**
42        * Gibt eine Anzahl Einheiten frei (subtrahiert). Wirft bei unzureichendem Bestand eine Exception.
43        */
44       public void releaseResources(ResourceType resourceType, int quantity) throws InsufficientResourcesException {
45           if (resourceType == null || quantity <= 0) return; // no-op für ungültige Parameter
46           int available = resources.getOrDefault(resourceType, 0);
47           if (available < quantity) {
48               throw new InsufficientResourcesException(resourceType.name(), quantity, available);
49           }
50           resources.put(resourceType, available - quantity);
51       }
52   
53       /**
54        * Berechnet den Gesamtwert aller Ressourcen.
55        */
56       public int getTotalValue() {
57           int totalValue = 0;
58           for (Map.Entry<ResourceType, Integer> entry : resources.entrySet()) {
59               totalValue += entry.getKey().getValue() * entry.getValue();
60           }
61           return totalValue;
62       }
63   
64       /**
65        * Ressourcentypen mit Wert je Einheit.
66        */
67       public enum ResourceType {
68           GOLD(10),
69           PLATINUM(25);
70   
71           private final int value;
72           ResourceType(int value) { this.value = value; }
73           public int getValue() { return value; }
74       }
75   
76       /**
77        * Kleiner Demo-Entry-Point entsprechend der Beschreibung.
78        */
79       public static void main(String[] args) {
80           ResourceStorage storage = new ResourceStorage();
81           storage.allocateResources(ResourceType.GOLD, 100);
82           storage.allocateResources(ResourceType.PLATINUM, 50);
83   
84           System.out.println("Gold: " + storage.getResourceUnits(ResourceType.GOLD));
85           System.out.println("Platin: " + storage.getResourceUnits(ResourceType.PLATINUM));
86           System.out.println("Gesamtwert: " + storage.getTotalValue());
87   
88           try {
89               storage.releaseResources(ResourceType.GOLD, 150); // Sollte Exception werfen
90           } catch (InsufficientResourcesException e) {
91               System.err.println(e.getMessage());
92           }
93   
94           try {
95               storage.releaseResources(ResourceType.GOLD, 50);
96               System.out.println("Gold nach Freigabe: " + storage.getResourceUnits(ResourceType.GOLD));
97           } catch (InsufficientResourcesException e) {
98               System.err.println(e.getMessage());
99           }
100      }
101  }
