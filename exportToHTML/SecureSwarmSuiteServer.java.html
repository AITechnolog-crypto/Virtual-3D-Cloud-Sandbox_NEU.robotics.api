147          static String bitcoinUri(String address, Double amountBtc){
148              if (amountBtc==null) return "bitcoin:"+address;
149              NumberFormat nf=NumberFormat.getInstance(Locale.US); nf.setGroupingUsed(false); nf.setMaximumFractionDigits(8);
150              return "bitcoin:"+address+"?amount="+nf.format(amountBtc);
151          }
152          static String ethereumUri(String address){ return "ethereum:"+address; }
153          static String epcSepaPayload(String name,String iban,Double amountEUR,String rem){
154              String version="001", encoding="1", service="SCT", bic="";
155              String amount = (amountEUR==null) ? "" : "EUR"+String.format(Locale.US,"%.2f",Math.max(0.01,amountEUR));
156              String r=(rem==null)?"":rem;
157              return String.join("\n", Arrays.asList("BCD",version,encoding,service,bic,name,iban.replace(" ",""),amount,"",r,"","") );
158          }
159          @SuppressWarnings("unchecked")
160          static boolean writePngIfZXing(String payload, File out, int size){
161              try {
162                  Class<?> bfClass   = Class.forName("com.google.zxing.BarcodeFormat");
163                  Object qrFormat    = Enum.valueOf((Class<Enum>) bfClass.asSubclass(Enum.class), "QR_CODE");
164                  Class<?> writerCls = Class.forName("com.google.zxing.qrcode.QRCodeWriter");
165                  Object writer      = writerCls.getDeclaredConstructor().newInstance();
166                  java.lang.reflect.Method encode = writerCls.getMethod("encode", String.class, bfClass, int.class, int.class, Map.class);
167                  Object bitMatrix    = encode.invoke(writer, payload, qrFormat, size, size, null);
168                  Class<?> bmClass    = Class.forName("com.google.zxing.common.BitMatrix");
169                  int w = (int) bmClass.getMethod("getWidth").invoke(bitMatrix);
170                  int h = (int) bmClass.getMethod("getHeight").invoke(bitMatrix);
171                  java.awt.image.BufferedImage img = new java.awt.image.BufferedImage(w,h,java.awt.image.BufferedImage.TYPE_INT_RGB);
172                  java.lang.reflect.Method get = bmClass.getMethod("get", int.class, int.class);
173                  for(int y=0;y<h;y++) for(int x=0;x<w;x++){ boolean on=(boolean)get.invoke(bitMatrix,x,y); img.setRGB(x,y,on?0x000000:0xFFFFFF); }
174                  javax.imageio.ImageIO.write(img,"png",out);
175                  System.out.println("[QR] PNG geschrieben → "+out.getAbsolutePath());
176                  return true;
177              } catch (ClassNotFoundException e) {
178                  System.out.println("[QR] ZXing nicht gefunden – PNG-Export übersprungen.");
179                  return false;
180              } catch (Throwable t) { System.out.println("[QR] Fehler: "+t.getMessage()); return false; }
181          }

362      static final class MetricsRegistry {
363          private final ConcurrentMap<String, AtomicLong> counters = new ConcurrentHashMap<>();
364          private final ConcurrentMap<String, AtomicLong> errors   = new ConcurrentHashMap<>();
365          private final ConcurrentMap<String, Histogram>   hists   = new ConcurrentHashMap<>();
366          private final AtomicLong leads = new AtomicLong(0);

368          void incReq(String path, String method){ key(counters, "requests_total{path=\""+path+"\",method=\""+method+"\"}").incrementAndGet(); }
369          void incErr(String path, String method){ key(errors, "errors_total{path=\""+path+"\",method=\""+method+"\"}").incrementAndGet(); }
370          void observe(String name, long ms){ hists.computeIfAbsent(name, k->Histogram.defaultMs()).observe(ms); }
371          void incLead(){ leads.incrementAndGet(); }

373          String prometheus(){
374              StringBuilder sb=new StringBuilder();
375              sb.append("# HELP requests_total Total HTTP requests\n# TYPE requests_total counter\n");
376              counters.forEach((k,v)-> sb.append(k).append(' ').append(v.get()).append('\n'));
377              sb.append("\n# HELP errors_total Total HTTP 5xx/4xx\n# TYPE errors_total counter\n");
378              errors.forEach((k,v)-> sb.append(k).append(' ').append(v.get()).append('\n'));
379              sb.append("\n# HELP http_request_duration_ms Histogram of request durations\n# TYPE http_request_duration_ms histogram\n");
380              for (Map.Entry<String,Histogram> e: hists.entrySet()) sb.append(e.getValue().asProm("http_request_duration_ms", e.getKey()));
381              sb.append("\n# HELP leads_total Total accepted leads\n# TYPE leads_total counter\nleads_total ").append(leads.get()).append('\n');
382              return sb.toString();
383          }
384          private AtomicLong key(ConcurrentMap<String,AtomicLong> m,String k){ return m.computeIfAbsent(k, kk->new AtomicLong(0)); }

386          static final class Histogram {
387              private final double[] bounds; private final long[] buckets; private long count=0; private double sum=0;
388              Histogram(double[] bounds){ this.bounds=bounds; this.buckets=new long[bounds.length+1]; }
389              static Histogram defaultMs(){ return new Histogram(new double[]{50,100,250,500,1000,2500,5000}); }
390              synchronized void observe(double v){ int i=0; while(i<bounds.length && v>bounds[i]) i++; buckets[i]++; count++; sum+=v; }
391              synchronized String asProm(String metric, String label){
392                  StringBuilder sb=new StringBuilder();
393                  for(int i=0;i<buckets.length;i++){
394                      String le = (i<bounds.length) ? String.valueOf(bounds[i]) : "+Inf";
395                      sb.append(metric).append("_bucket{le=\"").append(le).append("\",name=\"").append(label).append("\"} ").append(buckets[i]).append('\n');
396                  }
397                  sb.append(metric).append("_count{name=\"").append(label).append("\"} ").append(count).append('\n');
398                  sb.append(metric).append("_sum{name=\"").append(label).append("\"} ").append(String.format(Locale.US,"%.3f", sum)).append('\n');
399                  return sb.toString();
400              }
401          }

457          private void add(String path, HttpHandler h){
458              server.createContext(path, ex -> {
459                  long t0 = System.nanoTime();
460                  String method = ex.getRequestMethod().toUpperCase(Locale.ROOT);
461                  metrics.incReq(path, method);
462                  try {
463                      ex.getResponseHeaders().add("Access-Control-Allow-Origin", "*");
464                      ex.getResponseHeaders().add("Content-Type", "application/json; charset=utf-8");
465                      if ("OPTIONS".equals(method)) { send(ex, 204, ""); return; }
466                      h.handle(ex);
467                  } catch (Throwable e) {
468                      metrics.incErr(path, method);
469                      send(ex, 500, "{\"error\":\""+ContactInfo.esc(e.getMessage())+"\"}");
470                  } finally {
471                      long ms = (System.nanoTime()-t0)/1_000_000;
472                      metrics.observe(path, ms);
473                      ex.close();
474                  }
475              });
476          }

594      static Map<String,String> parseJsonFlat(String json){
595          // sehr einfache Parser für flache JSON-Objekte: {"k":"v", "n":123}
596          Map<String,String> m=new LinkedHashMap<>(); if(json==null) return m;
597          String s=json.trim(); if(!s.startsWith("{")||!s.endsWith("}")) return m; s=s.substring(1,s.length()-1).trim(); if(s.isEmpty()) return m;
598          int i=0; while(i<s.length()){
599              // key
600              while(i<s.length() && Character.isWhitespace(s.charAt(i))) i++;
601              if(i>=s.length()||s.charAt(i)!='"') break; int j=s.indexOf('"', i+1); if(j<0) break; String key=s.substring(i+1,j);
602              i=j+1; while(i<s.length() && (s.charAt(i)==' '||s.charAt(i)==':')) i++;
603              // value (string oder number/bool)
604              String val;
605              if(i<s.length() && s.charAt(i)=='"'){ int k=s.indexOf('"', i+1); if(k<0) break; val=s.substring(i+1,k); i=k+1; }
606              else { int k=i; while(k<s.length() && s.charAt(k)!=',' ) k++; val=s.substring(i,k).trim(); i=k; }
607              m.put(key, val);
608              while(i<s.length() && (s.charAt(i)==' '||s.charAt(i)==',')) i++;
609          }
610          return m;
611      }
