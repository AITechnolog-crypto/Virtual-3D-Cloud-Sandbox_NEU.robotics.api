1    package com.june.repository;
2    
3    import com.june.model.Drone;
4    import org.springframework.data.jpa.repository.JpaRepository;
5    import org.springframework.data.jpa.repository.Query;
6    import org.springframework.stereotype.Repository;
7    import java.time.LocalDateTime;
8    import java.util.List;
9    import java.util.Optional;
10   
11   @Repository
12   public interface DroneRepository extends JpaRepository<Drone, Long> {
13       
14       Optional<Drone> findByDroneId(String droneId);
15       List<Drone> findByDroneType(String droneType);
16       List<Drone> findByCurrentStatus(String currentStatus);
17       List<Drone> findByStealthLevelGreaterThanEqual(Integer minStealthLevel);
18       List<Drone> findByBatteryLevelLessThan(Integer threshold);
19       List<Drone> findByAssignedMission(String mission);
20       
21       @Query("SELECT d FROM Drone d WHERE " +
22              "d.positionX BETWEEN :minX AND :maxX AND " +
23              "d.positionY BETWEEN :minY AND :maxY AND " +
24              "d.positionZ BETWEEN :minZ AND :maxZ")
25       List<Drone> findDronesInArea(Double minX, Double maxX,
26                                      Double minY, Double maxY,
27                                      Double minZ, Double maxZ);
28       
29       @Query("SELECT d FROM Drone d WHERE d.currentStatus = 'IDLE' " +
30              "AND d.batteryLevel > :minBattery")
31       List<Drone> findAvailableDrones(Integer minBattery);
32       
33       @Query("SELECT d.droneType, COUNT(d) FROM Drone d GROUP BY d.droneType")
34       List<Object[]> countByType();
35       
36       @Query("SELECT d.currentStatus, COUNT(d) FROM Drone d GROUP BY d.currentStatus")
37       List<Object[]> countByStatus();
38       
39       List<Drone> findTop10ByOrderByMaxSpeedDesc();
40       List<Drone> findByLastActiveAfter(LocalDateTime date);
41       boolean existsByDroneId(String droneId);
42   }
