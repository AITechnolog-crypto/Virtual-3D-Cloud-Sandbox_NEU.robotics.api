54   <script src="https://unpkg.com/three@0.156.1/build/three.min.js"></script>

56   <script>
57   (() => {
58     const log = (m)=>{ document.getElementById('log').textContent = String(m) }
59   
60     const video = document.getElementById('cam')
61     const canvas = document.getElementById('stage')
62     const gate = document.getElementById('gate')
63     const btnStart = document.getElementById('start')
64     const btnFlip = document.getElementById('flip')
65     const btnReset = document.getElementById('reset')
66     const btnCube = document.getElementById('cube')
67     const btnRing = document.getElementById('ring')
68     const btnAxis = document.getElementById('axis')
69     const selQ = document.getElementById('quality')
70   
71     let facing = 'environment'
72     let stream = null
73   
74     // Three.js Grundsetup
75     const renderer = new THREE.WebGLRenderer({ canvas, antialias:true, alpha:true, powerPreference:'high-performance' })
76     const scene = new THREE.Scene()
77     const camera = new THREE.PerspectiveCamera(60, 1, 0.01, 1000)
78     camera.position.set(0,0,2)
79   
80     // Licht
81     const amb = new THREE.AmbientLight(0xffffff, 0.8); scene.add(amb)
82     const dir = new THREE.DirectionalLight(0xffffff, 0.6); dir.position.set(1,1,1); scene.add(dir)
83   
84     // Objekte
85     const group = new THREE.Group(); scene.add(group)
86     let mesh = null
87     let axes = null
88   
89     function makeCube(){
90       if (mesh) group.remove(mesh)
91       const g = new THREE.BoxGeometry(0.5,0.5,0.5)
92       const m = new THREE.MeshStandardMaterial({ color:0x55ccff, roughness:0.4, metalness:0.2 })
93       mesh = new THREE.Mesh(g,m)
94       group.add(mesh)
95     }
96     function makeRing(){
97       if (mesh) group.remove(mesh)
98       const g = new THREE.TorusGeometry(0.5,0.18,24,64)
99       const m = new THREE.MeshStandardMaterial({ color:0xffaa33, roughness:0.2, metalness:0.6 })
100      mesh = new THREE.Mesh(g,m)
101      group.add(mesh)
102    }
103    function toggleAxes(){
104      if (axes) { scene.remove(axes); axes = null; return }
105      axes = new THREE.AxesHelper(1); scene.add(axes)
106    }
107  
108    makeCube()
109  
110    // Resize
111    function fit(){
112      const w = window.innerWidth, h = window.innerHeight
113      let scale = 1
114      if (selQ.value === 'low') scale = 0.7
115      if (selQ.value === 'high') scale = 1
116      renderer.setSize(w*scale, h*scale, false)
117      renderer.setPixelRatio(Math.min(devicePixelRatio, selQ.value==='high'?2:1.5))
118      camera.aspect = w/h
119      camera.updateProjectionMatrix()
120      canvas.style.width = '100%'; canvas.style.height = '100%'
121    }
122    window.addEventListener('resize', fit)
123    selQ.addEventListener('change', fit)
124  
125    // Device Orientation Steuerung
126    let baseQuat = new THREE.Quaternion()
127    let currentQuat = new THREE.Quaternion()
128    let havePermissionAsked = false
129  
130    function requestMotionPermissionIfNeeded(){
131      const doE = window.DeviceOrientationEvent
132      if (doE && typeof doE.requestPermission === 'function' && !havePermissionAsked){
133        havePermissionAsked = true
134        return doE.requestPermission().catch(()=>null)
135      }
136      return Promise.resolve()
137    }
138  
139    function eulerToQuat(alpha, beta, gamma){
140      const deg2rad = Math.PI/180
141      const e = new THREE.Euler(beta*deg2rad, alpha*deg2rad, -gamma*deg2rad, 'YXZ')
142      const q = new THREE.Quaternion(); q.setFromEuler(e)
143      return q
144    }
145  
146    function onOrientation(ev){
147      if (ev.absolute === null && ev.alpha === null) return
148      const q = eulerToQuat(ev.alpha||0, ev.beta||0, ev.gamma||0)
149      currentQuat.copy(q).premultiply(baseQuat)
150      group.setRotationFromQuaternion(currentQuat)
151    }
152  
153    function resetOrientation(){
154      baseQuat.copy(group.quaternion.clone().invert())
155    }
156  
157    // Kamera starten
158    async function startCamera(){
159      try {
160        if (stream) stream.getTracks().forEach(t=>t.stop())
161        stream = await navigator.mediaDevices.getUserMedia({
162          video: { facingMode: facing, width:{ideal:1920}, height:{ideal:1080} },
163          audio: false
164        })
165        video.srcObject = stream
166        await video.play()
167        fit()
168        gate.style.display = 'none'
169        log('')
170      } catch(e){
171        log('Kamera Fehler ' + e)
172      }
173    }
174  
175    // Render-Loop
176    let tPrev = performance.now()
177    function tick(){
178      const t = performance.now()
179      const dt = (t - tPrev) / 1000
180      tPrev = t
181      if (mesh){
182        mesh.rotation.y += dt * 0.5
183      }
184      renderer.render(scene, camera)
185      requestAnimationFrame(tick)
186    }
187    tick()
188    fit()
189  
190    // UI Events
191    btnStart.addEventListener('click', async ()=>{
192      await requestMotionPermissionIfNeeded()
193      window.addEventListener('deviceorientation', onOrientation)
194      startCamera()
195    })
196    btnFlip.addEventListener('click', ()=>{
197      facing = (facing === 'environment') ? 'user' : 'environment'
198      startCamera()
199      // Spiegelung Frontkamera aktiv, Rückkamera unsichtbar gespiegelt
200      video.style.transform = facing==='user' ? 'scaleX(-1)' : 'scaleX(-1)'
201    })
202    btnReset.addEventListener('click', resetOrientation)
203    btnCube.addEventListener('click', makeCube)
204    btnRing.addEventListener('click', makeRing)
205    btnAxis.addEventListener('click', toggleAxes)
206  
207    // Fallback wenn keine Motion Events
208    setTimeout(()=>{ if (!('ondeviceorientation' in window)) log('Hinweis Gerät liefert keine Orientation Events – Rotation per Touch denkbar') }, 2000)
209  
210    // Touch-Drag Rotation als Backup
211    let drag=false, lx=0, ly=0
212    canvas.addEventListener('pointerdown', e=>{ drag=true; lx=e.clientX; ly=e.clientY })
213    canvas.addEventListener('pointerup',   ()=> drag=false)
214    canvas.addEventListener('pointermove', e=>{
215      if (!drag) return
216      const dx = (e.clientX - lx)/window.innerWidth
217      const dy = (e.clientY - ly)/window.innerHeight
218      group.rotation.y += dx * Math.PI
219      group.rotation.x += dy * Math.PI
220      lx=e.clientX; ly=e.clientY
221    })
222  })();
