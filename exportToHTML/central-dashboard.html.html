78   let map;
79   document.addEventListener('DOMContentLoaded', ()=>{
80     map = L.map('map', {zoomControl:true}).setView([48.137, 11.575], 5);
81     L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png',{

83       attribution: '&copy; OpenStreetMap'
84     }).addTo(map);
85   });
86   function centerOnMe(){
87     if (navigator.geolocation) {
88       navigator.geolocation.getCurrentPosition(pos => {
89         map.setView([pos.coords.latitude, pos.coords.longitude], 14);
90         L.marker([pos.coords.latitude, pos.coords.longitude]).addTo(map);
91       });
92     }
93   }
94   // Drohnen & Roboter
95   async function loadDronesRobots(){
96     const el = document.getElementById('droneRobotList');
97     el.innerHTML = '<div class="loading">Lade...</div>';
98     try {
99       const drones = await fetch('/api/drones').then(r=>r.json());
100      const robots = await fetch('/api/robots').then(r=>r.json());
101      let html = '<table class="data-table"><tr><th>Name</th><th>Typ</th></tr>';
102      drones.forEach(d => html += `<tr><td>${d.name||'Drohne'}</td><td>Drohne</td></tr>`);
103      robots.forEach(r => html += `<tr><td>${r.name||'Roboter'}</td><td>Roboter</td></tr>`);
104      el.innerHTML = html + '</table>';
105    } catch(e) {
106      el.innerHTML = '<div class="loading">Fehler beim Laden.</div>';
107    }
108  }
109  // Finanzen & Transaktionen
110  async function loadFinance(){
111    const el = document.getElementById('financeList');
112    el.innerHTML = '<div class="loading">Lade...</div>';
113    try {
114      const wallets = await fetch('/api/wallets').then(r=>r.json());
115      const transactions = await fetch('/api/transactions').then(r=>r.json());
116      let html = '<table class="data-table"><tr><th>Wallet</th><th>Balance</th></tr>';
117      wallets.forEach(w => html += `<tr><td>${w.walletName}</td><td>${Number(w.balance).toLocaleString('de-DE',{minimumFractionDigits:2,maximumFractionDigits:2})} ${w.currency}</td></tr>`);
118      html += '</table><br><table class="data-table"><tr><th>Typ</th><th>Betrag</th><th>Kategorie</th></tr>';
119      transactions.forEach(t => {
120        const color = t.transactionType==='INCOME'? '#00ff88':'#ff0044';
121        html += `<tr><td>${t.transactionType}</td><td style="color:${color}">${Number(t.amount).toLocaleString('de-DE',{minimumFractionDigits:2,maximumFractionDigits:2})} EUR</td><td>${t.category||''}</td></tr>`;
122      });
123      el.innerHTML = html + '</table>';
124    } catch(e) {
125      el.innerHTML = '<div class="loading">Fehler beim Laden.</div>';
126    }
127  }
