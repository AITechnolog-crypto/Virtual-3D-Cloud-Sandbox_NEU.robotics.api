1    package com.june.repository;

3    import com.june.model.Blueprint;
4    import org.springframework.data.jpa.repository.JpaRepository;
5    import org.springframework.data.jpa.repository.Query;
6    import org.springframework.stereotype.Repository;
7    import java.util.List;
8    import java.util.Optional;
9    
10   @Repository
11   public interface BlueprintRepository extends JpaRepository<Blueprint, Long> {
12       
13       Optional<Blueprint> findByNameAndVersion(String name, String version);
14       List<Blueprint> findByCategory(String category);
15       List<Blueprint> findByStatus(String status);
16       List<Blueprint> findByCreatedBy(String createdBy);
17       
18       @Query("SELECT b FROM Blueprint b WHERE b.name = :name " +
19              "ORDER BY b.version DESC LIMIT 1")
20       Optional<Blueprint> findLatestVersion(String name);
21       
22       List<Blueprint> findByNameOrderByVersionDesc(String name);
23       
24       @Query("SELECT b FROM Blueprint b WHERE " +
25              "LOWER(b.name) LIKE LOWER(CONCAT('%', :keyword, '%')) OR " +
26              "LOWER(b.description) LIKE LOWER(CONCAT('%', :keyword, '%'))")
27       List<Blueprint> searchBlueprints(String keyword);
28       
29       @Query("SELECT b.category, COUNT(b) FROM Blueprint b GROUP BY b.category")
30       List<Object[]> countByCategory();
31       
32       List<Blueprint> findByStatusIn(List<String> statuses);
33   }
