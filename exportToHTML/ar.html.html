28   <script src="https://unpkg.com/three@0.156.1/build/three.min.js"></script>
29   <script>
30   (function(){
31     const info  = document.getElementById('info');
32     const video = document.getElementById('cam');
33     const canvas= document.getElementById('stage');
34     const gate  = document.getElementById('gate');
35     const btnStart=document.getElementById('start');
36     const btnFlip =document.getElementById('flip');
37     const btnReset=document.getElementById('reset');
38     const btnObj  =document.getElementById('obj');
39   
40     let facing = 'environment';
41     let stream = null;
42     let target = null; // {lat,lng}
43   
44     const renderer = new THREE.WebGLRenderer({canvas, antialias:true, alpha:true, powerPreference:'high-performance'});
45     const scene = new THREE.Scene();
46     const camera = new THREE.PerspectiveCamera(60, 1, 0.01, 1000);
47     camera.position.set(0,0,2);
48     const amb = new THREE.AmbientLight(0xffffff, 0.8); scene.add(amb);
49     const dir = new THREE.DirectionalLight(0xffffff, 0.7); dir.position.set(1,1,1); scene.add(dir);
50     scene.add(new THREE.HemisphereLight(0xffffff, 0x223344, 0.3));
51   
52     const group = new THREE.Group(); scene.add(group);
53     let mesh = null;
54     function makeCube(){ if (mesh) group.remove(mesh); mesh = new THREE.Mesh(new THREE.BoxGeometry(0.5,0.5,0.5), new THREE.MeshStandardMaterial({color:0x55ccff, roughness:0.4, metalness:0.2})); group.add(mesh); }
55     function makeRing(){ if (mesh) group.remove(mesh); mesh = new THREE.Mesh(new THREE.TorusGeometry(0.5,0.18,24,64), new THREE.MeshStandardMaterial({color:0xffaa33, roughness:0.2, metalness:0.6})); group.add(mesh); }
56     makeCube();
57   
58     function fit(){ const w = innerWidth, h = innerHeight; renderer.setSize(w, h, false); renderer.setPixelRatio(Math.min(devicePixelRatio, 2)); camera.aspect = w/h; camera.updateProjectionMatrix(); }
59     addEventListener('resize', fit); fit();
60   
61     // Device Orientation
62     let baseQuat = new THREE.Quaternion();
63     let currentQuat = new THREE.Quaternion();
64     function eulerToQuat(a,b,g){ const d = Math.PI/180; const e = new THREE.Euler((b||0)*d,(a||0)*d,-(g||0)*d,'YXZ'); const q = new THREE.Quaternion(); q.setFromEuler(e); return q; }
65     function onOrientation(ev){ const q = eulerToQuat(ev.alpha, ev.beta, ev.gamma); currentQuat.slerp(q, 0.15).premultiply(baseQuat); group.setRotationFromQuaternion(currentQuat); }
66     function resetOrientation(){ baseQuat.identity(); }
67   
68     // Kamera
69     async function startCam(){
70       if (stream) stream.getTracks().forEach(t=>t.stop());
71       stream = await navigator.mediaDevices.getUserMedia({ video:{facingMode:facing,width:{ideal:1280},height:{ideal:720},frameRate:{ideal:30,max:30}}, audio:false });
72       video.srcObject = stream; await video.play();
73       video.style.transform = facing==='user' ? 'scaleX(-1)' : 'none';
74       gate.style.display = 'none';
75     }
76   
77     // Geolokation anfordern und an Hub melden
78     if ('geolocation' in navigator){
79       navigator.geolocation.watchPosition(p=>{
80         const lat = p.coords.latitude, lng = p.coords.longitude, acc = p.coords.accuracy;
81         try { window.parent.postMessage({type:'ar:position', lat, lng, acc}, '*'); } catch(_){ }
82         info.textContent = 'AR Position ' + lat.toFixed(5) + ' ' + lng.toFixed(5);
83       }, e=>{}, {enableHighAccuracy:true, maximumAge:5000, timeout:10000});
84     }
85   
86     // Ziel aus Hub empfangen
87     window.addEventListener('message', ev=>{
88       const m = ev.data || {};
89       if (m.type === 'hub:setTarget'){
90         target = {lat:m.lat, lng:m.lng};
91         info.textContent = 'Ziel gesetzt ' + m.lat.toFixed(5) + ' ' + m.lng.toFixed(5);
92       }
93     });
94   
95     // Render
96     let tPrev = performance.now();
97     function tick(){ const t = performance.now(), dt = (t - tPrev) / 1000; tPrev = t; if (mesh){ mesh.rotation.y += dt * 0.5; } renderer.render(scene, camera); requestAnimationFrame(tick); }
98     tick();
99   
100    // UI
101    btnStart.onclick = async ()=>{
102      try{
103        if (window.DeviceOrientationEvent && typeof DeviceOrientationEvent.requestPermission==='function'){
104          await DeviceOrientationEvent.requestPermission().catch(()=>null);
105        }
106        addEventListener('deviceorientation', onOrientation);
107        await startCam();
108        try { window.parent.postMessage({type:'ar:status', msg:'gestartet'}, '*'); } catch(_){ }
109      }catch(e){
110        try { window.parent.postMessage({type:'ar:status', msg:'fehler '+e}, '*'); } catch(_){}
111      }
112    };
113    btnFlip.onclick  = ()=>{ facing = (facing==='environment') ? 'user' : 'environment'; startCam(); };
114    btnReset.onclick = ()=> resetOrientation();
115    btnObj.onclick   = ()=>{ if (mesh && mesh.geometry.type==='BoxGeometry') makeRing(); else makeCube(); };
116  })();
117  </script>
