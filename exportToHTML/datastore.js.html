1    // datastore.js — schlanker Speicher mit drei Treibern (Local | File | Http)
2    // Minimal-invasive Einbindung: Polyfill für bus, Default-Store, Export/Import & Usage-Anzeige
3    (function(){
4      'use strict';
5    
6      // ---------- Utils
7      function safeJSONParse(s, fallback){ try{ return JSON.parse(s); }catch(_){ return fallback; } }
8      function lsGet(k, d){ try{ const v = localStorage.getItem(k); return v==null? d : v; }catch(_){ return d; } }
9      function lsSet(k, v){ try{ localStorage.setItem(k, v); }catch(_){ } }
10   
11     // ---------- Store Factory
12     const Store = (function(){
13       function load(key){ return safeJSONParse(lsGet(key, '[]'), []); }
14       function save(key, arr){ lsSet(key, JSON.stringify(arr)); }
15   
16       class Local {
17         constructor({ key='nur.events', budgetKB=256 }={}){
18           this.key = key;
19           this.cap = budgetKB * 1024;
20           this.buf = load(this.key);
21         }
22         append(ev){ this.buf.push(ev); this.gc(); save(this.key, this.buf); }
23         list(){ return this.buf.slice(); }
24         clear(){ this.buf = []; save(this.key, this.buf); }
25         usage(){ const bytes = (lsGet(this.key, '')||'').length; return { bytes, cap: this.cap }; }
26         gc(){ let s = JSON.stringify(this.buf); while (s.length > this.cap){ this.buf.shift(); s = JSON.stringify(this.buf); } }
27       }
28   
29       class File { // Export/Import als JSON-Datei
30         constructor(opts){ this.local = new Local(opts||{}); }
31         append(e){ this.local.append(e); }
32         list(){ return this.local.list(); }
33         clear(){ this.local.clear(); }
34         usage(){ return this.local.usage(); }
35         export(){ try{ const a=document.createElement('a'); a.href=URL.createObjectURL(new Blob([JSON.stringify(this.local.list())],{type:'application/json'})); a.download='events.json'; a.click(); setTimeout(()=>URL.revokeObjectURL(a.href), 0); }catch(_){}}
36         import(text){ const arr = safeJSONParse(text, null); if (Array.isArray(arr)){ this.local.clear(); arr.forEach(e=> this.local.append(e)); }
37         }
38       }
39   
40       class Http { // versucht POST, fällt bei Fehlern auf Local zurück
41         constructor({ base=location.origin+'/api', key='nur.events', budgetKB=256 }={}){ this.base = base; this.local = new Local({ key, budgetKB }); }
42         async append(ev){ this.local.append(ev); try{ await fetch(this.base + '/events', { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify(ev) }); }catch(_){ /* fallback already appended */ } }
43         async list(){ try{ const r = await fetch(this.base + '/events', { cache:'no-store' }); if (r.ok) return await r.json(); }catch(_){ } return this.local.list(); }
44         clear(){ this.local.clear(); }
45         async usage(){ return this.local.usage(); }
46       }
47   
48       function make(opts){ const d = (opts && opts.driver) || 'local';
49         if (d === 'http') return new Http(opts||{});
50         if (d === 'file') return new File(opts||{});
51         return new Local(opts||{});
52       }
53   
54       return { make };
55     })();

57     // Expose
58     try{ window.Store = Store; }catch(_){ }

60     // ---------- Mini-Bus Polyfill
61     (function(){
62       if (window.bus && typeof window.bus.on === 'function') return;
63       const listeners = Object.create(null);
64       window.bus = {
65         on(type, cb){ if(!listeners[type]) listeners[type]=[]; listeners[type].push(cb); return ()=> this.off(type, cb); },
66         off(type, cb){ const arr=listeners[type]||[]; const i=arr.indexOf(cb); if(i>=0) arr.splice(i,1); },
67         emit(type, payload){ const arr=(listeners[type]||[]).slice(); arr.forEach(fn=>{ try{ fn({ type, payload }); }catch(_){ } }); }
68       };
69     })();
70   
71     // ---------- Default-Store initialisieren (falls nicht vorhanden)
72     const DEFAULT_KEY = 'nur.events';
73     const DEFAULT_BUDGET_KB = Number(lsGet('nur.store.budgetKB','256')) || 256;
74     if (!window.__eventStore){
75       try{ window.__eventStore = Store.make({ driver: 'local', key: DEFAULT_KEY, budgetKB: DEFAULT_BUDGET_KB }); }catch(_){ }
76     }
77     if (!window.store){ try{ window.store = window.__eventStore; }catch(_){ } }
78   
79     // ---------- Event-Wiring: hal / learn / vision
80     try{
81       const store = window.__eventStore;
82       const append = (type, payload)=>{ try{ store.append({ t: Date.now(), type, data: payload }); }catch(_){ } };
83       window.bus.on('hal:sensors', m=> append('hal',   (m && m.payload)!=null ? m.payload : m));
84       window.bus.on('learn:stats', m=> append('learn', (m && m.payload)!=null ? m.payload : m));
85       window.bus.on('vision:blob', m=> append('vision',(m && m.payload)!=null ? m.payload : m));
86     }catch(_){ }
87   
88     // ---------- Bridge: NurTrio Testbench postMessage -> bus.emit
89     try{
90       window.addEventListener('message', (ev)=>{
91         const d = ev && ev.data; if (!d || d.ns !== 'NurTrioV1') return;
92         if (typeof d.type === 'string'){ window.bus.emit(d.type, { source:'postMessage', payload: d }); }
93       });
94     }catch(_){ }

96     // ---------- Export/Import Helper (Buttons oder Mini-Toolbox)
97     function ensureButtons(){
98       function bind(){
99         try{
100          const store = window.__eventStore;
101          const btnEx = document.getElementById('btnExport');
102          const btnIm = document.getElementById('btnImport');
103          if (btnEx && !btnEx.__wired){ btnEx.__wired=true; btnEx.addEventListener('click', ()=>{ if (store.export) store.export(); else { const a=document.createElement('a'); a.href=URL.createObjectURL(new Blob([JSON.stringify(store.list())],{type:'application/json'})); a.download='events.json'; a.click(); } }); }
104          if (btnIm && !btnIm.__wired){ btnIm.__wired=true; btnIm.addEventListener('click', ()=>{ const i=document.createElement('input'); i.type='file'; i.accept='application/json'; i.onchange=()=>{ const f=i.files && i.files[0]; if(!f) return; const r=new FileReader(); r.onload=()=>{ try{ if (store.import) store.import(r.result); else lsSet(DEFAULT_KEY, r.result); location.reload(); }catch(_){ } }; r.readAsText(f); }; i.click(); }); }
105          return !!(btnEx||btnIm);
106        }catch(_){ return false; }
107      }
108      if (bind()) return; // Buttons existieren
109      // Falls Buttons fehlen: kleine, dezente Toolbox einblenden (unten rechts)
110      const host = document.createElement('div');
111      host.id = 'storeToolbox';
112      host.style.cssText = 'position:fixed;right:12px;bottom:12px;z-index:99998;display:flex;gap:6px;';
113      const b1 = document.createElement('button'); b1.textContent='Export'; b1.id='btnExport';
114      const b2 = document.createElement('button'); b2.textContent='Import'; b2.id='btnImport';
115      [b1,b2].forEach(b=>{ b.style.cssText='background:rgba(255,255,255,.06);color:#e8f2ff;border:1px solid rgba(255,255,255,.15);padding:6px 10px;border-radius:10px;cursor:pointer' });
116      host.appendChild(b1); host.appendChild(b2);
117      document.addEventListener('DOMContentLoaded', ()=>{ try{ document.body.appendChild(host); bind(); }catch(_){ } });
118    }
119    ensureButtons();

121    // ---------- Usage Monitor + DevBadge-Integration
122    (function(){
123      let storeUsed = '';
124      async function update(){
125        try{ const u = await (window.__eventStore && window.__eventStore.usage ? window.__eventStore.usage() : {bytes:0,cap:1});
126          storeUsed = (u && u.cap) ? ((u.bytes/1024).toFixed(1) + '/' + (u.cap/1024).toFixed(0) + ' KB') : '';
127          window.__storeUsage = storeUsed;
128          // Update Dev Badge if present
129          const badge = document.getElementById('devBadge');
130          if (badge){
131            let tag = badge.querySelector('#storeUsageTag');
132            if (!tag){ tag = document.createElement('div'); tag.id='storeUsageTag'; tag.className='small'; tag.style.opacity='0.85'; tag.style.marginTop='2px'; badge.appendChild(tag); }
133            tag.textContent = '🗃️ store ' + storeUsed;
134          }
135        }catch(_){ }
136      }
137      setInterval(update, 1500); // alle 1.5s wie gefordert
138      // erste Initialisierung nach DOM bereit
139      if (document.readyState === 'loading'){ document.addEventListener('DOMContentLoaded', update); } else { update(); }
140    })();
