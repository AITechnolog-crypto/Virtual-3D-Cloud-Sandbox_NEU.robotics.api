214          double balance=0; final Path ledger = DATA.resolve("wallet.csv");
215          synchronized void credit(double a,String why){ if(a>0){ balance+=a; log("CREDIT",a,why);} }
216          synchronized boolean debit(double a,String why){ if(a<=0||a>balance) return false; balance-=a; log("DEBIT",a,why); return true; }
217          private void log(String t,double a,String m){ try{ Files.writeString(ledger, new Date()+";"+t+";"+String.format(Locale.US,"%.2f",a)+";"+m+"\n", StandardCharsets.UTF_8, StandardOpenOption.CREATE, StandardOpenOption.APPEND);}catch(Exception ignored){} }

219      enum EquityTier { NEEDY, STANDARD, WEALTHY }
220      static final class SocialEquity {
221          EquityTier classify(double incomeIdx){ if(incomeIdx>=0.75) return EquityTier.WEALTHY; if(incomeIdx<=0.35) return EquityTier.NEEDY; return EquityTier.STANDARD; }
222          double buyerPrice(EquityTier t,double p){ return switch(t){ case NEEDY->Math.max(0,p*0.9); case WEALTHY->p*1.2; default->p; }; }
223          double waterAidPrice(EquityTier t,double liters){ double base=liters*0.02; return (t==EquityTier.NEEDY)?0 : (t==EquityTier.WEALTHY? base*1.6 : base); }
224      }
225      static final class Marketplace {
226          static final class Offer { final String name; double basePerKg; int stockKg; Offer(String n,double b,int s){name=n;basePerKg=b;stockKg=s;} }
227          final List<Offer> offers=new ArrayList<>(); final Random r=new Random(); final Map<String,Integer> demand=new HashMap<>();
228          void add(Offer o){ offers.add(o); }
229          double demandIndex(String name){ int d=demand.getOrDefault(name,0)+1; demand.put(name,d); return 1.0+Math.min(1.2, d*0.04); }
230          double sell(String product, double kg, SocialEquity eq, Wallet wallet){
231              Offer o = offers.stream().filter(x->x.name.equals(product)&&x.stockKg>0).findFirst().orElse(null); if(o==null) return 0;
232              int sell=(int)Math.min(kg,o.stockKg); if(sell<=0) return 0;
233              double incomeIdx = 0.3 + r.nextDouble()*0.7; EquityTier tier = eq.classify(incomeIdx);
234              double dyn = o.basePerKg * demandIndex(o.name);
235              double price = eq.buyerPrice(tier, dyn); price = Math.max(o.basePerKg*0.7, Math.min(price, o.basePerKg*1.9));
236              double revenue = sell*price; o.stockKg-=sell; wallet.credit(revenue, "SALE "+product+" "+sell+"kg @"+String.format(Locale.US,"%.2f",price)+"/kg tier="+tier);
237              return revenue;
238          }
239      }
240      static final class AidRouter {
241          final SocialEquity eq; final FarmStore store; final Wallet wallet;
242          AidRouter(SocialEquity e, FarmStore s, Wallet w){ this.eq=e; this.store=s; this.wallet=w; }
243          void waterAid(double liters, double incomeIdx){
244              EquityTier t = eq.classify(incomeIdx);
245              if(store.waterLiters < liters){ System.out.println("[AID] nicht genug Wasser"); return; }
246              store.waterLiters -= liters;
247              double charge = eq.waterAidPrice(t, liters);
248              if(charge>0) wallet.credit(charge, "AID water "+(int)liters+"L tier="+t);
249              System.out.println("[AID] "+(int)liters+"L geliefert tier="+t+" charge="+String.format(Locale.US,"%.2f",charge));
250          }

273      public static void main(String[] args) {
274          ensure();
275          Map<String,String> cli = parseArgs(args);
276          String mode = cli.getOrDefault("mode","run");
277  
278          // --- Biometrie ---
279          Biometry bio = new Biometry();
280          if("enroll".equalsIgnoreCase(mode)){
281              if(args.length<2){ System.out.println("Usage: java GreenBoots_BioModules_Pro enroll <SHA256_HEX>"); return; }
282              bio.enroll(args[1].trim().toLowerCase());
283              System.out.println("Enrolled marker: "+args[1]); return;
284          }
285          if(!bio.load()){ System.out.println("Kein Marker. Bitte zuerst: java GreenBoots_BioModules_Pro enroll <SHA256_HEX>"); return; }
286          if(args.length<2 || !bio.check(args[1].trim().toLowerCase())){ System.out.println("Biometrie: Zugriff verweigert."); return; }
287  
288          // --- CLI ---
289          String zoneArg = cli.get("zone");
290          long units = parseLong(cli.getOrDefault("units","10000000000"), 10_000_000_000L);
291  
292          // --- Core/Context ---
293          EnergyManager energy = new EnergyManager();
294          SwarmField swarm = new SwarmField(units);
295          CoolingUnit cooling = new CoolingUnit();
296          Audit audit = new Audit();
297          Metrics metrics = new Metrics();
298          EdgePrivacy edge = new EdgePrivacy();
299          ZoneResolver zones = new ZoneResolver();
300          RecoveryPolicy policy = new RecoveryPolicy();
301          GreenHarvesters harvest = new GreenHarvesters(new Random(7));
302  
303          Wallet wallet = new Wallet();
304          Marketplace market = new Marketplace();
305          market.add(new Marketplace.Offer("Wheat", 0.85, 1000));
306          market.add(new Marketplace.Offer("Corn",  0.65, 800));
307          SocialEquity equity = new SocialEquity();
308          FarmStore store = new FarmStore();
309          AidRouter aid = new AidRouter(equity, store, wallet);
310  
311          Context ctx = new Context(energy, swarm, cooling, audit, metrics, edge, zones, policy, harvest,
312                                    wallet, market, equity, aid, store);
313  
314          // --- Modules ---
315          ExtensionHub hub = new ExtensionHub(ctx);
316          hub.register(new PollinatorGuardianModule());
317          hub.register(new ForestGuardianModule());
318          hub.register(new BirdAndBatSafeModule());
319          hub.register(new BlueCorridorModule());
320          hub.register(new TurtleAndGhostNetModule());
321          hub.register(new AmphibianHelperModule());
322          hub.register(new PalletSentinelModule());
323          hub.register(new SmartBinsModule());
324          hub.register(new CropGuardianModule());
325          hub.register(new ReefMicrolabModule());
326  
327          // --- Deploy ---
328          swarm.deploy();
329          audit.log("system", "GreenBoots PRO bereit: " + NumberFormat.getInstance(Locale.US).format(units) + " Units");
330          wallet.credit(120,"bootstrap");
331  
332          // --- Route ---
333          List<Position> route = (zoneArg==null) ? List.of(
334              new Position(48.135,11.582),  // CITY
335              new Position(50.000, 8.800),  // FARMLAND
336              new Position(47.500,13.000),  // FOREST
337              new Position(53.550, 9.993),  // COAST
338              new Position(47.950,12.300)   // WETLAND
339          ) : List.of(exampleFor(zoneArg));
340  
341          int tick=0;
342          for(Position p : route){
343              tick++;
344              String zone = zones.zoneTagFor(p);
345              if(!policy.allowStart(zone)){ audit.log("policy","skip zone="+zone+" @"+p); continue; }
346  
347              swarm.ensurePlatform();
348  
349              // Passives Harvest
350              double kJ=0; kJ+=harvest.harvestPlantMicrobialKJ(zone); kJ+=harvest.harvestSolarKJ(zone); energy.add(kJ);
351              audit.log("energy","harvest +"+round(kJ)+"kJ ("+zone+")");
352  
353              // Bio-Module
354              hub.autorun(p, zone);
355  
356              // Cooling
357              cooling.cool();
358  
359              // Commerce: Wenn genug Korn, verkaufe Batch
360              if(store.grainKg>650){
361                  double sellKg = Math.min(300, store.grainKg-500);
362                  store.grainKg -= sellKg;
363                  double rev = market.sell("Wheat", sellKg, equity, wallet);
364                  store.reinvest(rev);
365              }
366  
367              // Humanitäre Wasserhilfe: alle 4 Ticks ein bedürftiger Haushalt (Demo)
368              if(tick%4==0){
369                  double incomeIdx=0.28; aid.waterAid(160, incomeIdx);
370              }
371  
372              // Snapshot
373              snapshot(ctx, zone, tick);
374  
375              audit.log("status", "tick="+tick+" zone="+zone+" E="+round(energy.total())+"kJ T="+round(cooling.temp())+"C store(grain="+round(store.grainKg)+"kg, water="+round(store.waterLiters)+"L) wallet="+String.format(Locale.US,"%.2f",wallet.balance));
376  
377              // kleine Pause (Demo)
378              try{ Thread.sleep(350);}catch(InterruptedException ignored){}
379          }
380  
381          System.out.println("\n=== KPIs ===");
382          metrics.print();
383          System.out.println("\n=== Audit (Tail) ===");
384          for(String r: audit.tail()) System.out.println("• "+r);
385          System.out.println("\n[END] Gesamtenergie: "+round(energy.total())+" kJ | Wallet: "+String.format(Locale.US,"%.2f",wallet.balance));
386      }
