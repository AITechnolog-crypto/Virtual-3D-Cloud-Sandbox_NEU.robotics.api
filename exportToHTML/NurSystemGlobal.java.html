119  class SwarmEnergyBridge {
120      private static final double KWH_TO_KJ=3600.0;
121      private final IEnergyReservoir res;
122      public SwarmEnergyBridge(IEnergyReservoir r){res=r;}
123      public void distributeTo(List<SwarmRobot> sample,double kWhPerBot){
124          if(sample.isEmpty()) return;
125          double need=kWhPerBot*sample.size();
126          double take=res.takeEnergy(need);
127          double perBotKWh=take/sample.size();
128          double perBotKJ=perBotKWh*KWH_TO_KJ;
129          for(SwarmRobot bot:sample) bot.chargeKJ(perBotKJ);
130          System.out.println("Rückspeisung: "+perBotKWh+" kWh/Bot ("+perBotKJ+" kJ)");
131      }
132  }

156  class SwarmRobot {
157      private SolarPanel solar=new SolarPanel();
158      private PhotocatalyticConverter conv=new PhotocatalyticConverter();
159      private SatelliteCommunicator comm=new SatelliteCommunicator();
160      private MagneticFieldGenerator field=new MagneticFieldGenerator();
161      private PlasmaShield shield=new PlasmaShield();
162      private Location target; private double energy=0; private boolean useMagneticField=true;
163  
164      public SwarmRobot(Location l){ target=l; }
165      public void chargeKJ(double kj){ energy+=kj; System.out.println("Bot@"+target+" aufgeladen: "+energy+" kJ"); }
166      public void operate(){
167          if (!isReady()) {
168              handleError("Nicht genügend Energie zum Starten.");
169              return;
170          }
171          // Neue Betriebslogik gemäß Vorgabe
172          energy += solar.absorbSunlight();
173          conv.convertCO2(energy/10.0); // Verbraucht Energie (Verbuchung durch Aufrufer)
174          energy -= 5; // Beispielhafter Verbrauch für Konvertierung
175          comm.receiveCommands();
176          if (useMagneticField) {
177              field.generateField();
178              energy -= 2; // Feld‑Overhead
179          }
180          System.out.println("SwarmRobot ist in Betrieb. Aktueller Energielevel: " + energy);
181      }
182      private boolean isReady(){
183          // Minimaler Ready‑Check; kann später an echte Schwellen angepasst werden
184          return energy >= 0; // 0 erlaubt Start, negative Energie würde blockieren
185      }
186      private void handleError(String msg){
187          System.out.println(msg);
188      }
189      public void respondToThreat(String t){
190          switch(t){
191              case "Rakete": energy-=shield.activateKJ(t); break;
192              case "Laser": energy-=shield.activateKJ(t); energy+=conv.convertCO2KJ(); break;
193              case "EMP": comm.sendSync(); break;
194              case "Atombombe": energy-=shield.activateKJ(t)+field.neutralizeGravityKJ(); break;
195              case "Virus": System.out.println("Virus neutralisiert."); break;
196              case "Negative KI": System.out.println("Feindliche KI blockiert."); break;
197              default: System.out.println("Unbekannte Bedrohung: "+t); break;
198          }
199      }
200  }

272  class ApiConnector {
273      private static final String API_URL="https://api.openai.com/v1/chat/completions";
274      private final String apiKey;
275      public ApiConnector(String key){
276          String k = key;
277          if (k == null || k.isBlank()) {
278              try { k = System.getenv("OPENAI_API_KEY"); } catch (Exception ignored) {}
279              if (k == null || k.isBlank()) {
280                  try { k = System.getProperty("openai.api.key", ""); } catch (Exception ignored) {}
281              }
282          }
283          this.apiKey = (k == null) ? "" : k;
284      }
285      public String query(String prompt){
286          try{
287              if (apiKey == null || apiKey.isBlank()) return "API Fehler: OPENAI_API_KEY fehlt.";
288              HttpClient client=HttpClient.newHttpClient();
289              String body="{\"model\":\"gpt-4o-mini\",\"messages\":[{\"role\":\"user\",\"content\":\""+prompt+"\"}]}";
290              HttpRequest req=HttpRequest.newBuilder()
291                  .uri(URI.create(API_URL))
292                  .header("Content-Type","application/json")
293                  .header("Authorization","Bearer "+apiKey)
294                  .POST(HttpRequest.BodyPublishers.ofString(body,StandardCharsets.UTF_8))
295                  .build();
296              HttpResponse<String> res=client.send(req,HttpResponse.BodyHandlers.ofString());
297              return res.body();
298          }catch(Exception e){ return "API Fehler: "+e.getMessage(); }
299      }
300  }
