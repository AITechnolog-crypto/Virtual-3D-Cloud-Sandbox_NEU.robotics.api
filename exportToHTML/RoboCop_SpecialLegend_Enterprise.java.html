120      static final class EvidenceLocker {
121          final List<String> lines = new ArrayList<>();
122          void add(String line){ lines.add(line); }
123          File exportPackage(String name){
124              try {
125                  // Hash-Kette
126                  List<String> out = new ArrayList<>();
127                  String prev = "GENESIS";
128                  for (String l : lines) {
129                      String h = sha256(prev + "|" + l);
130                      out.add(h + "  " + l);
131                      prev = h;
132                  }
133                  out.add("CHAIN_HEAD="+prev);
134                  File f = new File(name);
135                  try (Writer w = new OutputStreamWriter(new FileOutputStream(f), StandardCharsets.UTF_8)){
136                      w.write(String.join("\n", out));
137                  }
138                  return f;
139              } catch (Exception e) {
140                  System.out.println("Evidence export fehlgeschlagen: "+e.getMessage());
141                  return new File(name);
142              }
143          }
144          private static String sha256(String s) throws Exception{
145              MessageDigest md = MessageDigest.getInstance("SHA-256");
146              byte[] d = md.digest(s.getBytes(StandardCharsets.UTF_8));
147              StringBuilder sb = new StringBuilder();
148              for (byte b : d) sb.append(String.format("%02x", b));
149              return sb.toString();
150          }
151      }

318              double litersDelivered = 0, charge=0;

392      enum PaymentKind { IBAN_EUR, BITCOIN, ETHEREUM, UNKNOWN }
393      static final class PaymentDetector {
394          private static final String IBAN_RE = "^[A-Z]{2}[0-9]{2}[A-Z0-9]{11,30}$";
395          private static final String ETH_RE  = "^0x[0-9a-fA-F]{40}$";
396          private static final String BTC_RE  = "^(bc1|[13])[a-zA-HJ-NP-Z0-9]{25,90}$";
397          static PaymentKind detect(String s){
398              if (s==null) return PaymentKind.UNKNOWN;
399              String x = s.replace(" ", "");
400              if (x.matches(ETH_RE)) return PaymentKind.ETHEREUM;
401              if (x.matches(BTC_RE)) return PaymentKind.BITCOIN;
402              if (x.matches(IBAN_RE)) return PaymentKind.IBAN_EUR;
403              return PaymentKind.UNKNOWN;
404          }
405      }
406      static final class QrPayloadFactory {
407          static String bitcoinUri(String address, Double amountBtc){
408              if (amountBtc == null) return "bitcoin:" + address;
409              NumberFormat nf = NumberFormat.getInstance(Locale.US); nf.setGroupingUsed(false); nf.setMaximumFractionDigits(8);

483          void route(double revenue, double ops){
484              revenue = Math.max(0,revenue); ops=Math.max(0,ops);
485              System.out.println(String.format(Locale.US,"[Route] Einnahmen=%.2f Kosten=%.2f Reserve=%.2f",revenue,ops,reserve));
486              double margin = revenue - ops;
487              if (margin < 0){
488                  double need = -margin;
489                  if (reserve >= need){ reserve -= need; System.out.println("[Route] Defizit "+euro(need)+" aus Reserve gedeckt → "+euro(reserve)); margin=0; }
490                  else { System.out.println("[Route] Defizit "+euro(need)+" NICHT gedeckt. Keine Ausschüttung."); return; }
491              }
492              double fill = Math.min(Math.max(0, p.reserveTarget - reserve), margin);
493              reserve += fill; double rest = margin - fill;
494              if (fill>0) System.out.println("[Route] Reserve +"+euro(fill)+" → "+euro(reserve));
495              if (rest<=0){ System.out.println("[Route] Kein Überschuss."); return; }
496              double toWallet = rest*p.wWallet, toTech = rest*p.wTech, toAid = rest*p.wAid, toCloud = rest*p.wCloud;
497              if (toWallet>0){ wallet.gutschreiben(toWallet); silos.wallet(toWallet); }
498              if (toCloud>0){ cloud.feed(toCloud); silos.cloud(toCloud); }
499              if (toTech>0) System.out.println("[R&D] Budget +"+euro(toTech));
500              if (toAid>0)  System.out.println("[Aid] Budget +"+euro(toAid));
501          }
502      }
