19   <script>
20   const NS='NurTrioV1';
21   let t0=performance.now(), x=0,y=0, th=0, n=0;
22   function send(){
23     const t=performance.now(); const dt=(t-t0)/1000; t0=t; n++;
24     const v=0.5, w=0.2; th+=w*dt; x+=Math.cos(th)*v*dt; y+=Math.sin(th)*v*dt;
25     const s={ ns:NS, type:'hal:sensors', pose:{x,y,theta:th}, imu:{yaw:th,pitch:0,roll:0}, prox:{front: 1.2+0.8*Math.sin(t*0.001)}, time:Date.now() };
26     parent.postMessage(s, '*');
27     if (n%3===0){ parent.postMessage({ ns:NS, type:'learn:stats', loss:Math.random()*0.1 }, '*'); }
28     if (n%5===0){ parent.postMessage({ ns:NS, type:'vision:blob', blobs: Math.floor(Math.random()*4) }, '*'); }
29     const el=document.getElementById('st'); if (el){ el.textContent = `t=${new Date().toLocaleTimeString()}  x=${x.toFixed(2)} y=${y.toFixed(2)} th=${th.toFixed(2)}  msgs=${n}`; }
30   }
31   let iv=null;
32   function start(){ if(iv) clearInterval(iv); iv=setInterval(send, 200); }
33   function stop(){ if(iv) clearInterval(iv); iv=null; }
34   window.addEventListener('visibilitychange',()=>{ if(document.hidden) stop(); else start(); });
35   start();
36   </script>
