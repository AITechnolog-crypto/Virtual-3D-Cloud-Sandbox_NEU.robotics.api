126      static final class QrPngWriter {
127          @SuppressWarnings("unchecked")
128          static boolean writePngIfZXing(String payload, File out, int size){
129              try{
130                  Class<?> bfClass=Class.forName("com.google.zxing.BarcodeFormat");
131                  Object qrFormat=Enum.valueOf((Class<Enum>)bfClass.asSubclass(Enum.class), "QR_CODE");
132                  Class<?> writerCls=Class.forName("com.google.zxing.qrcode.QRCodeWriter");
133                  Object writer=writerCls.getDeclaredConstructor().newInstance();
134                  Method encode=writerCls.getMethod("encode", String.class, bfClass, int.class, int.class, Map.class);
135                  Object bitMatrix=encode.invoke(writer, payload, qrFormat, size, size, null);
136                  Class<?> bmClass=Class.forName("com.google.zxing.common.BitMatrix");
137                  int w=(int)bmClass.getMethod("getWidth").invoke(bitMatrix);
138                  int h=(int)bmClass.getMethod("getHeight").invoke(bitMatrix);
139                  BufferedImage img=new BufferedImage(w,h,BufferedImage.TYPE_INT_RGB);
140                  Method get=bmClass.getMethod("get", int.class, int.class);
141                  for(int y=0;y<h;y++) for(int x=0;x<w;x++){ boolean on=(boolean)get.invoke(bitMatrix,x,y); img.setRGB(x,y, on?0x000000:0xFFFFFF);} 
142                  ImageIO.write(img, "png", out);
143                  System.out.println("[QR] PNG → "+out.getAbsolutePath());
144                  return true;
145              }catch(ClassNotFoundException e){
146                  System.out.println("[QR] ZXing nicht im Classpath – zeige Payload als Text.");
147                  return false;
148              }catch(Throwable t){
149                  System.out.println("[QR] Fehler: "+t.getMessage());
150                  return false;
151              }
152          }

253          /** Einnahmen → Kosten → Reserve → Quoten; nie negativ. */
254          void route(double revenue, double opCost){
255              revenue=Math.max(0,revenue); opCost=Math.max(0,opCost);
256              System.out.println(String.format(Locale.US,"[Route] € in=%.2f  op=%.2f  reserve=%.2f", revenue, opCost, reserve));
257              double margin = revenue - opCost;
258              if(margin < 0){
259                  double need=-margin;
260                  if(reserve>=need){ reserve-=need; System.out.println(String.format(Locale.US,"[Route] Defizit %.2f aus Reserve → %.2f",need,reserve)); margin=0; }
261                  else { System.out.println(String.format(Locale.US,"[Route] Defizit %.2f nicht gedeckt (Reserve %.2f). Keine Ausschüttung.",need,reserve)); return; }
262              }
263              double needRes=Math.max(0, policy.reserveTarget - reserve);
264              double toRes=Math.min(needRes, margin);
265              reserve += toRes; double rest = Math.max(0, margin - toRes);
266              if(toRes>0) System.out.println(String.format(Locale.US,"[Route] Reserve +%.2f → %.2f", toRes, reserve));
267              if(rest<=0){ System.out.println("[Route] Kein Rest nach Reserve."); return; }
268  
269              double toWallet = rest*policy.wWallet;
270              double toOrbit  = rest*policy.wOrbit;
271              double toTech   = rest*policy.wTech;
272              double toAid    = rest*policy.wAid;
273  
274              if(toWallet>0){ wallet.gutschreiben(toWallet); silos.speichereInWallet(toWallet); }
275              if(toOrbit>0){ orbit.rueckspeisenOrbit(toOrbit); silos.speichereInCloud(toOrbit); }
276              if(toTech>0){ donations.capture("bank", Fund.RND_TECH,   Math.round(toTech*100), "EUR"); }
277              if(toAid>0){ donations.capture("bank", Fund.HUMANITAER, Math.round(toAid*100), "EUR"); }
278  
279              System.out.println(String.format(Locale.US,"[Route] verteilt → Wallet=%.2f, Orbit=%.2f, Tech=%.2f, Aid=%.2f", toWallet,toOrbit,toTech,toAid));
280          }

322      static final class Regulator {
323          final Set<String> whitelist=new HashSet<>(List.of("Wheat","Corn"));
324          final Map<String,Double> floor = Map.of("Wheat",0.55,"Corn",0.40);
325          final int dailyKgLimit=4000;
326          final AtomicLong soldTodayKg=new AtomicLong(0);
327          LocalDate day=LocalDate.now();
328          boolean kycOk(double reputation){ return reputation>=0.5; }
329          synchronized boolean allowSell(String product,int kg,double price){
330              if(!LocalDate.now().equals(day)){ day=LocalDate.now(); soldTodayKg.set(0); }
331              if(!whitelist.contains(product)) return false;
332              if(price < floor.getOrDefault(product, 0.0)) return false;
333              if(soldTodayKg.get()+kg > dailyKgLimit) return false;
334              soldTodayKg.addAndGet(kg); return true;
335          }
336      }
337      static final class SocialEquity {
338          Tier classify(double incomeIdx){ if(incomeIdx>=0.75) return Tier.WEALTHY; if(incomeIdx<=0.35) return Tier.NEEDY; return Tier.STANDARD; }
339          double buyerPrice(Tier t,double p){ return switch(t){ case NEEDY->Math.max(0,p*0.9); case WEALTHY->p*1.2; default->p; }; }

341      static final class Marketplace {
342          final Map<Long,Offer> offers=new LinkedHashMap<>();
343          final Map<Long,Integer> demand=new HashMap<>();
344          final Random r=new Random();
345          void add(Offer o){ offers.put(o.id,o); }
346          double demandIndex(long id){ int d=demand.getOrDefault(id,0)+1; demand.put(id,d); return 1.0+Math.min(1.2, d*0.04); }
347          /** Verkauf inkl. CRM/KYC/Regulator – gibt Revenue zurück. */
348          double sell(Offer o, int kg, Customer buyer, SocialEquity eq, Regulator reg, boolean marketing, Orders orders){
349              if(o==null || kg<=0 || o.stockKg<=0) return 0;
350              int sell=Math.min(kg,o.stockKg);
351              if(!reg.kycOk(buyer.reputation)){ LOG.info("[KYC] abgelehnt für "+buyer.id); return 0; }
352  
353              double dyn=o.base*demandIndex(o.id);
354              if(marketing) dyn*=1.05;
355              Tier tier = eq.classify(buyer.incomeIdx);
356              double ppk = clamp(eq.buyerPrice(tier, dyn), o.base*0.7, o.base*1.9);
357  
358              if(!reg.allowSell(o.name, sell, ppk)){ LOG.info("[REG] Verkauf blockiert"); return 0; }
359  
360              double revenue = sell*ppk;
361              o.stockKg -= sell;
362              buyer.loyalty += Math.max(1, (int)(revenue/50.0));
363              String oid="ORD-"+UUID.randomUUID().toString().substring(0,8);
364              orders.record(oid, buyer.id, o.name, sell, ppk, revenue);
365              LOG.info(String.format(Locale.US,"[SALE] %s %dkg @%.2f → %.2f (tier=%s, loyalty=%d, stockLeft=%d)",
366                      o.name, sell, ppk, revenue, tier, buyer.loyalty, o.stockKg));
367              return revenue;
368          }

392      static final class Sensors {

394          List<SensorData> read(){

404      static final class IrrigationPlan { final int liters; IrrigationPlan(int l){liters=l;} }
405      static final class EnergyPlan { final double demand; EnergyPlan(double d){demand=d;} }
406      static final class Decision { final IrrigationPlan ip; final EnergyPlan ep; Decision(IrrigationPlan i,EnergyPlan e){ip=i;ep=e;} }
407  
408      static final class KI {
409          Decision decide(List<SensorData> d, Silos s, boolean koppel){
410              double soil=v(d,"soil_moisture"), sun=v(d,"sun_intensity"), tank=v(d,"cistern_liters"), rain=v(d,"rain_forecast");
411              double need=0; if(soil<28 && rain<2) need=600; else if(soil<35 && rain<3) need=320;
412              if(tank<800) need*=0.5; if(sun>0.8) need*=1.1; if(!koppel) need*=1.06;
413              IrrigationPlan ip=(need>0)? new IrrigationPlan((int)Math.round(need)) : null;
414              double demand=40+(sun*30); if(!koppel) demand*=0.95;
415              return new Decision(ip, new EnergyPlan(demand));
416          }
417          double v(List<SensorData> d,String key){ return d.stream().filter(x->x.type.equals(key)).mapToDouble(x->x.value).findFirst().orElse(0); }
418      }
419      static final class IrrigationSystem {
420          final Silos s; IrrigationSystem(Silos s){this.s=s;}
421          void execute(IrrigationPlan p){ if(p==null) return; if(s.waterLiters>=p.liters){ s.waterLiters-=p.liters; LOG.info("Bewässerung "+p.liters+"L"); } else LOG.warning("Zisterne leer"); }
422      }
423      static final class SolarSystem {
424          final Silos s; SolarSystem(Silos s){this.s=s;}
425          void execute(EnergyPlan e, boolean koppel){ double cover=koppel?14:10; if(s.energyUnits < e.demand-cover) LOG.warning("Energie knapp"); s.addEnergy(koppel?11:8); }
426      }

594              }catch(Exception e){ LOG.log(Level.WARNING,"entrepreneur loop error", e); }

600              System.out.println("\nShutdown ok.");
601          }));
602          Thread.currentThread().join();
603      }
